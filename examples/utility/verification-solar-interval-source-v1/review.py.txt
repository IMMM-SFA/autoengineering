"""Pin public provider documentation and recheck saved site metadata only."""
from pathlib import Path
from datetime import datetime,timezone
import hashlib,json,urllib.request
root=Path.cwd();out=root/'examples/utility/verification-solar-interval-source-v1'
class NoRedirect(urllib.request.HTTPRedirectHandler):
    def redirect_request(self,*a,**kw):raise ValueError('documentation redirects are not allowed')
opener=urllib.request.build_opener(NoRedirect,urllib.request.ProxyHandler({}))
requests=[]
def get(url,maximum):
    request=urllib.request.Request(url,headers={'User-Agent':'Autoengineering-metadata-review/1'})
    with opener.open(request,timeout=10) as response:
        raw=response.read(maximum+1)
        assert len(raw)<=maximum and response.status==200
    requests.append({'url':url,'bytes':len(raw),'maximum_bytes':maximum,'sha256':hashlib.sha256(raw).hexdigest()})
    return raw
api='https://api.github.com/repos/openEDI/documentation/commits?path=pvdaq.md&per_page=1'
head=json.loads(get(api,65536))[0]
commit=head['sha'];assert len(commit)==40 and all(c in '0123456789abcdef' for c in commit)
url=f'https://raw.githubusercontent.com/openEDI/documentation/{commit}/pvdaq.md'
raw=get(url,65536);text=raw.decode()
excerpt='Is the data left(L), center(C), or right(R) aligned during the acquisition interval'
assert excerpt in text
metadata_path=root/'examples/utility/proposals/confirmation-v2/metadata-refresh/2107_system_metadata.json'
metadata_raw=metadata_path.read_bytes()
assert hashlib.sha256(metadata_raw).hexdigest()=='17e4f7f448fcb2bec43f7610b40c1f7f4fc794ccc185e41dcde0514afba73ae5'
m=json.loads(metadata_raw)
ac=m['Metrics']['inv_01_ac_power_inv_149583']
intervals={section:{key:value['time_interval'] for key,value in m[section].items() if 'time_interval' in value}
           for section in ('Inverters','Meters','Other Instruments')}
assert all(v=='L' for group in intervals.values() for v in group.values())
assert ac['source_id'] is None and ac['aggregation_type']=='avg' and ac['source_type']=='INVERTER'
result={'scope':'Provider interval dictionary and cached hardware metadata only',
 'reviewed_at_utc':datetime.now(timezone.utc).isoformat(),'provider_document_commit':commit,
 'provider_document_url':f'https://github.com/openEDI/documentation/blob/{commit}/pvdaq.md',
 'provider_document_sha256':hashlib.sha256(raw).hexdigest(),'provider_document_bytes':len(raw),
 'provider_excerpt':excerpt,'documented_interval_codes':{'L':'left','C':'center','R':'right'},
 'documentation_requests':requests,'saved_site_metadata_sha256':hashlib.sha256(metadata_raw).hexdigest(),
 'saved_site_metadata_path':str(metadata_path.relative_to(root)),
 'observed_interval_code_counts':{s:len(v) for s,v in intervals.items()},
 'observed_interval_codes_unique':sorted({v for group in intervals.values() for v in group.values()}),
 'selected_ac_metric_id':ac['metric_id'],'selected_ac_source_type':ac['source_type'],
 'selected_ac_source_id':None,'selected_ac_aggregation':ac['aggregation_type'],
 'system_reported_resolution':m['System']['data_time_resolution'],
 'system_timezone_code':m['System']['timezone_code'],
 'resolved':'Generic L/C/R dictionary meaning',
 'unresolved':['Explicit selected-AC-to-inverter join','Sample support duration and timestamp/DST treatment',
               'Availability delays, quality masks, equipment topology, rights and exposure'],
 'inference_limit':'Equal device interval codes do not establish the missing channel join or its clock/support contract',
 'observation_files_requested':False,'new_private_labels_read':False,'model_calls':0,
 'task_admitted':False,'proposal_or_frozen_metadata_rewritten':False}
(out/'source-review.json').write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps({'provider_commit':commit,'provider_document_sha256':result['provider_document_sha256'],
                 'interval_code_counts':result['observed_interval_code_counts'],'interval_codes':result['observed_interval_codes_unique'],
                 'observation_files_requested':False,'task_admitted':False}))
