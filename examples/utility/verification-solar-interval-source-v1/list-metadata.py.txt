"""List only the selected site's root and metadata prefixes, never data objects."""
from pathlib import Path
import hashlib,json,urllib.request,urllib.parse,xml.etree.ElementTree as E
out=Path('examples/utility/verification-solar-interval-source-v1')
class NoRedirect(urllib.request.HTTPRedirectHandler):
    def redirect_request(self,*a,**kw):raise ValueError('redirect denied')
opener=urllib.request.build_opener(NoRedirect,urllib.request.ProxyHandler({}))
base='https://oedi-data-lake.s3.amazonaws.com/'
root='pvdaq/2023-solar-data-prize/2107_OEDI/'
results=[]
for index,prefix in enumerate((root,root+'metadata/')):
    url=base+'?'+urllib.parse.urlencode({'list-type':'2','prefix':prefix,'delimiter':'/','max-keys':'50'})
    request=urllib.request.Request(url,headers={'User-Agent':'Autoengineering-metadata-review/1'})
    try:
        with opener.open(request,timeout=10) as response:
            raw=response.read(131073);assert len(raw)<=131072 and response.status==200
        xml=E.fromstring(raw);ns={'s':'http://s3.amazonaws.com/doc/2006-03-01/'}
        files=[{'key':node.findtext('s:Key',namespaces=ns),'bytes':int(node.findtext('s:Size',namespaces=ns))}
               for node in xml.findall('s:Contents',ns)]
        directories=[node.findtext('s:Prefix',namespaces=ns) for node in xml.findall('s:CommonPrefixes',ns)]
        result={'url':url,'prefix':prefix,'status':'listed','response_sha256':hashlib.sha256(raw).hexdigest(),
                'response_bytes':len(raw),'truncated':xml.findtext('s:IsTruncated',namespaces=ns)=='true',
                'files':files,'directories':directories,'listed_object_bodies_requested':False}
        (out/f'listing-{index}.xml.txt').write_bytes(raw)
    except Exception as error:
        result={'url':url,'prefix':prefix,'status':'unavailable','error_type':type(error).__name__,
                'automatic_retry':False,'listed_object_bodies_requested':False}
    results.append(result)
(out/'metadata-listings.json').write_text(json.dumps({'maximum_requests':2,'results':results,
 'observation_objects_requested':False,'task_admitted':False},indent=2)+'\n')
print(json.dumps(results))
