from autoengineering.campaign.serialization import digest, unique_json

metadata = daemon_fixture.metadata
H = daemon_fixture.H


class Peer:
    def __init__(self, endpoint):
        self.socket = socket.socket(socket.AF_UNIX, socket.SOCK_STREAM)
        self.socket.bind(endpoint)
        self.socket.listen(16)
        self.socket.settimeout(0.1)
        self.stop = threading.Event()
        self.lock = threading.Lock()
        self.workers = []
        self.errors = []
        self.calls = []
        self.daemon = None
        self.response = b""
        self.packet = b""
        self.received = b""
        self.attached = False
        self.starts = 0
        self.inputs = 0
        self.problem = None
        self.exit_code = 0
        self.thread = threading.Thread(target=self.serve, daemon=True)
        self.thread.start()

    def serve(self):
        while not self.stop.is_set():
            try:
                stream, _ = self.socket.accept()
            except socket.timeout:
                continue
            except OSError:
                break
            worker = threading.Thread(target=self.handle, args=(stream,), daemon=True)
            self.workers.append(worker)
            worker.start()

    def handle(self, stream):
        try:
            with stream:
                stream.settimeout(0.1)
                raw = bytearray()
                while b"\r\n\r\n" not in raw:
                    if self.stop.is_set():
                        return
                    try:
                        chunk = stream.recv(4096)
                    except socket.timeout:
                        continue
                    if not chunk:
                        return
                    raw.extend(chunk)
                    assert len(raw) <= 65536
                header, body = bytes(raw).split(b"\r\n\r\n", 1)
                lines = header.decode().split("\r\n")
                method, path, _ = lines[0].split(" ")
                fields = dict(line.split(": ", 1) for line in lines[1:])
                size = int(fields.get("Content-Length", "0"))
                while len(body) < size:
                    body += stream.recv(size - len(body))
                assert len(body) == size
                path = path.removeprefix("/v1.48")
                with self.lock:
                    self.calls.append((method, path))
                if "/attach?" in path:
                    self.attached = True
                    stream.sendall(
                        b"HTTP/1.1 101 Switching Protocols\r\nConnection: Upgrade\r\n"
                        b"Upgrade: tcp\r\nContent-Type: application/vnd.docker.raw-stream\r\n\r\n"
                    )
                    data = bytearray()
                    while not self.stop.is_set():
                        try:
                            chunk = stream.recv(65536)
                        except socket.timeout:
                            continue
                        if not chunk:
                            break
                        data.extend(chunk)
                        assert len(data) <= 40 * 1024 * 1024 + 4 * 1024 * 1024
                    self.received = bytes(data)
                    if not data:
                        return  # Owner failed before its one stdin transfer.
                    if self.received != self.packet:
                        assert self.packet.startswith(self.received)
                        return  # Interrupted input cannot produce a result.
                    self.inputs += 1
                    if self.problem == "idle":
                        self.stop.wait(timeout=30)
                        return
                    with self.lock:
                        self.daemon.created["State"].update(
                            Status="exited",
                            Running=False,
                            Pid=0,
                            ExitCode=self.exit_code,
                            FinishedAt="2026-01-01T00:00:02Z",
                        )
                    output = self.response[:-1] if self.problem == "truncated" else self.response
                    diagnostic = b"AUTHORED-PRIVATE-CANARY!"
                    stream.sendall(struct.pack("!B3xI", 2, len(diagnostic)) + diagnostic)
                    stream.sendall(struct.pack("!B3xI", 1, len(output)) + output)
                    return
                with self.lock:
                    if path == f"/containers/{H}/start":
                        assert method == "POST" and self.attached
                        state = self.daemon.owner.replay()
                        assert state.phase == "start_intent" and state.events[2].dispatch
                        assert (
                            self.daemon.owner.root / "control/guest-watchdog/ready.json"
                        ).exists()
                        self.starts += 1
                        assert self.starts == 1
                        self.daemon.created["State"].update(
                            Status="running",
                            Running=True,
                            Pid=123,
                            StartedAt="2026-01-01T00:00:00Z",
                        )
                        if self.problem == "lost_start":
                            return
                        status, output = (304 if self.problem == "bad_start" else 204), b""
                    elif "/kill?" in path:
                        assert method == "POST" and path == f"/containers/{H}/kill?signal=SIGKILL"
                        self.daemon.created["State"].update(
                            Status="exited",
                            Running=False,
                            Pid=0,
                            ExitCode=137,
                            FinishedAt="2026-01-01T00:00:02Z",
                        )
                        status, output = 204, b""
                    elif method == "DELETE":
                        assert path == f"/containers/{H}?force=1&v=0"
                        self.daemon.exists = False
                        status, output = 204, b""
                    else:
                        status, output = self.daemon.request(
                            method, path, unique_json(body.decode()) if body else None
                        )
                stream.sendall(
                    f"HTTP/1.1 {status} Authored\r\nContent-Length: {len(output)}\r\n"
                    "Connection: close\r\n\r\n".encode()
                    + output
                )
        except (BrokenPipeError, ConnectionResetError):
            pass  # Expected when an interrupted owner closes its original stream.
        except BaseException as error:
            self.errors.append(error)

    def close(self):
        self.stop.set()
        self.socket.close()
        self.thread.join(timeout=3)
        for worker in self.workers:
            worker.join(timeout=3)
            assert not worker.is_alive()
        assert not self.thread.is_alive()
        assert not self.errors, self.errors


@contextmanager
def prepared(tmp_path, metadata):
    with tempfile.TemporaryDirectory(prefix="ae-dispatch-", dir="/tmp") as folder:
        endpoint = str(Path(folder) / "peer.sock")
        peer = Peer(endpoint)
        try:
            api = GuestDaemonAPI(
                endpoint, api_version="1.48", server_digest=digest(daemon_fixture.SERVER)
            )
            owner, store, source, drivers = daemon_fixture.input_fixture.prepared(
                tmp_path, metadata, policy=GuestStdinPolicy(), runtime_digest=api.identity
            )
            daemon_fixture.input_fixture.prepare(owner, store, source, drivers)
            peer.daemon = daemon_fixture.Daemon(owner)
            daemon_fixture.create(owner, store, api)
            peer.response, _ = bootstrap_fixture.response(tmp_path, owner)
            peer.packet = module.encode_request(owner)
            yield owner, store, api, peer
        finally:
            peer.close()


def run(owner, store, api, *, timeout=10, cancelled=None):
    with store.writer(), owner.writer():
        return module.GuestDispatchController(api).run(
            owner, store, deadline=time.monotonic() + timeout, cancelled=cancelled
        )


def test_single_start_attaches_first_and_replays_without_daemon(tmp_path, metadata, monkeypatch):
    with prepared(tmp_path, metadata) as (owner, store, api, peer):
        before = store.replay()
        result = run(owner, store, api)
        assert result.transport_status == "complete" and result.journal_phase == "cleaned"
        assert result.capture and result.watchdog.result == "absent"
        assert result.watchdog.reason == "owner_finished"
        assert not result.watchdog.kill_intended and result.watchdog.delete_intended
        assert peer.starts == peer.inputs == 1
        assert peer.received == peer.packet
        assert b"AUTHORED-PRIVATE-CANARY" not in peer.packet
        state = owner.replay()
        assert state.events[2].dispatch and state.input_package_digest
        assert state.observations["stopped"]["elapsed_seconds"] > 0
        assert store.replay() == before
        calls = deepcopy(peer.calls)
        with pytest.raises(ValueError, match="unstarted"):
            run(owner, store, api)
        assert peer.calls == calls
        relocated = tmp_path / "relocated"
        shutil.copytree(owner.root, relocated)
        cold = GuestOwnerJournal(relocated)

        def forbidden(*args, **kwargs):
            raise AssertionError("cold recovery must not execute, connect or resample")

        monkeypatch.setattr(module.time, "monotonic", forbidden)
        monkeypatch.setattr(module, "attach", forbidden)
        assert module.read_completion(cold) == result
        assert peer.calls == calls
        monkeypatch.undo()  # Restore the clock before the native peer joins.


@pytest.mark.parametrize("problem", ["lost_start", "bad_start", "truncated"])
def test_failed_dispatch_never_restarts_and_reconciles_original_stop(tmp_path, metadata, problem):
    with prepared(tmp_path, metadata) as (owner, store, api, peer):
        peer.problem = problem
        with pytest.raises(ValueError):
            run(owner, store, api)
        assert peer.starts == 1
        calls = deepcopy(peer.calls)
        with owner.writer():
            result = module.reconcile_dispatch(owner)
        assert result.transport_status == "missing" and result.journal_phase == "cleaned"
        assert result.watchdog.result == "absent"
        assert owner.replay().observations["stopped"]["elapsed_seconds"] is None
        assert peer.calls == calls
        assert peer.inputs == (1 if problem == "truncated" else 0)
        assert module.read_completion(owner) == result
        with pytest.raises(ValueError):
            run(owner, store, api)
        assert peer.calls == calls


@pytest.mark.parametrize(
    "point",
    [
        "before_guest_attach",
        "after_guest_attach",
        "before_start_intent",
        "after_start_intent",
        "before_daemon_start",
        "after_daemon_start",
        "start_response_saved",
        "before_started",
        "after_started",
        "before_guest_input",
        "after_guest_input",
        "guest_capture_saved",
        "before_stopped",
        "after_stopped",
        "before_cleanup_intent",
        "after_cleanup_intent",
        "before_cleaned",
        "after_cleaned",
        "before_dispatch_completion",
        "after_dispatch_completion",
    ],
)
def test_interruption_recovers_saved_observations_without_new_dispatch(tmp_path, metadata, point):
    with prepared(tmp_path, metadata) as (owner, store, api, peer):
        before = store.replay()

        def fail(stage):
            if stage == point:
                raise RuntimeError("authored dispatch interruption")

        owner.fault = fail
        with pytest.raises(RuntimeError, match="authored dispatch interruption"):
            run(owner, store, api)
        calls = deepcopy(peer.calls)
        reopened = GuestOwnerJournal(owner.root, suite_root=store.root)
        with reopened.writer():
            recovered = module.reconcile_dispatch(reopened)
        state = reopened.replay()
        if point in ("before_guest_attach", "after_guest_attach", "before_start_intent"):
            assert state.phase == "created" and peer.starts == 0
        elif point in ("after_start_intent", "before_daemon_start"):
            assert state.phase == "start_intent" and peer.starts == 0
        else:
            assert state.phase == "cleaned" and peer.starts == 1
            assert module.read_completion(reopened) == recovered
        assert store.replay() == before
        assert recovered.watchdog.result == "absent"
        assert peer.calls == calls
        with pytest.raises((ValueError, FileExistsError)):
            run(reopened, store, api)
        assert peer.calls == calls  # Neither attach nor START may be retried.


def test_appended_suite_cancellation_is_seen_during_transport(tmp_path, metadata):
    with prepared(tmp_path, metadata) as (owner, store, api, peer):
        sent = False

        def cancel_after_input():
            nonlocal sent
            if peer.inputs and not sent:
                sent = True
                grant = owner.manifest.job.grant
                store.append(
                    request_id="cancel-during-dispatch",
                    operation="cancel",
                    cell_digest=grant.cell_digest,
                    attempt_id="dispatch-cancel",
                    child_digest=grant.digest,
                    phase="confirmation",
                )
            return False  # The suite event itself must stop transport.

        with pytest.raises(ValueError):
            run(owner, store, api, cancelled=cancel_after_input)
        assert sent and store.replay()["cancelled"]
        calls = deepcopy(peer.calls)
        with owner.writer():
            recovered = module.reconcile_dispatch(owner)
        assert recovered.transport_status != "complete"
        assert recovered.watchdog.reason == "cancelled" and recovered.watchdog.result == "absent"
        assert owner.replay().observations["stopped"]["reason"] == "cancelled"
        assert peer.calls == calls and peer.starts == peer.inputs == 1


def test_original_deadline_stops_idle_output_without_reattachment(tmp_path, metadata):
    with prepared(tmp_path, metadata) as (owner, store, api, peer):
        peer.problem = "idle"
        with pytest.raises(ValueError):
            run(owner, store, api)
        calls = deepcopy(peer.calls)
        with owner.writer():
            recovered = module.reconcile_dispatch(owner)
        assert recovered.watchdog.reason == "deadline" and recovered.watchdog.result == "absent"
        assert recovered.transport_status == "missing"
        assert owner.replay().observations["stopped"]["reason"] == "timeout"
        assert peer.calls == calls and peer.starts == peer.inputs == 1


def test_cancel_after_observed_stop_does_not_credit_transport(tmp_path, metadata):
    with prepared(tmp_path, metadata) as (owner, store, api, peer):
        stopped = False

        def observe(stage):
            nonlocal stopped
            if stage == "after_stopped":
                stopped = True

        owner.fault = observe
        with pytest.raises(ValueError):
            run(owner, store, api, cancelled=lambda: stopped)
        calls = deepcopy(peer.calls)
        reopened = GuestOwnerJournal(owner.root, suite_root=store.root)
        original = reopened.replay().observations["stopped"]
        assert original["reason"] is None and original["elapsed_seconds"] > 0
        with reopened.writer():
            recovered = module.reconcile_dispatch(reopened)
        assert recovered.transport_status == "unverified" and recovered.capture
        assert recovered.watchdog.reason == "cancelled" and recovered.watchdog.result == "absent"
        assert reopened.replay().observations["stopped"] == original  # Never relabel an old event.
        assert peer.calls == calls and peer.starts == peer.inputs == 1


def test_integrated_watchdog_survives_actual_dispatch_owner_sigkill(tmp_path, metadata):
    with prepared(tmp_path, metadata) as (owner, store, api, peer):
        peer.problem = "idle"
        marker = tmp_path / "helper-input-sent"
        script = """
from pathlib import Path
import signal, sys, time
from autoengineering.campaign.serialization import digest
from autoengineering.utility.guest_daemon import GuestDaemonAPI
from autoengineering.utility.guest_dispatch import GuestDispatchController
from autoengineering.utility.guest_owner import GuestOwnerJournal
from autoengineering.utility.store import SuiteStore
suite = SuiteStore(Path(sys.argv[1]))
owner = GuestOwnerJournal(Path(sys.argv[2]), suite_root=suite.root)
server = {"ApiVersion":"1.48","Os":"linux","Arch":"amd64","Version":"authored"}
api = GuestDaemonAPI(sys.argv[3], api_version="1.48", server_digest=digest(server))
def interrupt(stage):
    if stage == "after_guest_input":
        Path(sys.argv[4]).write_text("authored-input-sent")
        signal.pause()
owner.fault = interrupt
with suite.writer(), owner.writer():
    GuestDispatchController(api).run(owner, suite, deadline=time.monotonic() + 20)
"""
        helper = subprocess.Popen(
            [
                sys.executable,
                "-I",
                "-c",
                script,
                str(store.root),
                str(owner.root),
                api._socket_path,
                str(marker),
            ],
            stdin=subprocess.DEVNULL,
            stdout=subprocess.DEVNULL,
            stderr=subprocess.DEVNULL,
            start_new_session=True,
        )
        guard_root = owner.root / "control/guest-watchdog"
        try:
            end = time.monotonic() + 12
            while not marker.exists() and helper.poll() is None and time.monotonic() < end:
                time.sleep(0.05)
            assert marker.exists(), helper.poll()
            assert peer.starts == 1
            helper.kill()  # Signal only this exact owned child, not an inferred PID.
            assert helper.wait(timeout=5) < 0
            end = time.monotonic() + 15
            while not (guard_root / "outcome.json").exists() and time.monotonic() < end:
                time.sleep(0.05)
            calls = deepcopy(peer.calls)
            with owner.writer():
                recovered = module.reconcile_dispatch(owner)
            assert recovered.watchdog.reason == "owner_lost"
            assert recovered.watchdog.result == "absent" and recovered.watchdog.kill_intended
            assert recovered.journal_phase == "cleaned" and recovered.transport_status == "missing"
            assert owner.replay().observations["stopped"]["elapsed_seconds"] is None
            assert peer.calls == calls and peer.starts == 1
            ready = module.GuestWatchdogReady.from_json((guard_root / "ready.json").read_text())
            end = time.monotonic() + 3
            while time.monotonic() < end:
                status = subprocess.run(
                    ["ps", "-p", str(ready.watchdog_pid), "-o", "stat="],
                    capture_output=True,
                    text=True,
                    timeout=2,
                )
                if status.returncode == 1 or status.stdout.strip().startswith("Z"):
                    break
                time.sleep(0.05)
            else:
                pytest.fail("original detached guard did not terminate after cleanup")
        finally:
            if helper.poll() is None:
                helper.kill()
            helper.wait(timeout=5)
