"""Relocated authored dispatch replay with model/process/I/O guards."""
import importlib.abc
from pathlib import Path
import os
import socket
import subprocess
import sys
import tarfile
import tempfile
import time

class NoModels(importlib.abc.MetaPathFinder):
    def find_spec(self, fullname, *args):
        if fullname.split('.')[0] in {'examples', 'torch', 'openmdao', 'pvlib', 'py_wake'}:
            raise AssertionError('cold replay cannot import a model')
sys.meta_path.insert(0, NoModels())
import numpy as np
from autoengineering.utility.guest_owner import GuestOwnerJournal
from autoengineering.utility.guest_dispatch import GuestDispatchCompletion, read_completion
bundle = Path(__file__).parent
cases = {'complete': ('authored-dispatched-owner.tar.gz', 'expected-completion.json'),
         'owner-loss': ('authored-dispatch-owner-loss.tar.gz', 'expected-owner-loss.json')}
archive_name, expected_name = cases[sys.argv[1] if len(sys.argv) == 2 else 'complete']
expected = GuestDispatchCompletion.from_json((bundle / expected_name).read_text())
with tempfile.TemporaryDirectory() as directory:
    destination = Path(directory).resolve()
    with tarfile.open(bundle / archive_name) as archive:
        archive.extractall(destination, filter='data')
    for path in destination.rglob('*'):
        path.chmod(0o700 if path.is_dir() else 0o600)
    owner = GuestOwnerJournal(destination / 'owner')
    def forbidden(*args, **kwargs):
        raise AssertionError('cold replay cannot execute, connect, load arrays or sample time')
    subprocess.Popen = socket.socket = time.monotonic = time.time = np.load = forbidden
    active = [True]
    def only_saved_reads(event, args):
        if not active[0]:
            return
        if event != 'open' or isinstance(args[0], int):
            return
        path = Path(args[0]).resolve()
        mode, flags = args[1:3]
        if not path.is_relative_to(destination):
            raise AssertionError('cold replay tried to read outside the relocated archive')
        if (mode is not None and any(c in mode for c in 'wax+')) or flags & (
            os.O_WRONLY | os.O_RDWR | os.O_CREAT | os.O_TRUNC | os.O_APPEND
        ):
            raise AssertionError('cold replay tried to write')
    sys.addaudithook(only_saved_reads)
    assert read_completion(owner) == expected
    print('Original dispatch completion agrees from relocated saved bytes only.')
    active[0] = False  # The read-only audit ends before temporary-directory cleanup.
