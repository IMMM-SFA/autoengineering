"""Eight fixed diagnostics on the coupled image, in a private policy namespace.

This does not edit the production image catalogue or admit a scientific runtime.
"""
import hashlib,json,os,re,uuid
from pathlib import Path
from autoengineering.campaign.serialization import canonical_json,digest
from autoengineering.utility.guest_daemon import GuestDaemonAPI
from examples.utility import guest_resources as policy,guest_resource_probe as probe
from examples.utility import guest_image_build as build,engineering_container_runtime as owner

root=Path.cwd();os.umask(0o077)
p=root/'examples/utility/verification-engineering-image-resources-v1'
work=root/'outputs/engineering-image-resources-v1';assert not work.exists();work.mkdir(mode=0o700)
source=Path(policy.__file__).read_text();support=Path(probe.__file__).read_text()
namespace={'__name__':'fixed_coupled_resource_policy','__file__':str(Path(policy.__file__))}
exec(compile(source,'archived-resource-policy','exec'),namespace)
def fixed_labels(image_id):
    if image_id!=owner.IMAGE:raise ValueError('diagnostic image differs')
    return dict(owner.LABELS)
namespace['image_labels']=fixed_labels
original_arguments=namespace['_arguments']
extra=('output_small','output_overrun')
def arguments(mode,nonce,canary):
    if mode in extra:
        if re.fullmatch('[0-9a-f]{32}',nonce) is None or canary is not None:raise ValueError('output probe args differ')
    else:original_arguments(mode,nonce,canary)
namespace['_arguments']=arguments
cases=[]
for mode in (*probe.MODES,*extra):
    nonce=uuid.uuid4().hex
    canary=str(work/('ae-host-canary-'+nonce)) if mode=='host_canary_denied' else None
    if canary:Path(canary).write_text('fresh authored public canary, no scientific data\n')
    body=support
    if mode in extra:
        body='import sys,json,hashlib,signal\nfrom pathlib import Path\n'
        body+='scope={"__name__":"fixed_context"}\nexec('+repr(support)+',scope)\n'
        body+='signal.alarm(45)\nobserved=scope["context"]()\n'
        body+='worker_path=Path("/opt/coupled-source/examples/utility/engineering_load_worker.py")\nraw=worker_path.read_bytes()\n'
        body+='assert hashlib.sha256(raw).hexdigest()=='+repr(owner.WORKER_SHA)+'\n'
        body+='worker={"__name__":"fixed_public_output_probe","__file__":str(worker_path)}\nexec(compile(raw,"fixed-public-worker","exec"),worker)\n'
        body+='capture=worker["capture"](b"",expected_sha256="1"*64,source_root=Path("/opt/coupled-source"),private_work=Path("/work"),mode='+repr('output-small' if mode=='output_small' else 'output-overrun')+')\n'
        body+='print(json.dumps({"schema":"utility-coupled-output-control-v1","mode":sys.argv[1],"nonce":sys.argv[2],"observed":observed,"capture":capture},sort_keys=True,separators=(",",":"),allow_nan=False),flush=True)\n'
    config=namespace['_configuration'](mode,nonce,body,canary=canary,image_id=owner.IMAGE)
    case={'mode':mode,'nonce':nonce,'name':'ae-guest-resource-'+nonce,'canary':canary,
          'body_sha256':hashlib.sha256(body.encode()).hexdigest(),'configuration':config}
    cases.append(case)
old=json.loads((root/'examples/utility/verification-engineering-linux-worker-v1/request.json').read_bytes())
api=GuestDaemonAPI('/Users/cam/.orbstack/run/docker.sock',api_version=old['api_version'],server_digest=old['server_digest'])
request={'schema':'utility-coupled-image-resource-request-v1','image_id':owner.IMAGE,'image_labels':dict(owner.LABELS),
    'cases':cases,'daemon':api.profile.to_dict(),'cli':'/usr/local/bin/docker','endpoint':'/Users/cam/.orbstack/run/docker.sock',
    'binary':build._binary('/usr/local/bin/docker'),'endpoint_identity':build._endpoint('/Users/cam/.orbstack/run/docker.sock'),
    'sources':{name:hashlib.sha256((root/name).read_bytes()).hexdigest() for name in (
        'examples/utility/guest_resources.py','examples/utility/guest_resource_probe.py','examples/utility/guest_image_build.py',
        'examples/utility/engineering_container_runtime.py','examples/utility/engineering_container_watchdog.py')},
    'namespace_overrides':['image_labels for the exact coupled image','two fixed public output mode argument checks'],
    'max_creations':8,'max_starts':8,'active_seconds_per_case':60,'cleanup_seconds_per_case':10,
    'max_output_bytes':16384,'scientific_starts':0,'private_source_reads':0,'automatic_retries':0,
    'runtime_qualified':False,'private_boundary_qualified':False}
for target in (p,work):(target/'request.json').write_text(json.dumps(request,indent=2)+'\n')
(p/'policy-source.py.txt').write_text(source);(p/'probe-source.py.txt').write_text(support)
print(canonical_json({'request_digest':digest(request),'cases':8,'image_id':owner.IMAGE,'private_source_reads':0}))
