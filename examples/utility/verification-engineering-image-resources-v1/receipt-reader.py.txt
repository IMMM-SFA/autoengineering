"""Replay exact-image diagnostic receipts using only public files and stdlib."""
from pathlib import Path
import hashlib,json,sys
root=Path(sys.argv[1]).resolve(strict=True)
raw=(root/'evidence-manifest.json').read_bytes();assert hashlib.sha256(raw).hexdigest()==sys.argv[2]
manifest=json.loads(raw)
for name,expected in manifest['files'].items():
    rel=Path(name);assert not rel.is_absolute() and '..' not in rel.parts
    path=root/rel;assert not any(p.is_symlink() for p in (path,*path.parents))
    data=path.read_bytes();assert len(data)==expected['bytes'] and hashlib.sha256(data).hexdigest()==expected['sha256']
def read(name):
    assert name in manifest['files'];return json.loads((root/name).read_bytes())
def canonical(v):return json.dumps(v,sort_keys=True,separators=(',',':'),allow_nan=False).encode()
def digest(v):return hashlib.sha256(canonical(v)).hexdigest()
q=read('request.json');summary=read('summary.json');image=read('image.json')
assert digest(q)==summary['request_digest']
assert q['image_id']==summary['image_id']==image['Id']=='sha256:62f6c7b58e34f4ec5d5517af49c8b9e1e7d1d7235f550384ac52c1281fb262ad'
assert image['Config']['Labels']==q['image_labels']
modes=['memory_limit','pid_limit','workspace_limit','read_only_root','network_denied','host_canary_denied','output_small','output_overrun']
assert [c['mode'] for c in q['cases']]==modes
assert len({c['nonce'] for c in q['cases']})==8
expected_observed={'effective_limits':{'memory.max':'1073741824','memory.swap.max':'0','pids.max':'16','cpu.max':'100000 100000'},
    'workspace_bytes':134217728,'root_read_only':True,'network_interfaces':['lo'],'uid':1000,'gid':1000,'seccomp':2,'capabilities':'none'}
for case, recorded in zip(q['cases'],summary['cases'],strict=True):
    mode=case['mode'];prefix=mode+'/'
    cfg=case['configuration'];body=cfg['Entrypoint'][-1]
    assert hashlib.sha256(body.encode()).hexdigest()==case['body_sha256']
    cid=read(prefix+'created.json')['Id'];assert cid==recorded['container_id']
    assert read(prefix+'create-intent.json')['configuration_digest']==digest(cfg)
    assert read(prefix+'start-intent.json')['container_id']==cid
    assert cfg['Image']==q['image_id'] and cfg['Cmd'][:2]==[mode,case['nonce']]
    for stage in ('before','terminal'):
        v=read(prefix+stage+'.json')
        assert (v['Id'],v['Image'],v['Name'])==(cid,q['image_id'],'/'+case['name'])
        for key,wanted in cfg.items():
            if key in ('HostConfig','Env','Volumes'):continue
            assert canonical(v['Config'].get(key))==canonical(wanted)
        assert sorted(v['Config']['Env'])==sorted(cfg['Env'])
        assert v['Config'].get('Volumes') in (None,{})
        host=v['HostConfig']
        for key,wanted in cfg['HostConfig'].items():
            actual=host.get(key)
            if key=='SecurityOpt' and actual==['no-new-privileges:true']:actual=['no-new-privileges']
            if wanted==[] and actual is None:actual=[]
            if key=='PortBindings' and actual is None:actual={}
            assert canonical(actual)==canonical(wanted),key
        assert host['ReadonlyRootfs'] is True and host['NetworkMode']=='none'
        assert host['Memory']==host['MemorySwap']==1073741824 and host['PidsLimit']==16
        assert host['NanoCpus']==1000000000 and host['Privileged'] is False
    terminal=read(prefix+'terminal.json')['State']
    assert terminal['Status']=='exited' and terminal['Running'] is False
    assert terminal['OOMKilled'] is (mode=='memory_limit')
    client=read(prefix+'client.json');output=(root/prefix/'output.log').read_bytes()
    assert client['output_bytes']==len(output)<=16384
    assert client['output_sha256']==hashlib.sha256(output).hexdigest()
    assert client['exit_code']==terminal['ExitCode']==(137 if mode=='memory_limit' else 0)
    cleanup=read(prefix+'cleanup.json')
    assert cleanup['container_id']==cid and cleanup['absent'] is True
    assert recorded['original_container_absent'] is True
    lines=output.decode().splitlines()
    if mode.startswith('output_'):
        assert len(lines)==1
        value=json.loads(lines[0]);capture=value['capture']
        assert value['schema']=='utility-coupled-output-control-v1' and value['observed']==expected_observed
        assert value['mode']==mode and value['nonce']==case['nonce']
        assert capture['schema']=='utility-coupled-load-output-v1' and capture['result'] is None
        assert capture['output_limit_bytes']==16384 and capture['private_boundary_qualified'] is False
        if mode=='output_small':
            assert capture['captured_bytes']==4096 and capture['child_exit_code']==0
            assert capture['output_limited'] is False and capture['child_signal'] is None
        else:
            assert capture['captured_bytes']==16384 and capture['child_signal']==25
            assert capture['child_exit_code'] is None and capture['output_limited'] is True
    else:
        assert len(lines)==(1 if mode=='memory_limit' else 2)
        ready=json.loads(lines[0])
        assert ready=={'schema':'utility-guest-resource-ready-v1','mode':mode,'nonce':case['nonce'],'observed':expected_observed}
        if mode=='memory_limit':
            assert recorded['observation']['result']=={'oom_killed':True,'exit_code':137}
        else:
            envelope=json.loads(lines[1]);result=envelope['result']
            assert envelope['mode']==mode and envelope['nonce']==case['nonce']
            assert result==recorded['observation']['result']
            if mode=='pid_limit':assert result['errno']=='EAGAIN' and result['pids_current']==16 and result['children']==15
            elif mode=='workspace_limit':assert result['errno']=='ENOSPC' and 0<result['written_bytes']<=134217728
            elif mode=='read_only_root':assert result=={'errno':'EROFS'}
            elif mode=='network_denied':assert result['errno'] in ('ENETUNREACH','EHOSTUNREACH') and result['attempts']==1
            else:assert result['errno'] in ('ENOENT','ENOTDIR')
assert summary['container_creations']==summary['container_starts']==8
assert summary['scientific_starts']==summary['private_source_reads']==0
assert summary['runtime_qualified'] is False and summary['private_boundary_qualified'] is False
print(json.dumps({'verified':True,'exact_image_resource_controls':6,'public_output_controls':2,
    'container_starts':8,'all_original_containers_absent':True,'runtime_qualified':False},sort_keys=True))
