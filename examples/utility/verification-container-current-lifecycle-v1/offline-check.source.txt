from pathlib import Path
import hashlib
import json
import shutil
import socket
import ssl
import subprocess
import sys
import zipfile

repository=Path.cwd();base=repository/'examples/utility/verification-container-current-lifecycle-v1'
sys.path.insert(0,str(repository/'scripts'))
from autoengineering.utility import private_container_runtime as module
from verify_utility_creation_owner_archive import verify as creation
from verify_utility_running_archive import verify as running

pins=json.loads((base/'core-source-hashes.json').read_bytes())
assert len(pins)==199
for name,digest in pins.items():assert hashlib.sha256(Path(name).read_bytes()).hexdigest()==digest
wheel=Path('/private/tmp/utility-recovery-chain-wheel-v1/autoengineering-0.2.0-py3-none-any.whl')
with zipfile.ZipFile(wheel) as built:
    for name in pins:assert built.read(str(Path(name).relative_to('src')))==Path(name).read_bytes()
relocated=Path('/private/tmp/utility-current-lifecycle-relocated-v1');relocated.mkdir()
archives=(('creation',creation,False),('running',running,False),('running/execution-timeout',running,True))
for name,reader,timeout_only in archives:
    target=relocated/name;target.mkdir(parents=True,exist_ok=True)
    shutil.copytree(base/name/'executions',target/'executions')
    shutil.copyfile(base/name/'execution-files.json',target/'execution-files.json')
def denied(*args,**kwargs):raise AssertionError('offline checker attempted network or execution')
socket.socket=denied;subprocess.Popen=denied;module.API.request=denied
reports={}
for name,reader,timeout_only in archives:
    target=relocated/name
    digest=hashlib.sha256((target/'execution-files.json').read_bytes()).hexdigest()
    reports[name]=reader(target,digest,execution_timeout_only=True) if timeout_only else reader(target,digest)
summary={'schema':'utility-current-lifecycle-offline-verification-v1','archives':reports,'core_sources_unchanged':199,'existing_wheel_core_files_match':199,'new_package_build':False,'relocated':True,'daemon_network_and_process_denied':True,'created_objects':11,'container_starts':5,'scoring_child_starts':None,'accepted_scientific_results':0,'replacement_attempts':0,'private_scoring_qualified':False}
(base/'offline-verification.json').write_text(json.dumps(summary,indent=2)+'\n')
print(json.dumps(summary))
