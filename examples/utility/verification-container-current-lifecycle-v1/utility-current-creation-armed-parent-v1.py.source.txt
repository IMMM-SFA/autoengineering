from pathlib import Path
import hashlib
import json
import signal
import subprocess
import sys
import time

from autoengineering.utility import private_container_runtime as module

repository=Path.cwd()
evidence=repository/'examples/utility/verification-container-current-lifecycle-v1/creation'
hashes=json.loads((evidence/'executed-source-hashes.json').read_bytes())
endpoint=Path('/Users/cam/.orbstack/run/docker.sock')
version=module.API(str(endpoint)).json('GET','/version')
server=module.sha(module.encode(version))
registered=module.FixedScorerRuntime(endpoint,server_sha256=server)
api_version=registered.preflight()
controls=[]
for kind,signum in (('cancel',signal.SIGTERM),('owner-loss',signal.SIGKILL)):
    name='after-arm-'+kind
    print('Running armed owner control: '+name,flush=True)
    root=Path('/private/tmp')/('utility-current-creation-owner-'+name+'-v1')
    root.mkdir(mode=0o700)
    work=root/'workspace';work.mkdir(mode=0o700)
    (work/'job.json').write_text('{"outcome":"rmse","units":"1","prediction":[0,1]}')
    (work/'labels.json').write_text('{"labels":[1,2],"mask":[true,true]}')
    settings={'endpoint':str(endpoint),'server_sha256':server,'runtime_identity':registered.identity,
              'api_version':api_version,'job_sha256':module.sha((work/'job.json').read_bytes()),
              'labels_sha256':module.sha((work/'labels.json').read_bytes()),
              'phase':'after-arm','signal':int(signum)}
    module.write(root/'control.json',settings)
    with (root/'creator.stdout').open('xb') as out,(root/'creator.stderr').open('xb') as err:
        child=subprocess.Popen([sys.executable,'-I','/private/tmp/utility-creation-owner-child-v2.py',
                                str(repository/'src'),str(root)],stdout=out,stderr=err,
                               env={'PATH':'/usr/bin:/bin','LANG':'C.UTF-8'})
        code=child.wait(timeout=30)
    assert code==-int(signum)
    directory=root/'container'
    plan=json.loads((directory/'plan.json').read_bytes())
    ready=json.loads((directory/'watchdog-ready.json').read_bytes())
    before=json.loads((directory/'before-start.json').read_bytes())
    module.verify_owned(before,plan)
    assert before['State']['Status']=='created' and before['State']['Running'] is False
    assert ready['container_id']==plan['container_id'] and ready['nonce']==plan['nonce']
    operations=[json.loads(line) for line in (root/'creator-requests.jsonl').read_bytes().splitlines()]
    assert len([op for op in operations if op['method']=='POST' and op['path'].startswith('/containers/create?')])==1
    assert len([op for op in operations if op['method']=='POST' and op['path'].endswith('/start')])==1
    # The start hook killed its caller before forwarding the recorded request.
    # The observer performs GET only. Automatic removal must come from watchdog EOF.
    api=module.API(str(endpoint),api_version)
    deadline=time.monotonic()+15
    while not (directory/'cleanup.json').exists():
        assert time.monotonic()<deadline,'original watchdog cleanup unresolved; do not restart'
        time.sleep(.05)
    receipt=json.loads((directory/'cleanup.json').read_bytes())
    cleanup_before=json.loads((directory/'cleanup-before.json').read_bytes())
    assert receipt['reason']=='owner_pipe_eof' and receipt['removed'] is True
    assert receipt['container_id']==plan['container_id'] and receipt['nonce']==plan['nonce']
    module.verify_owned(cleanup_before,plan)
    assert cleanup_before['State']['Status']=='created' and cleanup_before['State']['Pid']==0
    assert api.inspect(plan['container_id']) is None
    stage=directory/'stage';stage.chmod(0o755)
    for scope in (work,stage):
        for field in ('job.json','labels.json'): (scope/field).unlink()
    stage.chmod(0o555)
    record={'name':name,'root':str(root),'owner_pid':child.pid,'owner_exit':code,
            'phase':'after-arm','container_id':plan['container_id'],'nonce':plan['nonce'],
            'created_objects':1,'container_starts':0,'scoring_starts':0,
            'start_attempt_recorded_but_not_forwarded':True,'watchdog_pid':ready['watchdog_pid'],
            'cleanup_reason':receipt['reason'],'observer_delete_requests':0,
            'all_job_label_copies_deleted':True,'original_id_removed':True,'live_absence_confirmed':True}
    module.write(root/'control-result.json',record)
    controls.append(record)
    for path,digest in hashes.items():
        assert hashlib.sha256((repository/path).read_bytes()).hexdigest()==digest
    print(json.dumps(record,sort_keys=True),flush=True)
(evidence/'armed-controls.json').write_text(json.dumps({
    'schema':'synthetic-container-armed-owner-controls-v1','controls':controls,
    'created_objects':2,'container_starts':0,'scoring_starts':0,'private_scoring_qualified':False,
},indent=2)+'\n')
