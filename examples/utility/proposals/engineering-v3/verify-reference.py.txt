"""Read saved public development evidence without numerical model execution."""
from pathlib import Path
import builtins,hashlib,json,math,socket,subprocess,sys
original_import=builtins.__import__
def guarded_import(name,*args,**kwargs):
    if name.split('.')[0] in {'examples','openmdao','py_wake','pvlib','torch','botorch','tensorflow'}:
        raise AssertionError('model/example import denied')
    return original_import(name,*args,**kwargs)
def denied(*args,**kwargs):raise AssertionError('external execution denied')
builtins.__import__=guarded_import;socket.socket=denied;subprocess.Popen=denied
import autoengineering
from autoengineering.agents.batch import BatchStore
from autoengineering.campaign.serialization import digest,unique_json
from autoengineering.campaign.artifacts import read_bytes
from autoengineering.campaign.records import ArtifactRef
root=Path(sys.argv[1]).resolve() if len(sys.argv)>1 else Path(__file__).resolve().parent
package=Path(autoengineering.__file__).resolve().parent
initializing=True
def audit(event,args):
    if event.startswith(('socket.','subprocess.')):raise AssertionError('external execution denied')
    if event=='open':
        name,mode,flags=args
        if isinstance(name,(str,bytes)):
            path=Path(name).resolve()
            if not path.is_relative_to(root) and not (initializing and path.is_relative_to(package)):
                raise AssertionError('outside archived public metadata read denied')
            if mode and any(c in mode for c in 'wax+'):raise AssertionError('archive write denied')
sys.addaudithook(audit)
def load(path):return unique_json((root/path).read_text())
proposal=load('proposal.json');inputs=load('development-inputs.json');summary=load('evidence/summary.json');request=load('evidence/request.json')
assert digest(proposal)==summary['proposal_digest']==request['proposal_digest']
assert digest(inputs)==proposal['development_input_digest']==summary['input_digest']==request['input_digest']
assert digest(proposal['baseline']['model'])==summary['model_digest']==request['model_digest']
assert proposal['baseline']['source_current_A']==request['fixed_source_current_A']==.05
assert proposal['confirmation_seed_or_rows_generated'] is False and summary['confirmation_rows_generated'] is False
store=BatchStore(root/'evidence/batch');state=store.replay()
initializing=False
assert len(state.reservations)==len(state.observations)==24
assert state.finished['baseline'].status=='passed'
assert all(r.resource=='science' for r in state.reservations.values())
assert all(o.status=='success' and o.stop_verified for o in state.observations.values())
ref=ArtifactRef.from_dict(summary['receipt'])
assert ref in state.finished['baseline'].evidence
receipt=unique_json(read_bytes(store.root,ref).decode())
assert receipt['request_digest']==digest(request) and receipt['proposal_digest']==digest(proposal)
assert receipt['input_digest']==digest(inputs) and receipt['model_digest']==summary['model_digest']
assert len(receipt['case_results'])==24
by_case={case['scenario_id']:case for case in inputs['cases']}
assert len(by_case)==24
powers=[]
for row in receipt['case_results']:
    case=by_case[row['scenario_id']]
    assert row['forcing_digest']==digest(case['forcing']) and row['physical_and_reference_checks_passed']
    native=row['outputs'];reference=row['analytical_reference'];work=row['diagnostics']
    assert all(math.isfinite(value) for value in (*native.values(),*reference.values(),*work.values()))
    assert work['converged']==1 and work['residual_norm']<=1e-10 and work['residual_tolerance']==1e-10 and work['solver_iterations']<=100
    assert native['node2_voltage']>=.62
    power=.05*(native['node1_voltage']-case['forcing']['ground_voltage'])
    assert power==row['source_power_W'] and power>0
    voltage_error=max(abs(native[n]-reference[n]) for n in ('node1_voltage','node2_voltage'))
    power_error=abs(power-reference['source_power_W'])
    assert voltage_error==row['voltage_absolute_error_V'] and voltage_error<=1e-7
    assert power_error==row['power_absolute_error_W'] and power_error<=1e-8
    powers.append(power)
value=math.fsum(powers)/24
assert math.isclose(value,summary['mean_source_power_W'],rel_tol=0,abs_tol=1e-14)
assert receipt['mean_source_power_W']==summary['mean_source_power_W']==receipt['evaluation']['evaluation']['metrics']['source_power']
counts=receipt['counts']
assert counts['factory_starts']==counts['cleanups_returned']==counts['analytical_reference_solves']==24
assert counts['analytical_bisection_steps']==sum(r['analytical_reference']['steps'] for r in receipt['case_results'])<=4800
for observation in state.observations.values():
    assert len(observation.evidence)==1
    row=unique_json(read_bytes(store.root,observation.evidence[0]).decode())
    original=next(r for r in receipt['case_results'] if r['scenario_id']==row['scenario_id'])
    assert all(row[k]==v for k,v in original.items())
record=load('development-reference.json');gain=load('gain-reference.json')
assert record['value']==summary['mean_source_power_W'] and record['execution_receipt_digest']==ref.sha256
assert record['baseline_model_digest']==summary['model_digest'] and record['input_manifest_digest']==digest(inputs)
assert gain['target_proposal_digest']==digest(proposal) and gain['development_input_digest']==digest(inputs)
assert gain['reference']['sha256']==hashlib.sha256((root/'development-reference.json').read_bytes()).hexdigest()
assert gain['binding']['absolute_gain']==.05*record['value']==summary['proposed_absolute_useful_gain_W']
assert gain['binding']['status']=='reference_bound_not_admitted' and gain['confirmation_task_admitted'] is False
print(json.dumps({'saved_reference_verified':True,'coupled_observations':24,'mean_source_power_W':record['value'],'absolute_gain_W':gain['binding']['absolute_gain'],'new_model_calls':0,'task_admitted':False}))
