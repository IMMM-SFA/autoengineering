def counters(p):
    return {'component_calls':int(sum(c.iter_count+c.iter_count_apply for c in p.model.circuit.system_iter(recurse=True))),
            'component_linearize_calls':int(p.derivative_calls)}

def factory(ctx):
    assert ctx.model.digest==request['model_digest']
    stats['problem_instances']+=1;index=stats['problem_instances']
    p=original_factory(ctx);run=p.model.run_solve_nonlinear;cleanup=p.cleanup
    def counted(*args,**kwargs):
        if stats['coupled_evaluation_intents']>=64:raise RuntimeError('public conventional evaluation cap exhausted')
        stats['coupled_evaluation_intents']+=1;ordinal=stats['coupled_evaluation_intents']
        forcing={'ground_voltage':float(p.get_val('ground.V').item()),'resistor1_ohm':float(p.model.circuit.R1.options['R']),
                 'resistor2_ohm':float(p.model.circuit.R2.options['R'])}
        current=float(p.get_val('source.I').item());before=counters(p)
        identity=digest({'recipe':digest(recipe),'ordinal':ordinal,'problem':index,'forcing':forcing,'source_current_A':current})
        reservation=PhysicalReservation('conventional-'+identity[:32],'conventional','science',identity,None,batch.clock(),10.,0,0,0)
        def perform(_):
            begin=time.monotonic();raw=run(*args,**kwargs);work=load.diagnostics(p).to_dict();after=counters(p)
            outputs={name:float(p.get_val(path).item()) for name,path in (('node1_voltage','circuit.n1.V'),('node2_voltage','circuit.n2.V'))}
            record={'ordinal':ordinal,'problem_index':index,'phase':phase,'forcing':forcing,'source_current_A':current,
                'outputs':outputs,'work':work,'counter_before':before,'counter_after':after,
                'counter_delta':{key:after[key]-before[key] for key in before},'seconds':time.monotonic()-begin}
            return raw,record
        def observe(reservation,result):
            row=result[1];evidence=batch.evidence({'schema':'utility-engineering-v3-conventional-solve-v1','request_digest':digest(request),**row})
            good=row['work']['converged'] and row['work']['residual_norm']<=1e-10
            return PhysicalObservation(reservation.id,'success' if good else 'scientific_infeasible',True,0,{},None,'not_applicable',row['seconds'],(evidence,))
        raw,row=batch.execute(reservation,perform,observe)
        stats['coupled_evaluations_completed']+=1;solves.append(row)
        assert row['work']['converged'] and row['work']['residual_norm']<=1e-10
        assert row['work']['residual_tolerance']==1e-10 and row['work']['solver_iterations']<=100
        assert all(v>=0 for v in row['counter_delta'].values())
        print(json.dumps({'coupled_evaluation':ordinal,'phase':phase,'maximum_evaluations':64}),flush=True)
        return raw
    def close():
        totals={'problem_index':index,**counters(p)}
        value=cleanup();stats['cleanups_returned']+=1;problem_totals.append(totals);return value
    p.model.run_solve_nonlinear=counted;p.cleanup=close
    return p
native.factory=factory;load.factory=factory

        and work['residual_norm']<=1e-10 and work['residual_tolerance']==1e-10 and work['solver_iterations']<=100
        and values['node2_voltage']>=.62)
