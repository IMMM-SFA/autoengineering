from pathlib import Path

 def find_spec(self,fullname,*args):
  if fullname.split('.')[0] in {'examples','torch','openmdao','pvlib','py_wake'}:
   raise AssertionError('model import')
sys.meta_path.insert(0,NoModels())
import numpy
from autoengineering.utility.guest_owner import GuestOwnerJournal
from autoengineering.utility.guest_watchdog import GuestWatchdogOutcome,GuestWatchdogPlan,read_outcome
bundle=Path(__file__).parent
expected=GuestWatchdogOutcome.from_json((bundle/'expected-outcome.json').read_text())
with tempfile.TemporaryDirectory() as folder:
 root=Path(folder).resolve()
 with tarfile.open(bundle/'authored-startup-cleanup-owner.tar.gz') as archive:
  archive.extractall(root,filter='data')
 for path in root.rglob('*'):
  path.chmod(0o700 if path.is_dir() else 0o600)
 def forbidden(*args,**kwargs):raise AssertionError('execution, clock or array load')
 subprocess.Popen=socket.socket=time.monotonic=time.time=numpy.load=forbidden
 active=[True]
 def reads(event,args):
  if not active[0] or event!='open' or isinstance(args[0],int):return
  path=Path(args[0]).resolve();mode,flags=args[1:3]
  if not path.is_relative_to(root):raise AssertionError('private or unrelated read')
  writing=mode is not None and any(c in mode for c in 'wax+')
  writing=writing or flags&(os.O_WRONLY|os.O_RDWR|os.O_CREAT|os.O_TRUNC|os.O_APPEND)
  if writing:raise AssertionError('write')
 sys.addaudithook(reads)
 owner=GuestOwnerJournal(root/'owner')
 plan=GuestWatchdogPlan.from_json((owner.root/'control/guest-watchdog/plan.json').read_text())
 assert read_outcome(owner.root/'control/guest-watchdog',owner,plan)==expected
 print('Original watchdog startup cleanup agrees from relocated saved observations only.')
