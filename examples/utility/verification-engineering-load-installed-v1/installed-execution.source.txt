from pathlib import Path
import json
import os
import shutil
import sys

base = Path(sys.argv[1])
c = json.loads((base / "configuration.json").read_text())
sys.path.insert(0, c["installed"])
os.environ["PYTHONPATH"] = c["installed"]

from autoengineering.agents.batch import BatchCase, BatchSpec, BatchStore
from autoengineering.agents.hosts.fake import FakeHost
from autoengineering.agents.identity import host_environment_identity, host_source_identity
from autoengineering.agents.records import HostSelection, ResourceLimits
from autoengineering.campaign.artifacts import reference
from autoengineering.campaign.serialization import digest
from autoengineering.optimization import (
    BudgetSpec, ContinuousParameter, NoiseSpec, ObjectiveSpec, OptimizationRunSpec, SearchSpace, StudySpec,
)
from autoengineering.optimization.records import EvaluationAction
from autoengineering.utility.engineering_native import (
    NativeDriverSpec, SupervisedNativeDriver, native_cost_records, verify_native_report,
)
from autoengineering.utility.engineering_process import (
    EngineeringExecutionSpec, EngineeringScenario, SupervisedEngineeringExecutor, execution_identity,
)
from autoengineering.utility.studies import EngineeringStudySpec
from examples.utility.engineering import rules
import examples.utility.engineering_load as load
import examples.utility.engineering_load_native as native

assert Path(load.__file__).is_relative_to(c["installed"])
assert Path(native.__file__).is_relative_to(c["installed"])
source = Path(c["source"])
refs = tuple(reference(source, name, media_type="text/plain", producer="installed-load-control")
             for name in c["source_names"])
lock = reference(source, "pixi.lock", media_type="text/plain", producer="installed-load-control")
cases = (EngineeringScenario("authored-a", {"ground_voltage": .025,
            "resistor1_ohm": 110., "resistor2_ohm": 9000.}),
         EngineeringScenario("authored-b", {"ground_voltage": -.025,
            "resistor1_ohm": 90., "resistor2_ohm": 11000.}))
entry = "examples.utility.engineering_load:execute"
run = OptimizationRunSpec(StudySpec("installed-load-control", ObjectiveSpec("source_power", "minimize"),
    (), BudgetSpec(2, "physical_executions", 1), NoiseSpec(), "system", 17),
    SearchSpace((ContinuousParameter("source_current", .01, .1),)), "sobol", {},
    "autoengineering.utility.studies:engineering_evaluator")
study = EngineeringStudySpec(run, rules(.62, .001), tuple(s.scenario_id for s in cases),
    execution_identity(entry, refs, lock, cases, 10., 32768))
execution = EngineeringExecutionSpec("utility-engineering-execution-v1", study, entry,
                                     refs, lock, cases, 10., 32768)
spec = NativeDriverSpec("utility-native-driver-v1", execution,
    "examples.utility.engineering_load_native:execute", 20, 40, 10., 32768)
batch_spec = BatchSpec("1.0", "installed-load-control", host_source_identity(), host_environment_identity(),
    HostSelection("fake", "none", "not-used", "1.0"), FakeHost(()).capabilities(),
    digest({"host": "not_invoked"}), ResourceLimits(1, 1, 0, 0, 180., 15., 1048576, 1048576),
    3, (BatchCase("loads", "comparison", study.digest, 0, 3),))
batch = BatchStore.create(base / "batch", batch_spec)
batch.start_case("loads", study.digest)
runner = SupervisedEngineeringExecutor(execution, batch, "loads", source)
action = EvaluationAction("installed-fixed-load", {"source_current": .05}, "system", None, seed=17)
value = runner(action)
assert value.evaluation.status == "success" and len(value.scenarios) == 2
driver = SupervisedNativeDriver(spec, runner)
report = driver.run()
assert report.status == "success"
verify_native_report(batch, "loads", spec, report)
state = batch.replay()
assert len(state.reservations) == 3 and not state.pending
assert all(v.stop_verified for v in state.observations.values())

def forbidden(*args, **kwargs):
    raise AssertionError("completed recovery started a new process")

import autoengineering.utility.engineering_process as process_module
import autoengineering.utility.engineering_native as native_module
process_module.supervise_process = forbidden
native_module.supervise_process = forbidden
assert runner.recover_completed(action) == value
assert SupervisedNativeDriver(spec, runner).run(recover_only=True) == report
assert len(batch.replay().reservations) == 3
archive = Path(c["archive"])
(archive / "native-spec.json").write_text(spec.to_json())
(archive / "native-report.json").write_text(report.to_json())
(archive / "fixed-result.json").write_text(value.to_json())
shutil.copytree(batch.root, archive / "public-batch")
costs = {row.resource: row.quantity for row in native_cost_records(report, "0" * 64)}
summary = {"status": "passed", "load_module": load.__file__, "native_module": native.__file__,
    "fixed_scenario_processes": 2, "native_processes": 1, "original_reservations": 3,
    "recovery_added_starts": 0, "native_costs": costs,
    "inherited_dependencies": "interoperability", "final_scenarios_evaluated": False,
    "private_scoring_admitted": False}
(archive / "installed-execution.json").write_text(json.dumps(summary, indent=2) + "\n")
print(json.dumps(summary))
