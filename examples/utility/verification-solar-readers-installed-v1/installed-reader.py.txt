"""Installed solar readers from relocated authored bundles, without runtime access."""
import importlib.abc
import sys
from pathlib import Path

site=Path(sys.argv[1]).resolve()
root=Path(sys.argv[2]).resolve()
sys.path.insert(0,str(site))
sys.dont_write_bytecode=True

class NoModels(importlib.abc.MetaPathFinder):
    def find_spec(self,fullname,*args):
        if (fullname.split('.')[0] in {'torch','pvlib','openmdao','py_wake','topfarm','bmipy'}
            or fullname.startswith(('examples.leaf_river','examples.interoperability','examples.utility.hydrology_'))):
            raise AssertionError('model import during installed solar reading')

sys.meta_path.insert(0,NoModels())
import json,os,socket,subprocess,time
import numpy
from examples.utility.solar_inputs import read_development,read_reserved
for name,module in tuple(sys.modules.items()):
    if (name=='autoengineering' or name.startswith('autoengineering.')
        or name=='examples.utility' or name.startswith('examples.utility.solar_')):
        path=getattr(module,'__file__',None)
        if path is not None and not Path(path).resolve().is_relative_to(site):
            raise AssertionError('reader imported from outside installed target')
expected=json.loads((root/'expected.json').read_text())
kwargs=dict(expected_manifest_digest=expected['manifest_digest'],expected_preparer_sources=expected['sources'])
# Initialize metadata conversion before restricting all subsequent imports.
initial=read_development(root/'public',**kwargs)
public_inodes={(p.stat().st_dev,p.stat().st_ino) for p in (root/'public').rglob('*') if p.is_file()}
all_inodes={(p.stat().st_dev,p.stat().st_ino) for p in root.rglob('*') if p.is_file()}
phase={'reserved':False}

def forbidden(*args,**kwargs):
    raise AssertionError('execution, network, array loading or clock during installed reading')
subprocess.Popen=socket.socket=numpy.load=time.monotonic=time.time=forbidden

def saved(event,args):
    if event=='open':
        allowed=root if phase['reserved'] else root/'public'
        if isinstance(args[0],int):
            info=os.fstat(args[0])
            if (info.st_dev,info.st_ino) not in (all_inodes if phase['reserved'] else public_inodes):
                raise AssertionError('unscoped descriptor read')
        elif not Path(args[0]).resolve().is_relative_to(allowed):
            raise AssertionError('read outside current public/reserved phase')
        mode,flags=args[1:3]
        if any(c in (mode or '') for c in 'wax+') or flags & (os.O_WRONLY|os.O_RDWR|os.O_CREAT|os.O_TRUNC|os.O_APPEND):
            raise AssertionError('file write during installed reading')
    elif event=='import' or event.startswith(('socket.','subprocess.','os.exec')) or event in ('os.system','os.fork','os.posix_spawn'):
        raise AssertionError('import, network or execution after initialization')

sys.addaudithook(saved)
development=read_development(root/'public',**kwargs)
assert development==initial
assert len(development.forcing.start)==12
assert sum(development.labels['fitting'].score_mask)==1
assert development.forcing.segment_id[6:9]==(None,1,1)
phase['reserved']=True
reserved=read_reserved(root/'public',root/'private',**kwargs)
assert len(reserved.forcing.start)==16
assert sum(reserved.labels['confirmation'].score_mask)==4
assert development.qualification==reserved.qualification=='not_admitted'
print(json.dumps({'installed_target_used':True,'development_rows':12,'fitting_rows':1,'selection_rows':4,
                  'confirmation_rows':4,'new_model_starts':0,'new_processes_after_initialization':0,
                  'source_fallback':False,'data_scope':'relocated authored bundles','task_admitted':False}))
