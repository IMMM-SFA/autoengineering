"""Read the relocated authored journal; no actual runtime qualification."""
import json
from pathlib import Path
import socket
import subprocess
import tarfile
import tempfile
import time

from autoengineering.utility.guest_owner import GuestOwnerJournal

bundle = Path(__file__).parent
expected = json.loads((bundle / 'expected-projections.json').read_text())
with tempfile.TemporaryDirectory() as directory:
    destination = Path(directory).resolve()
    with tarfile.open(bundle / 'authored-owner-journal.tar.gz') as archive:
        archive.extractall(destination, filter='data')
    root = destination / 'owner'
    for path in [root, *root.rglob('*')]:
        path.chmod(0o700 if path.is_dir() else 0o600)
    def forbidden(*args, **kwargs):
        raise AssertionError('replay cannot execute, connect or sample a new clock')
    subprocess.Popen = socket.socket = time.monotonic = forbidden
    owner = GuestOwnerJournal(root)
    assert owner.manifest.digest == expected['owner_digest']
    assert [p.to_dict() for p in owner.projections()] == expected['projections']
    assert len(owner.replay().events) == 7
    assert owner.replay().phase == 'cleaned'
    print('Relocated authored owner journal agrees; no runtime admission.')
