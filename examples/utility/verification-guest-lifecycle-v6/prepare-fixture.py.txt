"""Stage a fresh reusable copy of the already trained v6 development fixture.

No fitting, model evaluation, driver generation or daemon action occurs here.
The owner template is authored metadata, not an operating-system observation.
"""
import hashlib,json,os,shutil
from pathlib import Path
from autoengineering.utility.confirmation import read_selected_state
from autoengineering.utility.guest_owner import GuestOwnerManifest
from autoengineering.utility.store import SuiteStore
from examples.utility import guest_inference_smoke as m
root=Path.cwd();os.umask(0o077)
p=root/'examples/utility/verification-guest-lifecycle-v6'
original=root/'outputs/guest-inference-current-fixture-v6'
out=root/'outputs/guest-current-frozen-fixture-v6'
assert not out.exists()
out.mkdir(mode=0o700)
(out/'owner').mkdir(mode=0o700)
raw=(original/'template/owner/owner.json').read_bytes()
(out/'owner/owner.json').write_bytes(raw)
store=SuiteStore(original/'authored-guest-metadata/composite')
assert not store.replay()['confirmation_accesses']
job=GuestOwnerManifest.from_json(raw.decode()).job
bound=read_selected_state(store.root,job.grant.inference.selected_state)
m.copy_json_tree(store.root,out/'suite')
for ref in bound.model.source_bundle:
 target=out/'suite'/ref.path
 m.private_parents(out/'suite',target.relative_to(out/'suite').parent)
 target.write_bytes((store.root/ref.path).read_bytes())
for name in ('source','drivers'):
 shutil.copytree(root/'outputs/guest-inference-v6'/name,out/name)
assert not list((out/'suite').rglob('*.npz'))
assert not list((out/'source').rglob('*.npz'))
assert [x.name for x in (out/'drivers').rglob('*.npz')]==['drivers.npz']
for directory in out.rglob('*'):
 if directory.is_dir():directory.chmod(0o700)
files={str(f.relative_to(out)):{'sha256':hashlib.sha256(f.read_bytes()).hexdigest(),'bytes':f.stat().st_size} for f in out.rglob('*') if f.is_file()}
m.save(p,'fixture-manifest.json',{'purpose':'separately counted current-image controls over exposed authored inputs','fit_sha256':bound.training.fit.sha256,'suite_digest':store.spec.digest,'files':files,'new_fitting_starts':0,'new_model_starts':0,'private_observations':0})
print('Reused the saved v6 fit and selection; no new fitting or model execution.')
