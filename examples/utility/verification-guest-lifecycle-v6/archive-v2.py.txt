"""Archive completed v6 lifecycle records, then verify relocated metadata only."""
import hashlib,json,os,shutil,subprocess,sys,tarfile
from pathlib import Path
from autoengineering.campaign.serialization import digest
from examples.utility import guest_inference_smoke as fixture
from examples.utility import guest_cancellation_reader,guest_running_cancellation_reader,guest_response_loss_reader
root=Path.cwd(); os.umask(0o077)
proof=root/'examples/utility/verification-guest-lifecycle-v6'
items=[('unstarted','guest-cancellation-controls-v6','guest-cancellation-controls-v1',guest_cancellation_reader),('running','guest-running-cancellation-v6','guest-running-cancellation-v1',guest_running_cancellation_reader),('response-loss','guest-response-loss-v6','guest-response-loss-v1',guest_response_loss_reader)]
def copy_scope(src,dst,*,suite=False):
 if dst.exists():
  return  # Preserve an already copied scope during metadata-only finalization.
 dst.mkdir(mode=0o700)
 for p in src.iterdir():
  if p.is_file() and p.suffix in ('.json','.txt'):shutil.copyfile(p,dst/p.name)
 if (src/'api-observations').exists():shutil.copytree(src/'api-observations',dst/'api-observations')
 if (src/'work').exists():
  owners=[p for p in (src/'work').iterdir() if p.is_dir() and (p/'owner.json').is_file()]
  assert len(owners)==1
  assert not any(p.is_symlink() for p in owners[0].rglob('*'))
  shutil.copytree(owners[0],dst/'owner')
 if suite:shutil.copytree(src/'suite',dst/'suite')
results=[]
for kind,name,prior,reader in items:
 run=root/'outputs'/name
 data=proof/kind
 copy_scope(run,data,suite=kind=='running')
 if kind!='running':
  for case in json.loads((run/'request.json').read_text())['case_requests']:
   copy_scope(run/case,data/case)
 for directory in data.rglob('*'):
  if directory.is_dir():directory.chmod(0o700)
 done=json.loads((data/'complete.json').read_text()); expected=digest(done)
 if kind=='running' and not (data/'helper.py.txt').exists():
  original_helper=root/'examples/utility/guest_owner_loss_controls.py'
  control=json.loads((data/'control-request.json').read_text())
  assert hashlib.sha256(original_helper.read_bytes()).hexdigest()==control['helper_sha256']
  shutil.copyfile(original_helper,data/'helper.py.txt')
 reader.read_completion(data,expected_digest=expected)
 cold=proof/(kind+'-cold.py.txt')
 shutil.copyfile(root/'examples/utility'/('verification-'+prior)/'cold-recovery.py.txt',cold)
 files={str(p.relative_to(data)):{'sha256':hashlib.sha256(p.read_bytes()).hexdigest(),'bytes':p.stat().st_size} for p in data.rglob('*') if p.is_file()}
 assert not any(Path(n).name in ('fitting.npz','comparison.npz','private-labels.json') for n in files)
 # Input-transfer archives contain only declared source, fit metadata and exposed drivers.
 for name in files:
  if name.endswith('.tar'):
   with tarfile.open(data/name) as a:
    assert all(Path(m.name).name not in ('fitting.npz','comparison.npz','private-labels.json') for m in a.getmembers())
 archive=proof/(kind+'-public-proof.tar.gz')
 if not archive.exists():
  with tarfile.open(archive,'w:gz') as a:
   for name in sorted(files):a.add(data/name,arcname=name,recursive=False)
 relocated=Path('/private/tmp')/('utility-lifecycle-v6-'+kind)
 if not relocated.exists():
  relocated.mkdir(mode=0o700)
  with tarfile.open(archive) as a:a.extractall(relocated,filter='data')
 assert all(hashlib.sha256((relocated/n).read_bytes()).hexdigest()==v['sha256'] for n,v in files.items())
 for directory in relocated.rglob('*'):
  if directory.is_dir():directory.chmod(0o700)
 command=[sys.executable,'-B','-c',"import runpy,sys; script=sys.argv.pop(1); runpy.run_path(script,run_name='__main__')",str(cold),str(relocated),expected]
 if (proof/(kind+'-cold.stdout.txt')).exists():
  assert (proof/(kind+'-cold.stderr.txt')).read_text()==''
  json.loads((proof/(kind+'-cold.stdout.txt')).read_text())
  p=subprocess.CompletedProcess(command,0)
 else:
  p=subprocess.run(command,cwd=root,capture_output=True,text=True)
  (proof/(kind+'-cold.stdout.txt')).write_text(p.stdout);(proof/(kind+'-cold.stderr.txt')).write_text(p.stderr)
  assert p.returncode==0,p.stderr
 (proof/(kind+'-manifest.json')).write_text(json.dumps(files,indent=2,sort_keys=True)+'\n')
 result={'kind':kind,'completion_digest':expected,'request_digest':done['request_digest'],'container_creations':done['container_creations'],'container_starts':done['container_starts'],'charged_inference_starts':done.get('charged_inference_starts',0),'all_passed':done['all_passed'],'all_original_containers_absent':True,'archive_sha256':hashlib.sha256(archive.read_bytes()).hexdigest(),'archive_files':len(files),'uncompressed_bytes':sum(x['bytes'] for x in files.values()),'cold_exit_code':p.returncode,'cold_evidence_relocated':True,'cold_models_processes_network_arrays_clocks_writes_outside_reads_denied':True,'cold_initialization_before_guard':True,'cold_dependencies':'inherited'}
 results.append(result)
summary={'scope':'Five actual lifecycle controls of the v6 candidate image over the existing authored exposed fit','image_id':'sha256:5fcc625ec3f746219ad50dbe7cd49886362702a17e211e1ffbefc39dd44f4188','controls':results,'container_creations':sum(x['container_creations'] for x in results),'container_starts':sum(x['container_starts'] for x in results),'charged_inference_starts':sum(x['charged_inference_starts'] for x in results),'model_results_observed':0,'model_inference_starts_total':None,'model_elapsed_seconds':None,'new_fitting_starts':0,'scoring_starts':0,'private_observations':0,'automatic_retries':0,'parent_composite_reservations_remain_charged':True,'owner_process_sigkill_qualified':False,'watchdog_sigkill_qualified':False,'during_model_computation_cancellation_qualified':False,'host_scope':'Local OrbStack Linux amd64 emulation on macOS ARM64','runtime_qualified':False,'independent_confirmation':False,'full_part_5_complete':False}
fixture.save(proof,'summary.json',summary)
for name in ('guest_cancellation_controls.py','guest_cancellation_reader.py','guest_running_cancellation.py','guest_running_cancellation_reader.py','guest_response_loss_controls.py','guest_response_loss_reader.py','guest_owner_loss_controls.py','guest_inference_smoke.py'):
 shutil.copyfile(root/'examples/utility'/name,proof/(name+'.txt'))
print(json.dumps(summary))
