"""Owned local diagnostic starts, with a separate owner-pipe cleanup process.

This candidate controller accepts only the six fixed resource probes. It never
pulls an image, mounts host files, runs a model or reads scientific labels.
Unknown mutations remain charged. No command resumes or retries an attempt.
"""

from __future__ import annotations

import argparse
import hashlib
import os
from pathlib import Path
import re
import select
import subprocess
import sys
import time
from uuid import uuid4

from autoengineering.campaign.artifacts import confined_file
from autoengineering.campaign.serialization import canonical_json, digest, unique_json
from autoengineering.campaign.store import checked_directory, durable_write
from autoengineering.utility.guest_daemon import GuestDaemonAPI
from examples.utility import guest_identity, guest_image_build as build, guest_resources as policy


def save(root, name, value):
    durable_write(root / name, canonical_json(value).encode())


def read(root, name):
    path = confined_file(root, name)
    if path.stat().st_size > 1024**2:
        raise ValueError("diagnostic metadata exceeds envelope")
    return unique_json(path.read_text())


def sources():
    return {
        name: hashlib.sha256(Path(__file__).with_name(name).read_bytes()).hexdigest()
        for name in ("guest_resource_runner.py", "guest_resources.py", "guest_resource_probe.py")
    }


def api_for(request, bindings):
    api = GuestDaemonAPI(
        bindings["endpoint"],
        api_version=request["daemon"]["api_version"],
        server_digest=request["daemon"]["server_digest"],
    )
    if api.profile.to_dict() != request["daemon"]:
        raise ValueError("diagnostic daemon implementation differs")
    return api


def prepare(identity_work, *, identity_digest, output, work):
    original = checked_directory(identity_work)
    completion = guest_identity.read_completion(original, expected_digest=identity_digest)
    profile = completion["profile"]
    if (
        profile["image_id"] != policy.IMAGE
        or profile["python"]["build_digest"] != policy.probe.PYTHON_BUILD
    ):
        raise ValueError("diagnostic image or interpreter differs")
    old_request, old = read(original, "request.json"), read(original, "bindings.json")
    output, work = Path(output).absolute(), Path(work).absolute()
    for path in (output, work):
        build.guest_image._public_directory(path.parent)
        if path.exists() or path.is_symlink():
            raise ValueError("diagnostic output and work must be absent")
        if path.is_relative_to(original) or original.is_relative_to(path):
            raise ValueError("diagnostic scopes overlap identity evidence")
    if output.is_relative_to(work) or work.is_relative_to(output):
        raise ValueError("diagnostic output and work overlap")
    if (
        build._binary(old["cli"]) != old_request["binary"]
        or build._endpoint(old["endpoint"]) != old_request["endpoint"]
    ):
        raise ValueError("diagnostic client or endpoint changed")
    bindings = {"cli": old["cli"], "endpoint": old["endpoint"], "work": str(work), "canaries": {}}
    cases = []
    for mode in policy.probe.MODES:
        nonce = uuid4().hex
        canary = str(output / ("ae-host-canary-" + nonce)) if mode == "host_canary_denied" else None
        if canary is not None:
            bindings["canaries"][nonce] = canary
        cases.append(
            {
                "mode": mode,
                "nonce": nonce,
                "configuration_digest": digest(policy.configuration(mode, nonce, canary=canary)),
            }
        )
    api = GuestDaemonAPI(
        old["endpoint"],
        api_version=old_request["runtime"]["api_version"],
        server_digest=old_request["runtime"]["server_digest"],
    )
    request = {
        "schema": "utility-guest-resource-request-v1",
        "identity_digest": identity_digest,
        "profile": profile,
        "cases": cases,
        "sources": sources(),
        "daemon": api.profile.to_dict(),
        "bindings_digest": digest(bindings),
        "binary": old_request["binary"],
        "endpoint": old_request["endpoint"],
        "active_seconds": 60,
        "cleanup_seconds": 10,
        "creations_reserved": 6,
        "starts_reserved": 6,
        "automatic_retries": 0,
        "scientific_starts": 0,
        "runtime_qualified": False,
    }
    output.mkdir(mode=0o700)
    save(output, "request.json", request)
    save(output, "bindings.json", bindings)
    for name in request["sources"]:
        durable_write(output / (name + ".txt"), Path(__file__).with_name(name).read_bytes())
    for path in bindings["canaries"].values():
        durable_write(Path(path), b"fresh authored diagnostic canary; no scientific data\n")
    return request


def load(bundle, expected_digest):
    bundle = checked_directory(bundle)
    request, bindings = read(bundle, "request.json"), read(bundle, "bindings.json")
    if (
        digest(request) != expected_digest
        or digest(bindings) != request["bindings_digest"]
        or request["sources"] != sources()
    ):
        raise ValueError("diagnostic request, bindings or sources changed")
    if (
        request["active_seconds"],
        request["cleanup_seconds"],
        request["creations_reserved"],
        request["starts_reserved"],
        request["automatic_retries"],
        request["scientific_starts"],
        request["runtime_qualified"],
    ) != (60, 10, 6, 6, 0, 0, False):
        raise ValueError("diagnostic bounds differ")
    if [case["mode"] for case in request["cases"]] != list(policy.probe.MODES):
        raise ValueError("diagnostic case set differs")
    if len({case["nonce"] for case in request["cases"]}) != 6:
        raise ValueError("diagnostic nonce reused")
    for case in request["cases"]:
        canary = bindings["canaries"].get(case["nonce"])
        if case["configuration_digest"] != digest(
            policy.configuration(case["mode"], case["nonce"], canary=canary)
        ):
            raise ValueError("diagnostic command or policy changed")
    return bundle, request, bindings


def inspect(api, target, deadline):
    if not re.fullmatch(r"(?:[0-9a-f]{64}|ae-guest-resource-[0-9a-f]{32})", target):
        raise ValueError("diagnostic inspection target differs")
    status, raw = api._request("GET", f"/containers/{target}/json", deadline=deadline)
    if status == 404:
        return None
    if status != 200:
        raise ValueError("diagnostic container observation is unknown")
    return unique_json(raw.decode())


def cleanup(api, case, work, *, canary, deadline):
    """One lease and at most one DELETE. Never renew the original allowance."""
    lease = work / "cleanup-lease"
    lease.mkdir(mode=0o700)
    save(work, "cleanup-intent.json", {"deadline": deadline, "nonce": case["nonce"]})
    known = None
    if (work / "created.json").exists():
        known = read(work, "created.json")["Id"]
        if not isinstance(known, str) or not re.fullmatch(r"[0-9a-f]{64}", known):
            raise ValueError("diagnostic original container ID differs")
    target = known or "ae-guest-resource-" + case["nonce"]
    value = inspect(api, target, deadline)
    cid = None
    if value is not None:
        cid = value.get("Id")
        if known is not None and cid != known:
            raise ValueError("diagnostic cleanup original container ID differs")
        policy.verify_owned(value, case["mode"], case["nonce"], cid=cid, canary=canary)
        save(work, "delete-intent.json", {"container_id": cid, "nonce": case["nonce"]})
        status, _ = api._request("DELETE", f"/containers/{cid}?force=true", deadline=deadline)
        if status not in (204, 404):
            raise ValueError("diagnostic deletion outcome is unknown")
        if inspect(api, cid, deadline) is not None:
            raise ValueError("diagnostic container absence is unverified")
    # A 404 after an uncertain create is not proof that a late creation cannot
    # happen. The original create response and matching CID must be present.
    if known is None:
        raise ValueError("diagnostic creation outcome is unknown")
    result = {
        "container_id": known,
        "absent": True,
        "nonce": case["nonce"],
        "delete_attempts": int(cid is not None),
        "owner_loss_qualified": False,
    }
    save(work, "cleanup.json", result)
    return result


def watchdog(bundle, expected_digest, index, reader, ready):
    bundle, request, bindings = load(bundle, expected_digest)
    if type(index) is not int or not 0 <= index < 6:
        raise ValueError("diagnostic watchdog case differs")
    case = request["cases"][index]
    work = checked_directory(Path(bindings["work"]) / case["mode"])
    intent = read(work, "intent.json")
    if intent["request_digest"] != expected_digest or intent["deadline"] - intent["began_at"] != 60:
        raise ValueError("diagnostic watchdog original deadline differs")
    api = api_for(request, bindings)
    os.write(ready, b"R")
    os.close(ready)
    reason = "deadline"
    try:
        while time.monotonic() < intent["deadline"]:
            available, _, _ = select.select(
                [reader], [], [], min(0.1, max(0, intent["deadline"] - time.monotonic()))
            )
            if available and not os.read(reader, 1):
                reason = "owner_pipe_closed"
                break
    finally:
        os.close(reader)
    save(work, "watchdog-trigger.json", {"reason": reason, "at": time.monotonic()})
    deadline = min(time.monotonic() + 10, intent["deadline"] + 10)
    try:
        cleanup(api, case, work, canary=bindings["canaries"].get(case["nonce"]), deadline=deadline)
    except Exception:
        save(work, "cleanup-unknown.json", {"nonce": case["nonce"], "retries": 0})
        raise


def run_case(bundle, request, bindings, root, index):
    case, expected = request["cases"][index], digest(request)
    work = root / case["mode"]
    work.mkdir(mode=0o700)
    began = time.monotonic()
    deadline = began + 60
    save(
        work,
        "intent.json",
        {
            "request_digest": expected,
            "began_at": began,
            "deadline": deadline,
            "creations_reserved": 1,
            "starts_reserved": 1,
        },
    )
    api = api_for(request, bindings)
    canary = bindings["canaries"].get(case["nonce"])
    config = policy.configuration(case["mode"], case["nonce"], canary=canary)
    server = api._json("GET", "/version", deadline=deadline)
    if digest(server) != request["daemon"]["server_digest"]:
        raise ValueError("diagnostic server changed")
    image = api._json("GET", f"/images/{policy.IMAGE}/json", deadline=deadline)
    if (image.get("Id"), image.get("Os"), image.get("Architecture")) != (
        policy.IMAGE,
        "linux",
        "amd64",
    ):
        raise ValueError("diagnostic image changed")
    save(work, "server.json", server)
    save(work, "image.json", image)
    reader, writer = os.pipe()
    ack_reader, ack_writer = os.pipe()
    child = None
    try:
        child = subprocess.Popen(
            [
                sys.executable,
                "-m",
                "examples.utility.guest_resource_runner",
                "watchdog",
                str(bundle),
                expected,
                str(index),
                str(reader),
                str(ack_writer),
            ],
            pass_fds=(reader, ack_writer),
            stdin=subprocess.DEVNULL,
            stdout=subprocess.DEVNULL,
            stderr=subprocess.DEVNULL,
            start_new_session=True,
        )
        os.close(reader)
        reader = None
        os.close(ack_writer)
        ack_writer = None
        available, _, _ = select.select(
            [ack_reader], [], [], min(10, max(0, deadline - time.monotonic()))
        )
        if not available or os.read(ack_reader, 1) != b"R":
            raise ValueError("diagnostic watchdog is not ready")
        save(work, "watchdog-ready.json", {"request_digest": expected, "pid": child.pid})
        save(
            work,
            "create-intent.json",
            {"configuration_digest": digest(config), "nonce": case["nonce"]},
        )
        created = api._json(
            "POST",
            "/containers/create?name=ae-guest-resource-" + case["nonce"],
            config,
            expected=201,
            deadline=deadline,
        )
        cid = created.get("Id")
        if not isinstance(cid, str) or not re.fullmatch(r"[0-9a-f]{64}", cid):
            raise ValueError("diagnostic created ID is unknown")
        save(work, "created.json", created)
        before = inspect(api, cid, deadline)
        save(work, "before.json", before)
        state = policy.verify_owned(before, case["mode"], case["nonce"], cid=cid, canary=canary)
        if (
            state.get("Status") != "created"
            or state.get("Running") is not False
            or state.get("Pid") != 0
            or state.get("OOMKilled") is not False
            or state.get("ExitCode") != 0
            or state.get("Dead") is not False
            or any(
                state.get(k) not in (None, "", "0001-01-01T00:00:00Z")
                for k in ("StartedAt", "FinishedAt")
            )
        ):
            raise ValueError("diagnostic creation already started or is unverified")
        save(work, "start-intent.json", {"container_id": cid, "nonce": case["nonce"], "starts": 1})
        argv = [
            bindings["cli"],
            "--config",
            str(root / "config"),
            "--host",
            "unix://" + bindings["endpoint"],
            "start",
            "--attach",
            cid,
        ]
        status = build._run(
            argv,
            env={
                "PATH": "/usr/bin:/bin",
                "HOME": str(root / "home"),
                "DOCKER_CONFIG": str(root / "config"),
            },
            cwd=root,
            deadline=deadline,
            log=work / "output.log",
        )
        save(work, "client.json", {"exit_code": status, "argv_digest": digest(argv)})
        terminal = inspect(api, cid, deadline)
        save(work, "terminal.json", terminal)
        raw = confined_file(work, "output.log").read_bytes()
        observation = policy.verify_terminal(
            terminal, raw, case["mode"], case["nonce"], cid=cid, canary=canary
        )
        save(work, "observation.json", observation)
    finally:
        os.close(writer)
        for fd in (reader, ack_reader, ack_writer):
            if fd is not None:
                os.close(fd)
        if child is not None:
            try:
                child.wait(timeout=max(0.01, deadline + 10 - time.monotonic()))
            except subprocess.TimeoutExpired:
                # The cleanup process is independent. Do not terminate it or
                # manufacture another cleanup allowance when its outcome is unknown.
                save(work, "watchdog-unobserved.json", {"pid": child.pid, "retries": 0})
    result = {
        "observation": read(work, "observation.json"),
        "cleanup": read(work, "cleanup.json"),
        "elapsed_seconds": time.monotonic() - began,
    }
    if result["elapsed_seconds"] > 70:
        raise ValueError("diagnostic original allowance exceeded")
    save(work, "complete.json", result)
    return result


def execute(bundle, *, expected_digest):
    bundle, request, bindings = load(bundle, expected_digest)
    if (
        build._binary(bindings["cli"]) != request["binary"]
        or build._endpoint(bindings["endpoint"]) != request["endpoint"]
    ):
        raise ValueError("diagnostic client or endpoint changed")
    root = Path(bindings["work"])
    if root.exists() or root.is_symlink():
        raise ValueError("diagnostic attempt is consumed; automatic retry is forbidden")
    root.mkdir(mode=0o700)
    for name in ("config", "home"):
        (root / name).mkdir(mode=0o700)
    durable_write(root / "config" / "config.json", b"{}")
    save(root, "request.json", request)
    save(root, "bindings.json", bindings)
    results = []
    for index in range(6):
        results.append(run_case(bundle, request, bindings, root, index))
    result = {
        "schema": "utility-guest-resource-completion-v1",
        "request_digest": expected_digest,
        "cases": results,
        "physical_container_starts": 6,
        "scientific_starts": 0,
        "runtime_qualified": False,
        "owner_loss_qualified": False,
    }
    save(root, "complete.json", result)
    return result


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    sub = parser.add_subparsers(dest="command", required=True)
    prep = sub.add_parser("prepare")
    for name in ("identity_work", "output", "work"):
        prep.add_argument(name, type=Path)
    prep.add_argument("--identity-digest", required=True)
    run = sub.add_parser("execute")
    run.add_argument("bundle", type=Path)
    run.add_argument("--expected-digest", required=True)
    watch = sub.add_parser("watchdog")
    watch.add_argument("bundle", type=Path)
    watch.add_argument("expected_digest")
    for name in ("index", "reader", "ready"):
        watch.add_argument(name, type=int)
    args = parser.parse_args()
    if args.command == "prepare":
        result = prepare(
            args.identity_work,
            identity_digest=args.identity_digest,
            output=args.output,
            work=args.work,
        )
    elif args.command == "execute":
        result = execute(args.bundle, expected_digest=args.expected_digest)
    else:
        watchdog(args.bundle, args.expected_digest, args.index, args.reader, args.ready)
        return
    print(canonical_json({"digest": digest(result), "schema": result["schema"]}))


if __name__ == "__main__":
    main()
