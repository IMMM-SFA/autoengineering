"""Diagnostic policy and adverse evidence checks, without a container start."""

from copy import deepcopy
import json

import pytest

from examples.utility import guest_resources as module

NONCE, CID = "a" * 32, "b" * 64
CANARY = "/authored/ae-host-canary-" + NONCE
RESULTS = {
    "pid_limit": {"errno": "EAGAIN", "pids_current": 16, "children": 14},
    "workspace_limit": {"errno": "ENOSPC", "written_bytes": 134217728},
    "read_only_root": {"errno": "EROFS"},
    "network_denied": {"errno": "ENETUNREACH", "attempts": 1},
    "host_canary_denied": {"errno": "ENOENT"},
}


def authored(mode):
    kwargs = {"canary": CANARY} if mode == "host_canary_denied" else {}
    config = module.configuration(mode, NONCE, **kwargs)
    host = config.pop("HostConfig")
    value = {
        "Id": CID,
        "Image": module.IMAGE,
        "Name": "/ae-guest-resource-" + NONCE,
        "Config": config,
        "HostConfig": host,
        "Mounts": [],
        "RestartCount": 0,
        "State": {
            "Status": "exited",
            "Running": False,
            "Dead": False,
            "Paused": False,
            "Restarting": False,
            "Pid": 0,
            "OOMKilled": mode == "memory_limit",
            "ExitCode": 137 if mode == "memory_limit" else 0,
            "StartedAt": "2026-10-31T00:00:00Z",
            "FinishedAt": "2026-10-31T00:00:05Z",
        },
    }
    ready = {
        "schema": "utility-guest-resource-ready-v1",
        "mode": mode,
        "nonce": NONCE,
        "observed": deepcopy(module.OBSERVED),
    }
    lines = [ready]
    if mode != "memory_limit":
        lines.append(
            {
                "schema": "utility-guest-resource-result-v1",
                "mode": mode,
                "nonce": NONCE,
                "result": deepcopy(RESULTS[mode]),
            }
        )
    return value, lines, kwargs


def encoded(lines):
    return ("\n".join(json.dumps(line) for line in lines) + "\n").encode()


@pytest.mark.parametrize("mode", module.probe.MODES)
def test_authored_terminal_observations_are_not_runtime_admission(mode):
    value, lines, kwargs = authored(mode)
    result = module.verify_terminal(value, encoded(lines), mode, NONCE, cid=CID, **kwargs)
    assert result["runtime_qualified"] is False and result["owner_loss_qualified"] is False


@pytest.mark.parametrize(
    "field,altered",
    [
        ("Memory", 0),
        ("MemorySwap", -1),
        ("PidsLimit", 0),
        ("NanoCpus", 0),
        ("NetworkMode", "host"),
        ("ReadonlyRootfs", False),
        ("Privileged", True),
        ("PidMode", "host"),
        ("IpcMode", "host"),
        ("CapAdd", ["SYS_ADMIN"]),
        ("SecurityOpt", ["seccomp=unconfined"]),
        ("AutoRemove", True),
        ("Binds", ["/private:/private"]),
        ("VolumesFrom", ["another"]),
        ("ExtraHosts", ["test:127.0.0.1"]),
        ("GroupAdd", ["0"]),
        ("UsernsMode", "host"),
        ("Runtime", "unapproved"),
        ("Tmpfs", {"/work": "rw,size=1g"}),
    ],
)
def test_policy_drift_refuses_before_start(field, altered):
    value, _, _ = authored("memory_limit")
    value["HostConfig"][field] = altered
    with pytest.raises(ValueError):
        module.verify_owned(value, "memory_limit", NONCE, cid=CID)

@pytest.mark.parametrize("oom,code", [(False, 137), (True, 143), (False, 0), (True, True)])
def test_memory_requires_specific_oom_state(oom, code):
    value, lines, kwargs = authored("memory_limit")

        module.verify_terminal(value, encoded(lines), "memory_limit", NONCE, cid=CID, **kwargs)


@pytest.mark.parametrize(
    "mode,bad",
    [
        ("pid_limit", {"errno": "EAGAIN", "pids_current": 16, "children": 0}),
        ("pid_limit", {"errno": "EAGAIN", "pids_current": 16, "children": True}),
        ("pid_limit", {"errno": "EAGAIN", "pids_current": 15, "children": 13}),
        ("workspace_limit", {"errno": "ENOSPC", "written_bytes": 0}),
        ("workspace_limit", {"errno": "ENOSPC", "written_bytes": 134217729}),
        ("read_only_root", {"errno": "EACCES"}),
        ("network_denied", {"errno": "ECONNREFUSED", "attempts": 1}),
        ("network_denied", {"errno": "ENETUNREACH", "attempts": True}),
        ("host_canary_denied", {"errno": "EACCES"}),
        ("host_canary_denied", {"errno": "ENOENT", "private_path": CANARY}),
    ],
)
def test_other_failures_cannot_pass_as_enforcement(mode, bad):
    value, lines, kwargs = authored(mode)
    lines[1]["result"] = bad
    with pytest.raises(ValueError):
        module.verify_terminal(value, encoded(lines), mode, NONCE, cid=CID, **kwargs)


@pytest.mark.parametrize(
    "change",
    ["missing_ready", "extra_line", "wrong_nonce", "wrong_limits", "duplicate_key", "oversized"],
)
def test_malformed_or_unbound_stdout_refuses(change):
    mode = "read_only_root"
    value, lines, kwargs = authored(mode)
    if change == "missing_ready":
        lines.pop(0)
    elif change == "extra_line":
        lines.append(lines[-1])
    elif change == "wrong_nonce":
        lines[0]["nonce"] = "c" * 32
    elif change == "wrong_limits":
        lines[0]["observed"]["effective_limits"]["memory.max"] = "max"
    output = encoded(lines)
    if change == "duplicate_key":
        output = output.replace(b'"result": {', b'"mode": "read_only_root", "result": {')
    elif change == "oversized":
        output += b" " * module.MAX_OUTPUT
    with pytest.raises(ValueError):
        module.verify_terminal(value, output, mode, NONCE, cid=CID, **kwargs)


def test_docker_semantic_nulls_and_security_spelling():
    value, _, _ = authored("memory_limit")
    for field in (
        "Binds",
        "Mounts",
        "VolumesFrom",
        "Devices",
        "DeviceRequests",
        "CapAdd",
        "PortBindings",
    ):
        value["HostConfig"][field] = None
    value["HostConfig"]["SecurityOpt"] = ["no-new-privileges:true"]
    module.verify_owned(value, "memory_limit", NONCE, cid=CID)


def test_only_dedicated_diagnostic_profile_accepts_canary_metadata():
    with pytest.raises(ValueError):
        module.configuration("memory_limit", NONCE, canary=CANARY)
    with pytest.raises(ValueError):
        module.configuration("host_canary_denied", NONCE)
    with pytest.raises(ValueError):
        module.configuration("host_canary_denied", NONCE, canary="/private/labels")
    assert (
        module.configuration("host_canary_denied", NONCE, canary=CANARY)["HostConfig"]["Mounts"]
        == []
    )


def test_inherited_image_version_is_cleared_not_used_as_interpreter_identity():
    config = module.configuration("memory_limit", NONCE)
    assert {"GPG_KEY=", "PYTHON_VERSION=", "PYTHON_SHA256="} <= set(config["Env"])
    value, _, _ = authored("memory_limit")
    value["Config"]["Env"].remove("PYTHON_VERSION=")
    value["Config"]["Env"].append("PYTHON_VERSION=3.12.16")
    with pytest.raises(ValueError, match="environment"):
        module.verify_owned(value, "memory_limit", NONCE, cid=CID)
