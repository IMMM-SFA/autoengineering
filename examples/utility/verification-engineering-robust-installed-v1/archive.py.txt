"""Copy completed public receipts and verify them from an installed base runtime."""
import hashlib,json,os,shutil,subprocess,sys
from pathlib import Path
root=Path.cwd();os.umask(0o077)
p=root/'examples/utility/verification-engineering-robust-installed-v1'
base=Path('/private/tmp/utility-robust-installed-v1')
data=p/'public-evidence';data.mkdir(mode=0o700)
for name in ('native-spec.json','native-report.json','scientific-evaluation.json','installed-execution.json'):
 shutil.copyfile(base/'execution-proof'/name,data/name)
shutil.copytree(base/'execution-proof/public-batch',data/'batch')
shutil.copyfile(base/'execution-request.json',p/'execution-request.json')
for d in data.rglob('*'):
 if d.is_dir():d.chmod(0o700)
relocated=Path('/private/tmp/utility-robust-installed-v1-public');shutil.copytree(data,relocated)
s=(root/'examples/utility/verification-engineering-robust-native-v1/cold-reader-v2.py.txt').read_text()
s=s.replace("batch,'first'","batch,'loads'").replace("{'first':'0'*64,'second':'1'*64}","{'loads':'0'*64}").replace("cases/first/engineering-processes","cases/loads/engineering-processes")
(p/'cold-reader.py.txt').write_text(s)
command=[sys.executable,'-I','-B',str(p/'cold-reader.py.txt'),str(relocated),'/private/tmp/utility-load-cost-report-v1/installed']
r=subprocess.run(command,cwd='/private/tmp',capture_output=True,text=True)
(p/'cold.stdout.txt').write_text(r.stdout);(p/'cold.stderr.txt').write_text(r.stderr)
assert r.returncode==0,r.stderr
files={str(f.relative_to(data)):hashlib.sha256(f.read_bytes()).hexdigest() for f in data.rglob('*') if f.is_file()}
(p/'evidence-manifest.json').write_text(json.dumps(files,indent=2,sort_keys=True)+'\n')
execution=json.loads((data/'installed-execution.json').read_text())
build=json.loads((p/'build-install.json').read_text())
summary={'scope':'Selected-source installed robust-load recipe on bundled exposed authored data, outside checkout','wheel_sha256':build['wheel_sha256'],'package_python_sources':232,'core_python_sources':225,'domain_python_sources':7,'wheel_and_installed_payloads_match_current_inputs':True,'execution':execution,'cold_reader':json.loads(r.stdout),'cold_reader_exit_code':r.returncode,'cold_evidence_relocated':True,'cold_runtime':'existing installed core-only wheel with base dependencies; no example or OpenMDAO imports','fixture':'examples/utility/fixtures/engineering_robust_cases.json','fixture_sha256':hashlib.sha256((root/'examples/utility/fixtures/engineering_robust_cases.json').read_bytes()).hexdigest(),'original_checkout_content_denied_in_scientific_interpreter':True,'native_code_staged_from_verified_wheel':True,'network_or_dependency_installation':False,'evidence_files':len(files),'independent_confirmation':False,'private_boundary_admitted':False,'linux_execution_qualified':False,'full_distribution_release_qualified':False,'full_part_5_complete':False}
(p/'summary.json').write_text(json.dumps(summary,indent=2)+'\n')
print(json.dumps({'wheel':build['wheel_sha256'],'installed_execution':execution['status'],'cold_reader_exit':r.returncode,'evidence_files':len(files)}))
