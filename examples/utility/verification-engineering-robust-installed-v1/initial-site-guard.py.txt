import os,sys
from pathlib import Path
_original=Path('/Users/cam/projects/autoengineering/.worktrees/agentic-architecture')
_dependencies=Path('/Users/cam/Library/Caches/rattler/cache/envs/autoengineering-2706574624935427595/envs/interoperability')
def _source_guard(event,args):
    if event=='open' and isinstance(args[0],(str,bytes,os.PathLike)):
        p=Path(os.fsdecode(args[0])).absolute()
        if p.is_relative_to(_original) and not p.is_relative_to(_dependencies):
            raise RuntimeError('original checkout content read denied')
sys.addaudithook(_source_guard)
