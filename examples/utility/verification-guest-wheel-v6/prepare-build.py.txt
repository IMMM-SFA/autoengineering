"""Prepare a fresh bounded local build request, without dispatching Docker."""
from pathlib import Path
import json,sys
sys.path.insert(0,str(Path.cwd()))
from autoengineering.campaign.serialization import digest
from examples.utility.guest_image_build import prepare
root=Path.cwd();out=root/'examples/utility/verification-guest-wheel-v6'
context=json.loads((out/'context-result.json').read_text())
old=root/'examples/utility/prepared-guest-build-v4'
bindings=json.loads((old/'bindings.json').read_text())
request=prepare(root/context['context_path'],context_digest=context['context_digest'],
 cli=bindings['cli'],buildx=bindings['buildx'],base_manifest=old/'base-manifest.raw',endpoint=bindings['endpoint'],
 work=root/'outputs/guest-image-build-v6',output=root/'examples/utility/prepared-guest-build-v6')
result={'request_digest':digest(request),'request_path':'examples/utility/prepared-guest-build-v6/request.json',
 'schema':request['schema'],'context_digest':request['context_digest'],
 'client_deadline_seconds':request['client_deadline_seconds'],'build_commands':request['build_commands'],
 'maximum_base_pull_commands':request['base_pull_commands'],'automatic_retries':request['automatic_retries'],
 'scientific_starts':request['scientific_starts'],'provider_calls':request['provider_calls'],
 'independent_label_accesses':request['independent_label_accesses'],
 'docker_commands_dispatched':0,'image_built':False,'runtime_qualified':False,
 'dispatch_deferred_while_full_default_running':True,
 'required_followup':'Explicit-environment derivative and current-image controls; no historical admission transfer'}
(out/'build-request-result.json').write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps(result))
