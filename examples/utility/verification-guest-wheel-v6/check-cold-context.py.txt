"""Check a relocated context once, then preserve and refuse a changed wheel."""
from pathlib import Path
import json,shutil,subprocess,sys
root=Path.cwd();out=root/'examples/utility/verification-guest-wheel-v6'
result=json.loads((out/'context-result.json').read_text())
stage=Path('/private/tmp/utility-guest-wheel-v6-v2')
context=stage/'relocated-context';shutil.copytree(root/result['context_path'],context)
reader=stage/'cold-context-reader.py.txt';shutil.copyfile(out/reader.name,reader)
command=[sys.executable,'-I','-B',str(reader),str(context),str(root),result['context_digest']]
p=subprocess.run(command,cwd=stage,capture_output=True,text=True)
(out/'cold-context.stdout.txt').write_text(p.stdout);(out/'cold-context.stderr.txt').write_text(p.stderr)
assert p.returncode==0,p.stderr
cold=json.loads(p.stdout);cold['exit_code']=p.returncode
(out/'cold-context-result.json').write_text(json.dumps(cold,indent=2)+'\n')
wheel=next((context/'build/wheel').glob('*.whl'))
with wheel.open('ab') as stream:stream.write(b'changed-control-byte')
bad=subprocess.run(command,cwd=stage,capture_output=True,text=True)
assert bad.returncode!=0
(out/'tamper-context-result.json').write_text(json.dumps({'changed_payload_rejected':True,'exit_code':bad.returncode,
 'new_model_calls':0,'changed_control_scope_retained':str(context),'original_context_changed':False},indent=2)+'\n')
print(json.dumps({'cold_context':cold,'changed_payload_rejected':True}))
