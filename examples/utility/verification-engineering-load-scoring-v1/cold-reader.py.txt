"""Verify recorded development work and scope correction without models or private results."""
import sys,json,hashlib,os,socket,subprocess,time
from pathlib import Path
root=Path(sys.argv[1]).resolve()
expected=json.loads((root.parent/'expected-public.json').read_text())
old=(root.parent/'executed-source/engineering_load_scoring.py.txt').read_bytes()
new=(root.parent/'current-kernel.py.txt').read_bytes()
assert hashlib.sha256(old).hexdigest()==expected['executed_kernel_sha256']
assert hashlib.sha256(new).hexdigest()==expected['current_kernel_sha256']
assert old.replace(b'explicit compute_totals calls, not all Newton Jacobian operations',b'component _linearize invocations, including Newton linearization; not FLOPs')==new
inodes={(p.stat().st_dev,p.stat().st_ino) for p in root.rglob('*') if p.is_file()}
def denied(*args,**kwargs):raise RuntimeError('process network or clock denied')
subprocess.Popen=socket.socket=time.time=time.monotonic=denied

def audit(event,args):
    if event=='open':
        path,mode,flags=args
        if flags&(os.O_WRONLY|os.O_RDWR|os.O_CREAT|os.O_TRUNC|os.O_APPEND):raise RuntimeError('write denied')
        if isinstance(path,int):
            st=os.fstat(path)
            if (st.st_dev,st.st_ino) not in inodes:raise RuntimeError('outside descriptor denied')
        elif not Path(os.fsdecode(path)).absolute().is_relative_to(root):raise RuntimeError('outside read denied')
    if event=='import' or event.startswith(('socket.','subprocess.','os.exec','os.fork','os.posix_spawn')):
        raise RuntimeError('import or execution denied')
sys.addaudithook(audit)
def load(name):return json.loads((root/name).read_text())
def digest(value):return hashlib.sha256(json.dumps(value,sort_keys=True,separators=(',',':'),allow_nan=False).encode()).hexdigest()
for name,sha in expected['files'].items():assert hashlib.sha256((root/name).read_bytes()).hexdigest()==sha
q=load('request.json');r=load('result.json');work=load('case-work.json');correction=load('counter-scope-correction.json')
assert digest(q)==expected['request_digest']==r['request_digest']==load('execution-intent.json')['request_digest']
assert digest(r)==expected['result_digest']
assert q['pins']['files']['examples/utility/engineering_load_scoring.py']==expected['executed_kernel_sha256']
assert q['model_starts_cap']==q['policy']['maximum_cases']==24
assert q['solver_iterations_per_case_cap']==q['policy']['fixed_solver_maximum_iterations']==100
assert r['attempted_model_starts']==r['completed_model_starts']==len(work)==24
assert r['score_status']=='success' and r['feasible'] and all(w['required_checks_passed'] for w in work)
for name,value in r['inner_work'].items():assert sum(w['work'][name] for w in work)==value
assert all(w['work']['solver_iterations']<=100 for w in work)
assert correction['original_private_completion_digest']==r['complete_private_digest']
assert correction['derivative_counter_value']==r['inner_work']['derivative_calls']==1950
assert correction['numeric_results_changed'] is False and correction['new_model_starts']==0
assert all(r[k] is False for k in ('task_admitted','private_boundary_qualified','independent_confirmation','model_outputs_exported'))
assert not any(k in sys.modules for k in ('numpy','openmdao','scipy'))
print(json.dumps({'saved_development_work_verified':True,'completed_model_starts':24,'solver_iterations':390,
                  'component_calls':6360,'component_linearize_calls':1950,'counter_scope_correction_verified':True,
                  'new_models':0,'private_source_or_model_output_reads':0,'processes':0,'network':0,'task_admitted':False}))
