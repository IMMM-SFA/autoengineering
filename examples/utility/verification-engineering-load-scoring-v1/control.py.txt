"""One capped development evaluation on already exposed control inputs, no private service admission."""
import sys,hashlib,json,platform
from pathlib import Path
from importlib.metadata import version
sys.path.insert(0,str(Path(__file__).resolve().parents[3]))
from autoengineering.campaign.serialization import digest,unique_json
from autoengineering.campaign.store import checked_directory,durable_write
from autoengineering.utility.store import SuiteStore
from autoengineering.campaign.artifacts import reference
from examples.utility import engineering_load_scoring as kernel,engineering_load as load
from examples.utility import engineering_scenarios as source
ROOT=Path(__file__).resolve().parents[3]
INPUT=ROOT/'outputs/engineering-source-binding-v1'
FILES=('examples/utility/engineering_load_scoring.py','examples/utility/engineering_load.py',
       'examples/utility/engineering.py','examples/interoperability/circuit/model.py',
       'examples/interoperability/circuit/fixture.py','src/autoengineering/adapters/openmdao.py',
       'src/autoengineering/adapters/python.py')
def pins():
    return {'python':platform.python_version(),'packages':{n:version(n) for n in ('openmdao','numpy','scipy','networkx')},
            'files':{n:hashlib.sha256((ROOT/n).read_bytes()).hexdigest() for n in FILES},
            'controller':hashlib.sha256(Path(__file__).read_bytes()).hexdigest()}
def save(root,name,value):durable_write(root/name,source.canonical_json(value).encode())
phase,output=sys.argv[1],Path(sys.argv[2]).absolute()
if phase=='prepare':
    output.mkdir(mode=0o700)
    metadata=unique_json((INPUT/'public/manifest.json').read_text())
    current=.05
    context=load.context(current,output/'work',source_root=ROOT)
    policy=kernel.LoadScorePolicy('utility-coupled-load-score-v1',context.model.digest,kernel.source_identity(),.62)
    request={'scope':'Previously exposed generated control inputs; fixed-current development kernel only',
             'scope_digest':digest(str(output)),'policy':policy.to_dict(),'pins':pins(),
             'input_manifest_digest':digest(metadata),'input_private_digest':metadata['private_record_digest'],
             'source_current_A':current,'model_starts_cap':24,'solver_iterations_per_case_cap':100,
             'automatic_retries':0,'task_admitted':False,'private_boundary_qualified':False,
             'os_deadline_qualified':False,'aggregate_rss_qualified':False}
    save(output,'request.json',request)
    print(json.dumps({'status':'prepared','request_digest':digest(request)}))
else:
    checked_directory(output)
    request=unique_json((output/'request.json').read_text())
    assert digest(request)==sys.argv[3] and request['scope_digest']==digest(str(output)) and request['pins']==pins()
    assert not (output/'execution-intent.json').exists()
    policy=kernel.LoadScorePolicy.from_dict(request['policy'])
    store=SuiteStore(INPUT/'suite')
    assert unique_json((INPUT/'public/manifest.json').read_text())['private_record_digest']==request['input_private_digest']
    save(output,'execution-intent.json',{'request_digest':digest(request),'model_starts_reserved':24,'automatic_retries':0})
    rows=source.read_private(store,INPUT/'public',INPUT/'private',expected_manifest_digest=request['input_manifest_digest'],
                            expected_selection_digest=store.replay()['frozen'])
    private=output/'private-results';private.mkdir(mode=0o700)
    print('Evaluating fixed design on 24 already exposed development cases',flush=True)
    score=kernel.score_loads(request['source_current_A'],rows,private,ROOT,policy=policy,execution_digest=digest(request))
    complete=unique_json((private/'complete.json').read_text()) if (private/'complete.json').exists() else None
    failure=unique_json((private/'failed.json').read_text()) if (private/'failed.json').exists() else None
    result={'request_digest':digest(request),'score_status':score.status,'feasible':score.feasible,
            'complete_private_digest':digest(complete) if complete else None,'failure_private_digest':digest(failure) if failure else None,
            'attempted_model_starts':(complete or failure)['attempted_model_starts'],
            'completed_model_starts':(complete or failure)['completed_model_starts'],
            'inner_work':complete['inner_work'] if complete else None,
            'task_admitted':False,'private_boundary_qualified':False,'independent_confirmation':False,
            'model_outputs_exported':False,'runtime':request['pins']['packages']}
    save(output,'result.json',result)
    print(json.dumps(result))
