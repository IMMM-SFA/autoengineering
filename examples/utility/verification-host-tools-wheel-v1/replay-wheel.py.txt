"""Fresh isolated CLI and metadata readers outside the checkout; no installation."""
from pathlib import Path
import json,os,shutil,subprocess,sys,tarfile,tempfile,zipfile
root=Path.cwd().resolve();out=root/'examples/utility/verification-host-tools-wheel-v1';cases=[]
with tempfile.TemporaryDirectory(prefix='host-tools-wheel-',dir='/private/tmp') as temporary:
 work=Path(temporary).resolve();package=work/'package';package.mkdir(mode=0o700)
 with zipfile.ZipFile(next(out.glob('*.whl'))) as archive:archive.extractall(package)
 modules=['examples.utility.guest_image','examples.utility.guest_image_context','examples.utility.guest_archive_download','examples.utility.guest_image_build','autoengineering.cli']
 for number,name in enumerate(modules):
  code="import sys,importlib;from pathlib import Path;sys.path.insert(0,sys.argv[1]);m=importlib.import_module(sys.argv[2]);assert Path(m.__file__).resolve().is_relative_to(Path(sys.argv[1]));sys.argv=[sys.argv[2],'--help'];m.cli(['--help']) if hasattr(m,'cli') else m.main()"
  result=subprocess.run([sys.executable,'-I','-c',code,str(package),name],cwd=work,capture_output=True,text=True,env={**os.environ,'PYTHONDONTWRITEBYTECODE':'1'})
  (out/f'cli-{number}.log').write_text(result.stdout+result.stderr)
  if result.returncode:raise RuntimeError(result.stderr)
  assert 'Usage' in result.stdout or 'usage' in result.stdout
  cases.append({'case':name+' help','exit_code':result.returncode,'isolated':True,'code_from_extracted_wheel':True})
 saved=work/'saved';saved.mkdir(mode=0o700)
 shutil.copytree(root/'outputs/guest-runtime-archives-v1',saved/'cache')
 shutil.copytree(root/'outputs/guest-build-context-v1',saved/'context')
 authored=saved/'authored';authored.mkdir(mode=0o700)
 with tarfile.open(out/'authored-build-metadata.tar.gz') as archive:archive.extractall(authored,filter='data')
 for path in authored.rglob('*'):path.chmod(0o700 if path.is_dir() else 0o600)
 shutil.copyfile(out/'cold-reader.py.txt',work/'reader.py')
 shutil.copyfile(out/'expected-build.json',work/'expected.json')
 result=subprocess.run([sys.executable,'-I',str(work/'reader.py'),str(package),str(saved),str(work/'expected.json')],cwd=work,capture_output=True,text=True,env={**os.environ,'PYTHONDONTWRITEBYTECODE':'1'})
 (out/'cold-reader.log').write_text(result.stdout+result.stderr)
 if result.returncode:raise RuntimeError(result.stderr)
 print(result.stdout.strip())
 cases.append({'case':'guarded cache/context/authored-build replay','exit_code':result.returncode,'isolated':True,'code_from_extracted_wheel':True})
(out/'reader-results.json').write_text(json.dumps(cases,indent=2)+'\n')
print('Five isolated CLI help checks and three guarded saved-result readers passed outside the checkout.')
