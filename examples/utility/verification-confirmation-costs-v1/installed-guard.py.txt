import sys
from pathlib import Path
blocked = Path("/Users/cam/projects/autoengineering")
allowed = Path('/Users/cam/Library/Caches/rattler/cache/envs/autoengineering-2706574624935427595/envs/default')
def audit(event,args):
 if event == "open" and isinstance(args[0],(str,bytes)):
  value=Path(args[0].decode() if isinstance(args[0],bytes) else args[0]).resolve()
  if value.is_relative_to(blocked) and not value.is_relative_to(allowed):
   raise RuntimeError("offline guard denied original checkout access")
 if event in ("subprocess.Popen","os.system","os.posix_spawn","socket.__new__","socket.connect","socket.bind"):
  raise RuntimeError("offline guard denied process or socket")
sys.addaudithook(audit)
