"""Run the installed reader and reject removed corner work."""
from pathlib import Path
import json
import shutil
import subprocess
import sys

source = Path(sys.argv[1]).resolve()
out = source / 'examples/utility/verification-engineering-envelope-registration-v1'
stage = Path('/private/tmp/utility-engineering-envelope-installed-v1')
assert not Path.cwd().resolve().is_relative_to(source)
wrong = stage / 'wrong'
shutil.copytree(stage / 'public', wrong)
path = wrong / 'registration-result.json'
record = json.loads(path.read_text()); record['support_solves'] = []
path.write_text(json.dumps(record))
def run(public):
    return subprocess.run([sys.executable, '-I', '-B', str(out / 'read-installed.py.txt'),
                           str(stage / 'installed'), str(public)], capture_output=True, text=True)
good = run(stage / 'public')
assert good.returncode == 0, good.stderr
bad = run(wrong)
assert bad.returncode != 0 and 'native support solves need unique bounded identities' in bad.stderr
result = {'schema': 'utility-engineering-envelope-installed-replay-v1',
          'outside_checkout': True, 'relocated_result': 'passed',
          'dropped_support_call': 'rejected', 'reader': json.loads(good.stdout),
          'actual_full_job_source_closure_qualified': False}
(out / 'reader-verification.json').write_text(json.dumps(result, indent=2) + '\n')
print(json.dumps(result))
