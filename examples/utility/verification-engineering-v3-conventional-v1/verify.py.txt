"""Replay public native-profile receipts; never invoke a driver or model."""
from pathlib import Path
import builtins,hashlib,json,math,socket,subprocess,sys
original_import=builtins.__import__
def guarded_import(name,*args,**kwargs):
    if name.split('.')[0] in {'examples','openmdao','py_wake','pvlib','torch','botorch','tensorflow'}:
        raise AssertionError('scientific import denied')
    return original_import(name,*args,**kwargs)
def denied(*a,**k):raise AssertionError('external execution denied')
builtins.__import__=guarded_import;socket.socket=denied;subprocess.Popen=denied
import autoengineering
from autoengineering.agents.batch import BatchStore
from autoengineering.campaign.artifacts import read_bytes
from autoengineering.campaign.records import ArtifactRef
from autoengineering.campaign.serialization import digest,unique_json
root=Path(sys.argv[1]).resolve() if len(sys.argv)>1 else Path(__file__).resolve().parent
package=Path(autoengineering.__file__).resolve().parent
initializing=True
def audit(event,args):
    if event.startswith(('socket.','subprocess.')):raise AssertionError('external execution denied')
    if event=='open':
        name,mode,flags=args
        if isinstance(name,(str,bytes)):
            path=Path(name).resolve()
            if not path.is_relative_to(root) and not (initializing and path.is_relative_to(package)):
                raise AssertionError('outside public archive read denied')
            if mode and any(c in mode for c in 'wax+'):raise AssertionError('archive write denied')
sys.addaudithook(audit)
def load(path):return unique_json((root/path).read_text())
review=load('profile-review.json');proposal=load('definition/proposal.json');inputs=load('definition/development-inputs.json')
stores=[BatchStore(root/f'profile-{n}/batch') for n in (1,2,3)]
states=[store.replay() for store in stores]
initializing=False
assert [len(s.reservations) for s in states]==[1,10,35]
assert [len(s.observations) for s in states]==[1,10,35]
assert [s.finished['conventional'].status for s in states]==['failed','failed','passed']
assert all(not s.pending and s.active_case is None for s in states)
assert sum(len(s.reservations) for s in states)==review['native_model_invocations_charged']==46<=64
assert next(iter(states[0].observations.values())).status=='scientific_infeasible'
activity=0;retained=0
for index,(store,state) in enumerate(zip(stores,states),1):
    request=load(f'profile-{index}/request.json')
    assert request['proposal_digest']==digest(proposal) and request['input_digest']==digest(inputs)
    assert request['batch_spec_digest']==store.spec.digest
    for observation in state.observations.values():
        assert observation.stop_verified and len(observation.evidence)==1
        row=unique_json(read_bytes(store.root,observation.evidence[0]).decode())
        assert row['request_digest']==digest(request)
        before=row['counter_before'];after=row['counter_after']
        delta={key:after[key]-before[key] for key in before}
        assert all(type(v) is int and v>=0 for v in delta.values()) and delta==row['counter_delta']
        touched=any(delta.values());activity+=touched;retained+=not touched
        if index>1:
            work=row['pass_work']
            assert work['circuit_work_observed']==touched and work['design_eligibility']=='not_assessed'
            assert work['component_calls']==delta['component_calls'] and work['component_linearize_calls']==delta['component_linearize_calls']
            assert work['nonlinear_iterations_this_pass']==(row['work']['solver_iterations'] if touched else 0)
assert activity==review['observed_circuit_work_passes']==43
assert retained==review['retained_circuit_state_passes']==3
summary=load('profile-3/summary.json');recipe=load('definition/conventional-recipe-v3.json')
assert summary['recipe_digest']==digest(recipe) and recipe['target_proposal_digest']==digest(proposal)
ref=ArtifactRef.from_dict(summary['receipt']);assert ref in states[2].finished['conventional'].evidence
receipt=unique_json(read_bytes(stores[2].root,ref).decode())
assert receipt['recipe_digest']==digest(recipe) and receipt['proposal_digest']==digest(proposal)
assert receipt['native_design']['driver_success'] is True
current=receipt['selected_current_A'];assert .01<=current<=.1
assert len(receipt['public_checks'])==24
by_case={row['scenario_id']:row for row in inputs['cases']}
for row in receipt['public_checks']:
    assert row['forcing_digest']==digest(by_case[row['scenario_id']]['forcing'])
checks=[*receipt['public_checks'],*receipt['native_design']['scenario_solves']]
for row in checks:
    outputs=row['outputs'];work=row['work']
    assert all(math.isfinite(v) for v in outputs.values()) and work['converged']
    assert work['residual_norm']<=1e-10 and work['residual_tolerance']==1e-10 and work['solver_iterations']<=100
    assert outputs['node2_voltage']>=.62
corner=receipt['native_design']['scenario_solves'][0]['outputs']['node2_voltage']
assert (corner>=.62001)==summary['native_optimizer_stricter_target_feasible']==review['exact_buffered_target_check']==False
baseline_source=load('baseline/source.json');baseline_ref=load('baseline/development-reference.json')
raw=(root/'baseline/receipt.json').read_bytes()
assert hashlib.sha256(raw).hexdigest()==baseline_source['execution_artifact']['sha256']==baseline_ref['execution_receipt_digest']
baseline=unique_json(raw.decode());assert baseline['proposal_digest']==digest(proposal) and baseline['input_digest']==digest(inputs)
base={row['scenario_id']:row['source_power_W'] for row in baseline['case_results']}
powers=[]
for row in receipt['public_checks']:
    case=by_case[row['scenario_id']]
    value=current*(row['outputs']['node1_voltage']-case['forcing']['ground_voltage'])
    assert value==row['source_power_W'] and 0<value<=1.01*base[row['scenario_id']]
    powers.append(value)
mean=math.fsum(powers)/24
assert mean==receipt['mean_source_power_W']==summary['mean_source_power_W']==review['public_mean_source_power_W']
assert baseline['mean_source_power_W']==receipt['baseline_source_power_W']
assert baseline['mean_source_power_W']-mean==receipt['development_gain_W']==review['development_gain_W']
rows=receipt['solve_records'];assert len(rows)==35
counts=receipt['counts'];assert counts['problem_instances']==counts['cleanups_returned']==26
assert sum(row['pass_work']['nonlinear_iterations_this_pass'] for row in rows)==receipt['nonlinear_solver_iterations']==480
for key in ('component_calls','component_linearize_calls'):
    total=sum(row[key] for row in receipt['problem_totals'])
    inner=sum(row['counter_delta'][key] for row in rows)
    assert total==receipt['counter_totals_per_problem'][key]
    assert total-inner==receipt['between_solve_counter_deltas'][key]>=0
assert review['failed_profiles_complete_counter_totals'] is None
assert review['complete_cost_evidence'] is False and review['task_admitted'] is False and review['utility_verdict']=='not_attempted'
print(json.dumps({'native_invocations_charged':46,'circuit_work_passes':43,'retained_state_passes':3,'selected_current_A':current,
    'public_mean_source_power_W':mean,'development_gain_W':receipt['development_gain_W'],'exact_buffered_target_check':False,
    'scientific_minimum_voltage_checks':True,'new_models_or_driver_calls':0,'task_admitted':False}))
