"""Check installed metadata replay and rejection of dropped failed-profile charges."""
from pathlib import Path
import hashlib,json,shutil,subprocess,sys,zipfile
source,installed,target=map(lambda x:Path(x).resolve(),sys.argv[1:])
archive=source/'examples/utility/verification-engineering-v3-conventional-v1'
assert not target.exists();target.mkdir(mode=0o700)
public=target/'public';wrong=target/'wrong'
shutil.copytree(archive,public);shutil.copytree(archive,wrong)
record=json.loads((wrong/'profile-review.json').read_text());record['native_model_invocations_charged']=35
(wrong/'profile-review.json').write_text(json.dumps(record,sort_keys=True,indent=2)+'\n')
wheel=source/'examples/utility/verification-wind-cost-report-v1/autoengineering-0.2.0-py3-none-any.whl'
assert hashlib.sha256(wheel.read_bytes()).hexdigest()=='72a1fda0bf81f8160e209265ba4f469fd50dbf9ebbb19a71fc1f43a38e0edc57'
verified=0
with zipfile.ZipFile(wheel) as z:
    for name in z.namelist():
        if name.startswith('autoengineering/') and not name.endswith('/'):
            assert (installed/name).read_bytes()==z.read(name);verified+=1
code='import sys,runpy; installed,script,root=sys.argv[1:]; sys.path.insert(0,installed); sys.argv=[script,root]; runpy.run_path(script,run_name="__main__"); from pathlib import Path; assert all(Path(module.__file__).resolve().is_relative_to(Path(installed)) for name,module in tuple(sys.modules.items()) if (name=="autoengineering" or name.startswith("autoengineering.")) and getattr(module,"__file__",None))'
assert not Path.cwd().resolve().is_relative_to(source)
script=archive/'verify.py.txt'
def run(root):return subprocess.run([sys.executable,'-I','-B','-c',code,str(installed),str(script),str(root)],capture_output=True,text=True)
good=run(public);assert good.returncode==0,good.stderr
bad=run(wrong);assert bad.returncode!=0 and 'AssertionError' in bad.stderr
result={'schema':'utility-engineering-v3-conventional-installed-reader-v1','outside_checkout':True,'relocated_evidence':'passed',
    'all_loaded_core_modules_from_installed_target':True,'installed_package_payloads_verified':verified,'wheel_sha256':hashlib.sha256(wheel.read_bytes()).hexdigest(),
    'failed_profile_charge_removal':'rejected','reader_result':json.loads(good.stdout),'wheel_rebuilt':False,'dependencies_installed':False,
    'new_native_driver_invocations':0,'new_circuit_or_reference_solves':0}
(archive/'reader-verification.json').write_text(json.dumps(result,sort_keys=True,indent=2)+'\n')
print(json.dumps(result))
