from pathlib import Path
import json,os,subprocess,sys
root=Path.cwd();archive=root/'examples/utility/verification-engineering-v3-conventional-v1'
nodeids=json.loads((archive/'sandbox-regression-limits.json').read_text())['failed_nodeids']
command=[sys.executable,'-B','-m','pytest',*nodeids,'-q','--tb=short','-p','no:cacheprovider','--basetemp=/private/tmp/engineering-v3-conventional-native-tests-v1','--junitxml=/private/tmp/engineering-v3-conventional-native-tests-v1.xml']
env=dict(os.environ);env['PATH']=str(Path(sys.executable).parent)+os.pathsep+env.get('PATH','')
with (archive/'process-tests.log').open('x') as output:
    result=subprocess.run(command,env=env,stdout=output,stderr=subprocess.STDOUT,text=True)
print(json.dumps({'returncode':result.returncode,'nodeids':len(nodeids),'log':'process-tests.log'}),flush=True)
sys.exit(result.returncode)
