"""Recheck exactly the confirmed launcher-path failure without changing source."""
from pathlib import Path
from datetime import datetime,timezone
import hashlib,json,os,shutil,subprocess,sys
import xml.etree.ElementTree as E
root=Path.cwd();out=root/'examples/utility/verification-current-base-v6'
completion=json.loads((out/'completion.json').read_text())
assert completion['exit_code']==1 and completion['changed_pins']==[]
xml=E.parse(out/'junit.xml').getroot()
failed=[]
for case in xml.iter('testcase'):
    failure=case.find('failure')
    if failure is not None:
        assert (case.get('classname'),case.get('name'))==('tests.test_research.TestRunComponent','test_command_kind_npz_contract')
        failed.append('tests/test_research.py::TestRunComponent::test_command_kind_npz_contract')
        assert '127' in failure.get('message','')
assert failed==['tests/test_research.py::TestRunComponent::test_command_kind_npz_contract']
assert '/bin/sh: python: command not found' in (out/'pytest.log').read_text()
run_env=os.environ.copy();run_env['PATH']=str(Path(sys.executable).parent)+os.pathsep+run_env.get('PATH','')
resolved=shutil.which('python',path=run_env['PATH'])
assert resolved is not None and Path(resolved).resolve()==Path(sys.executable).resolve()
command=[sys.executable,'-B','-X','pycache_prefix=/private/tmp/utility-path-recheck-v6-cache','-m','pytest',
         *failed,'-q','-p','no:cacheprovider','--basetemp=/private/tmp/utility-path-recheck-v6',
         '--junitxml='+str(out/'path-recheck.xml'),'--tb=short']
request={'command':command,'child_python':resolved,'child_path_prefix':str(Path(sys.executable).parent),
 'fresh_bytecode_lookup_prefix':True,'source_changes':False,'full_suite_rerun':False,
 'reason':'Terminal full-suite failure confirmed bare python command was absent',
 'created_at_utc':datetime.now(timezone.utc).isoformat()}
(out/'path-recheck-request.json').write_text(json.dumps(request,indent=2)+'\n')
p=subprocess.run(command,cwd=root,env=run_env,capture_output=True,text=True)
(out/'path-recheck.stdout.txt').write_text(p.stdout);(out/'path-recheck.stderr.txt').write_text(p.stderr)
pins=json.loads((out/'source-pins.json').read_text())
changed=[n for n,sha in pins.items() if not (root/n).is_file() or hashlib.sha256((root/n).read_bytes()).hexdigest()!=sha]
result={'exit_code':p.returncode,'rechecked_node_ids':failed,'changed_pins':changed,
 'original_full_run_preserved':True,'test_and_production_source_unchanged':True}
(out/'path-recheck-result.json').write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps(result))
assert p.returncode==0 and changed==[]
