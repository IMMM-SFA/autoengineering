"""Retain the authored guest composite public suite and original score/costs."""
from pathlib import Path
import json,sys,tarfile
from autoengineering.utility.store import SuiteStore
from autoengineering.utility.costs import join_confirmation_costs
from autoengineering.utility.scoring import read_receipt
origin=Path(sys.argv[1]).resolve()
store=SuiteStore(origin);state=store.replay()
assert len(state['confirmation_scores'])==1
score_id=next(iter(state['confirmation_scores'].values()))
receipt=read_receipt(origin,score_id)
assert receipt.score.status=='success'
for path in origin.rglob('*'):
 if path.is_file():
  assert path.name not in {'labels.json','drivers.npz','guest-inputs.tar','private-labels.json'}
  assert b'AUTHORED-PRIVATE-CANARY' not in path.read_bytes()
bundle=Path(__file__).parent
with tarfile.open(bundle/'authored-guest-composite-public.tar.gz','w:gz') as archive:
 archive.add(origin,arcname='public')
expected=dict(score_id=score_id,receipt=receipt.to_dict(),
 costs={row.resource:row.quantity for row in join_confirmation_costs(store)})
(bundle/'expected-public.json').write_text(json.dumps(expected,indent=2)+'\n')
print('Original authored guest composite public score and costs retained.')
