@contextmanager
def prepared(tmp_path, metadata, *, worker_status="success"):
    with fixture.prepared(tmp_path, metadata, authorize=False) as (executor, original, peer, _):
        private = tmp_path / "private-scoring"
        private.mkdir(mode=0o700)
        clock = original.inference.forcing.ports[0].time
        (private / "labels.json").write_text(
            json.dumps(
                {
                    "labels": [0.0] * len(clock.dates),
                    "mask": list(clock.scoring_mask),
                }
            )
        )
        labels = reference(private, "labels.json", media_type="application/json", producer="author")
        outcome = OutcomeSpec(
            "engineering-outcome-v1", "rmse", "mm/day", "minimize", "mean", "1" * 64
        )
        runtime = RecordedAuthoredRuntime()
        scorer = GuestCompositeRMSEScorer(
            executor,
            private,
            labels=labels,
            runtime=runtime,
            inference=original.inference,
            partition_digest=original.partition_digest,
            outcome_contract=outcome,
        )
        grant = replace(original, scorer_digest=scorer.identity)
        service = GuestCompositeConfirmationService(
            executor.store, scorer, scorer_digest=scorer.identity, outcome_contract=outcome
        )

        def configure(stage):
            if stage == "guest_executor_inputs_ready":
                owner = GuestOwnerJournal(
                    executor.work_root / grant.digest, suite_root=executor.store.root
                )
                peer.daemon = fixture.fixture.daemon_fixture.Daemon(owner)
                peer.packet = fixture.fixture.module.encode_request(owner)
                peer.response, _ = fixture.fixture.bootstrap_fixture.response(
                    tmp_path, owner, status=worker_status
                )

            elif stage == "composite_inference_committed":
                # The original inference/guard have finished. Close the authored
                # HTTP threads before the fixed score worker exercises fork().
                peer.close()

        executor.store.fault = configure
        yield executor, grant, peer, scorer, service, runtime, configure


def test_guest_chain_preserves_parent_grant_and_both_container_counts(tmp_path, metadata):
    with prepared(tmp_path, metadata) as (executor, grant, peer, scorer, service, runtime, _):
        result = service.confirm(grant)
        assert result.score.status == "success" and result.score.value == pytest.approx(1.0)
        assert peer.starts == peer.inputs == runtime.starts == 1
        assert isinstance(result.composite.registration, GuestCompositeScorerRegistration)
        assert (
            result.composite.registration.digest
            == result.measurement.scorer_digest
            == grant.scorer_digest
        )
        child = ConfirmationGrant.from_json(
            read_bytes(executor.store.root, result.composite.inner_grant).decode()
        )
        assert child.scorer_digest != grant.scorer_digest and child.scientific_starts == 1
        assert result.composite.inner_execution.grant_digest == child.digest
        state = executor.store.replay()
        assert state["phase_used"][("fixed_evaluations", "confirmation", "scientific_starts")] == 2
        costs = {row.resource: row.quantity for row in join_confirmation_costs(executor.store)}
        assert costs["confirmation_inference_starts"] == costs["confirmation_scoring_starts"] == 1
        assert (
            costs["confirmation_inference_container_starts"]
            == costs["confirmation_scoring_container_starts"]
            == 1
        )
        assert costs["confirmation_container_starts"] == 2
        assert costs["confirmation_container_lifecycle_duration"] > 0.1
        assert "private-scoring" not in result.to_json() and "labels.json" not in result.to_json()
        assert service.confirm(grant) == result
        assert peer.starts == runtime.starts == 1
        copied = tmp_path / "public-copy"
        shutil.copytree(executor.store.root, copied)
        for path in (executor.work_root, executor.driver_root, scorer.private_root):
            shutil.rmtree(path)
        replayed = SuiteStore(copied)
        assert replayed.replay() == state
        assert {r.resource: r.quantity for r in join_confirmation_costs(replayed)} == costs


@pytest.mark.parametrize("kind", ["vector", "local_service", "local_scorer"])
def test_guest_profile_requires_explicit_guest_service_before_access(tmp_path, metadata, kind):
    with prepared(tmp_path, metadata) as (executor, grant, peer, scorer, _, runtime, _):
        before = executor.store.replay()
        if kind == "vector":
            other = ConfirmationService(
                executor.store,
                scorer,
                scorer_digest=scorer.identity,
                outcome_contract=scorer.outcome_contract,
            )
        elif kind == "local_service":
            other = CompositeConfirmationService(
                executor.store,
                scorer,
                scorer_digest=scorer.identity,
                outcome_contract=scorer.outcome_contract,
            )
        else:
            with pytest.raises(ValueError, match="cannot admit a guest"):
                CompositeRMSEScorer(
                    executor,
                    scorer.private_root,
                    labels=scorer.labels,
                    runtime=runtime,
                    inference=grant.inference,
                    partition_digest=grant.partition_digest,
                    outcome_contract=scorer.outcome_contract,
                )
            assert executor.store.replay() == before and not peer.calls
            return
        with pytest.raises(ValueError):
            other.confirm(grant)
        assert executor.store.replay() == before and not peer.calls and runtime.starts == 0


@pytest.mark.parametrize("point", ["composite_inference_committed", "confirmation_score_created"])
def test_recovery_never_starts_a_missing_inner_score(tmp_path, metadata, point):
    with prepared(tmp_path, metadata) as (executor, grant, peer, _, service, runtime, configure):

        def fail(stage):
            configure(stage)
            if stage == point:
                raise KeyboardInterrupt("authored composite stage loss")

        executor.store.fault = fail
        with pytest.raises(KeyboardInterrupt):
            service.confirm(grant)
        calls = tuple(peer.calls)
        costs = {r.resource: r.quantity for r in join_confirmation_costs(executor.store)}
        assert costs["confirmation_inference_container_starts"] == 1
        executor.store.fault = lambda stage: None
        if point == "composite_inference_committed":
            assert runtime.starts == 0
            assert costs["confirmation_scoring_container_starts"] is None
            assert costs["confirmation_container_starts"] is None
            with pytest.raises(ValueError):
                service.recover(grant)
            assert runtime.starts == 0
        else:
            assert runtime.starts == 1
            recovered = service.recover(grant)
            assert recovered.score.value == pytest.approx(1.0) and recovered.recovery
            assert runtime.starts == 1
        assert peer.starts == 1 and tuple(peer.calls) == calls


@pytest.mark.parametrize(
    "status", ["scientific_infeasible", "model_failure", "infrastructure_failure"]
)
def test_failed_inference_publishes_failure_without_starting_private_score(
    tmp_path, metadata, status
):
    with prepared(tmp_path, metadata, worker_status=status) as (
        executor,
        grant,
        peer,
        _,
        service,
        runtime,
        _,
    ):
        receipt = service.confirm(grant)
        assert receipt.score.status == status and receipt.score.value is None
        assert receipt.composite.inner_grant is None and receipt.composite.inner_execution is None
        assert peer.starts == 1 and runtime.starts == 0
        transition, (_, checked, _, _) = read_stage_transition(executor.store, grant)
        assert checked.status == status and transition.inner_grant is None
        costs = {r.resource: r.quantity for r in join_confirmation_costs(executor.store)}
        assert costs["confirmation_container_starts"] == 1
        assert (
            costs["confirmation_scoring_starts"]
            == costs["confirmation_scoring_container_starts"]
            == 0
        )


def test_cancelled_stage_boundary_never_reads_labels_or_starts_score(
    tmp_path, metadata, monkeypatch
):
    from autoengineering.utility import container_scoring

    with prepared(tmp_path, metadata) as (
        executor,
        grant,
        peer,
        scorer,
        service,
        runtime,
        configure,
    ):
        original = container_scoring.read_bytes
        private_reads = 0

        def guard(root, ref, **kwargs):
            nonlocal private_reads
            if root == scorer.private_root:
                private_reads += 1
                raise AssertionError("cancelled stage cannot open labels")
            return original(root, ref, **kwargs)

        def cancel(stage):
            configure(stage)
            if stage == "composite_inference_committed":
                executor.store.append(
                    request_id="authored-stage-cancel",
                    operation="cancel",
                    cell_digest=grant.cell_digest,
                    attempt_id="authored-cancel",
                    child_digest=grant.digest,
                    phase="confirmation",
                )

        executor.store.fault = cancel
        monkeypatch.setattr(container_scoring, "read_bytes", guard)
        with pytest.raises(ValueError):
            service.confirm(grant)
        assert (
            executor.store.replay()["cancelled"]
            and not executor.store.replay()["confirmation_scores"]
        )
        assert peer.starts == 1 and runtime.starts == private_reads == 0
        with pytest.raises(ValueError):
            service.recover(grant)
        assert runtime.starts == private_reads == 0


def test_changed_live_guest_registration_is_refused_before_reservation(tmp_path, metadata):
    with prepared(tmp_path, metadata) as (executor, grant, peer, _, service, runtime, _):
        before = executor.store.replay()
        executor.api.profile = replace(executor.api.profile, server_digest="2" * 64)
        with pytest.raises(ValueError, match="registration changed"):
            service.confirm(grant)
        assert executor.store.replay() == before
        assert not peer.calls and runtime.starts == 0


@pytest.mark.parametrize("member", ["registration", "marker"])
def test_public_guest_handoff_refuses_altered_initialization(tmp_path, metadata, member):
    with prepared(tmp_path, metadata) as (executor, grant, _, _, service, _, _):
        service.confirm(grant)
        destination = tmp_path / "tampered-public"
        shutil.copytree(executor.store.root, destination)
        if member == "marker":
            (destination / "guest-executor-inits" / (grant.digest + ".complete.json")).unlink()
        else:
            path = destination / "guest-executor-inits" / (grant.digest + ".json")
            doc = json.loads(path.read_text())
            doc["registration"]["implementation_digest"] = "2" * 64
            path.write_text(json.dumps(doc))
        with pytest.raises(ValueError):
            SuiteStore(destination).replay()
