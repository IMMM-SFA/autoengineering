"""Build the current core-only wheel and install into an isolated offline target."""
from pathlib import Path
import hashlib, json, os, shutil, subprocess, sys, zipfile
root=Path.cwd();out=root/'examples/utility/verification-load-cost-report-v1'
stage=Path('/private/tmp/utility-load-cost-report-v1');stage.mkdir(mode=0o700)
source=stage/'source';source.mkdir(mode=0o700)
old=json.loads((root/'examples/utility/verification-engineering-load-costs-v1/wheel-input-pins.json').read_text())
names={n for n in old if not n.startswith('examples/')}
names.update(str(p.relative_to(root)) for p in (root/'src/autoengineering').rglob('*.py'))
pins={n:hashlib.sha256((root/n).read_bytes()).hexdigest() for n in sorted(names)}
for name in names:
    dest=source/name;dest.parent.mkdir(mode=0o700,parents=True,exist_ok=True);shutil.copyfile(root/name,dest)
(out/'wheel-input-pins.json').write_text(json.dumps(pins,indent=2)+'\n')
env=os.environ.copy();env['SOURCE_DATE_EPOCH']='1793318400'
build=[sys.executable,'-I','-B','-c',"from setuptools.build_meta import build_wheel; build_wheel('../wheel')"]
r=subprocess.run(build,cwd=source,env=env,capture_output=True,text=True)
(out/'build.stdout.txt').write_text(r.stdout);(out/'build.stderr.txt').write_text(r.stderr)
assert r.returncode==0
wheel=next((stage/'wheel').glob('*.whl'))
with zipfile.ZipFile(wheel) as z:
    python_files=[n for n in z.namelist() if n.endswith('.py')]
    assert len(python_files)==225 and all(n.startswith('autoengineering/') for n in python_files)
    for name in z.namelist():
        if '.dist-info/' not in name:
            assert hashlib.sha256(z.read(name)).hexdigest()==pins['src/'+name]
saved=out/wheel.name;shutil.copyfile(wheel,saved)
target=stage/'installed'
install=[sys.executable,'-I','-m','pip','install','--no-index','--no-deps','--no-compile','--target',str(target),str(wheel)]
r2=subprocess.run(install,cwd=stage,capture_output=True,text=True)
(out/'install.stdout.txt').write_text(r2.stdout);(out/'install.stderr.txt').write_text(r2.stderr)
assert r2.returncode==0
with zipfile.ZipFile(wheel) as z:
    for name in z.namelist():
        if '.dist-info/' not in name:
            assert (target/name).read_bytes()==z.read(name)
shutil.copytree(out/'public',stage/'public')
result={'wheel_sha256':hashlib.sha256(saved.read_bytes()).hexdigest(),'python_files':225,'example_python_files':0,
        'copied_inputs':len(pins),'all_package_payloads_match_inputs':True,'installed_payloads_match_wheel':True,
        'build_exit_code':r.returncode,'install_exit_code':r2.returncode,'target':str(target),
        'public_scope':str(stage/'public'),'network_or_dependency_installation':False,
        'inherited_native_dependencies':True,'release_or_linux_qualification':False}
(out/'build-install-result.json').write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps(result))
