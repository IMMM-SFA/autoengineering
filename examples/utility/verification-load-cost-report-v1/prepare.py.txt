"""Prepare public report inputs from the preceding verified native archive."""
from pathlib import Path
import hashlib, json, shutil, sys
sys.path.insert(0, str(Path.cwd()))
from autoengineering.utility.load_costs import CostEvidenceSpec, LoadCostInput, LoadCostRegistration
from autoengineering.utility.scoring import ConfirmationGrant
from autoengineering.campaign.records import ArtifactRef
from autoengineering.utility.store import SuiteStore
from autoengineering.utility.report import accounting_report, render_markdown
root=Path.cwd(); out=root/'examples/utility/verification-load-cost-report-v1'
old=root/'examples/utility/verification-engineering-load-costs-v1'
native=root/'examples/utility/verification-engineering-load-callback-v1'
expected=json.loads((native/'expected-public.json').read_text())
for name,sha in expected['files'].items():
    assert hashlib.sha256((native/'public'/name).read_bytes()).hexdigest()==sha
public=out/'public'; public.mkdir(mode=0o700)
shutil.copytree(native/'public/suite',public/'suite')
old_input=json.loads((old/'public/inputs.json').read_text())
grant=ConfirmationGrant.from_dict(old_input['grant'])
registration=LoadCostRegistration.from_dict(old_input['registration'])
work=ArtifactRef.from_dict(old_input['work_artifact'])
inputs=CostEvidenceSpec('utility-cost-inputs-v1',grant.suite_digest,(LoadCostInput(grant,registration,work),))
(public/'cost-inputs.json').write_text(inputs.to_json()+'\n')
store=SuiteStore(public/'suite')
legacy=accounting_report(store); enriched=accounting_report(store,cost_inputs=inputs)
(public/'expected-legacy.json').write_text(json.dumps(legacy,sort_keys=True,separators=(',',':'))+'\n')
(public/'expected-accounting.json').write_text(json.dumps(enriched,sort_keys=True,separators=(',',':'))+'\n')
(public/'expected-accounting.md').write_text(render_markdown(enriched))
pins={str(p.relative_to(public)):hashlib.sha256(p.read_bytes()).hexdigest() for p in sorted(public.rglob('*')) if p.is_file()}
(out/'public-pins.json').write_text(json.dumps(pins,indent=2)+'\n')
print(json.dumps({'public_files':len(pins),'new_native_models':0,'cost_inputs_digest':inputs.digest,'full_cost_evidence':enriched['complete_cost_evidence']}))
