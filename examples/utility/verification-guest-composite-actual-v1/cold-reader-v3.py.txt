import sys
import importlib.abc
from pathlib import Path

class NoModels(importlib.abc.MetaPathFinder):
    def find_spec(self, fullname, path=None, target=None):
        if fullname.split('.')[0] in {'torch','pvlib','openmdao','py_wake','topfarm','bmipy'} or fullname.startswith(('examples.leaf_river','examples.utility.hydrology')):
            raise RuntimeError('model import denied')

sys.meta_path.insert(0, NoModels())
import json
import os
import socket
import subprocess
import time
import numpy
from autoengineering.utility.store import SuiteStore
from autoengineering.utility.costs import join_confirmation_costs
from autoengineering.utility import composite_scoring, guest_archive, guest_receipts, inference_execution
from autoengineering.utility import private_container_runtime, container_scoring, controller, guest_executor
from autoengineering.agents import batch
from autoengineering.utility import baseline_selection
root=Path(sys.argv[1]).resolve()
expected=json.loads(Path(sys.argv[2]).read_text())
inodes={(p.stat().st_dev,p.stat().st_ino) for p in root.rglob('*') if p.is_file()}
def denied(*args,**kwargs): raise RuntimeError('process network clock or archive access denied')
subprocess.Popen=socket.socket=numpy.load=time.time=time.monotonic=denied

def audit(event,args):
    if event=='open':
        path,mode,flags=args
        if flags & (os.O_WRONLY|os.O_RDWR|os.O_CREAT|os.O_TRUNC|os.O_APPEND):
            raise RuntimeError('write denied')
        if isinstance(path,int):
            st=os.fstat(path)
            if (st.st_dev,st.st_ino) not in inodes:raise RuntimeError('outside descriptor denied')
        else:
            candidate=Path(os.fsdecode(path)).absolute()
            if not candidate.is_relative_to(root):raise RuntimeError('outside file denied')
    if event=='import' or event.startswith(('socket.','subprocess.','os.exec','os.fork','os.posix_spawn')):
        raise RuntimeError('new import or execution denied')
sys.addaudithook(audit)
store=SuiteStore(root)
state=store.replay()
costs={r.resource:r.quantity for r in join_confirmation_costs(store)}
assert costs==expected['costs']
assert len(state['confirmation_scores'])==1
assert costs['confirmation_container_starts']==2
print(json.dumps({'cold_public_replay':True,'confirmed_scores':1,'inference_starts':costs['confirmation_inference_starts'],'scoring_starts':costs['confirmation_scoring_starts'],'container_starts':costs['confirmation_container_starts'],'new_processes':0,'network':False,'private_labels_read':False,'numpy_archives_loaded':False,'model_imports':False,'writes':False}))
