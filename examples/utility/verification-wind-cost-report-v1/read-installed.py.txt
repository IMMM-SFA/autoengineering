"""Read wind costs through installed core and CLI, without examples or models."""

import argparse
import builtins
import json
from pathlib import Path
import socket
import subprocess
import sys

parser = argparse.ArgumentParser()
parser.add_argument("--installed", type=Path, required=True)
parser.add_argument("--public", type=Path, required=True)
parser.add_argument("--output", type=Path, required=True)
args = parser.parse_args()
installed, public, output = args.installed.resolve(), args.public.resolve(), args.output.resolve()
output.mkdir(mode=0o700)
sys.path.insert(0, str(installed))
original_import = builtins.__import__


def guarded_import(name, *args, **kwargs):
    if name.split(".")[0] in {
        "examples", "py_wake", "h5py", "netCDF4", "matplotlib", "torch", "botorch", "tensorflow",
    }:
        raise AssertionError("installed reader attempted an example or optional model import")
    return original_import(name, *args, **kwargs)


def denied(*args, **kwargs):
    raise AssertionError("installed reader attempted a process or network")


builtins.__import__ = guarded_import
socket.socket = denied
subprocess.Popen = denied

import autoengineering
from click.testing import CliRunner
from autoengineering.cli import cli
from autoengineering.campaign.serialization import canonical_json
from autoengineering.utility.load_costs import CostEvidenceSpec, join_cost_inputs
from autoengineering.utility.report import accounting_report, render_markdown
from autoengineering.utility.store import SuiteStore

assert Path(autoengineering.__file__).resolve().is_relative_to(installed)
assert not list(public.rglob("*.f64le"))
store = SuiteStore(public / "suite")
inputs = CostEvidenceSpec.from_json((public / "cost-inputs.json").read_text())
head = store.replay()["head"]
before = accounting_report(store)
records = join_cost_inputs(store, inputs)
quantities = {record.resource: record.quantity for record in records}
assert quantities["wind_model_starts"] == quantities["wind_case_attempts"] == 3
assert quantities["wind_private_reads"] == 3 and quantities["wind_private_bytes"] == 420864
assert quantities["wind_cpu_time"] is quantities["wind_peak_rss"] is quantities["wind_inner_wake_calls"] is None
document = accounting_report(store, cost_inputs=inputs)
assert document["complete_cost_evidence"] is False
assert document["cost_inputs_digest"] == inputs.digest
assert accounting_report(store) == before
assert canonical_json(accounting_report(store)) == canonical_json(before)
(output / "default-accounting.json").write_text(json.dumps(before, indent=2, sort_keys=True) + "\n")
(output / "accounting.json").write_text(json.dumps(document, indent=2, sort_keys=True) + "\n")
(output / "accounting.md").write_text(render_markdown(document))
report = CliRunner().invoke(cli, ["utility", "report", str(store.root), "--cost-inputs", str(public / "cost-inputs.json")])
assert report.exit_code == 0, report.output
assert json.loads(report.output) == document
assert report.output.strip() == canonical_json(document)
verify = CliRunner().invoke(cli, ["utility", "verify", str(store.root), "--cost-inputs", str(public / "cost-inputs.json")])
assert verify.exit_code == 0, verify.output
assert json.loads(verify.output)["cost_inputs_digest"] == inputs.digest
(output / "cli-report.json").write_text(report.output)
(output / "cli-verify.json").write_text(verify.output)
assert store.replay()["head"] == head
for name, module in tuple(sys.modules.items()):
    if name == "autoengineering" or name.startswith("autoengineering."):
        if getattr(module, "__file__", None):
            assert Path(module.__file__).resolve().is_relative_to(installed), name
result = {
    "status": "passed", "installed_package": str(Path(autoengineering.__file__).resolve()),
    "cost_inputs_schema": inputs.schema, "wind_costs": quantities,
    "default_report_unchanged": True, "new_suite_events": 0,
    "example_imports": 0, "optional_model_imports": 0, "model_starts": 0,
    "private_weather_reads": 0, "network_requests": 0, "full_cost_evidence": False,
}
(output / "result.json").write_text(json.dumps(result, indent=2, sort_keys=True) + "\n")
print(json.dumps(result, sort_keys=True))
