"""Copy public completion metadata only; no private rows, seed or packet files."""
from pathlib import Path
import hashlib,json,shutil
from autoengineering.campaign.serialization import canonical_json
from autoengineering.utility.store import SuiteStore
from autoengineering.utility.load_costs import CostEvidenceSpec
from autoengineering.utility.report import accounting_report,render_markdown

root=Path.cwd();p=root/'examples/utility/verification-engineering-supervised-confirmation-v1'
work=root/'outputs/engineering-supervised-confirmation-v1'
complete=json.loads((p/'complete.json').read_bytes())
request=json.loads((p/'request.json').read_bytes())
grant=request['grant'];grant_digest=complete['execution']['grant_digest']
public=p/'public';public.mkdir(mode=0o700)
shutil.copytree(work/'suite',public/'suite')
runtime=work/'callback-work'/grant_digest/'runtime'
audit=public/'runtime';audit.mkdir(mode=0o700)
for name in ('intent.json','create-intent.json','created.json','before.json','start-intent.json',
             'client.json','terminal.json','output.json','stop.json','delete-intent.json',
             'cleanup.json','complete.json','watchdog-plan.json','watchdog-ready.json','watchdog-outcome.json'):
    shutil.copyfile(runtime/name,audit/name)
shutil.copyfile(work/'callback-work'/grant_digest/'complete.json',public/'original-completion-pin.json')
shutil.copyfile(p/'cost-inputs.json',public/'cost-inputs.json')
shutil.copyfile(p/'request.json',public/'request.json')
shutil.copyfile(p/'complete.json',public/'complete.json')
shutil.copyfile(p/'watchdog-process-absence.json',public/'watchdog-process-absence.json')
store=SuiteStore(public/'suite')
inputs=CostEvidenceSpec.from_json((public/'cost-inputs.json').read_text())
legacy=accounting_report(store);report=accounting_report(store,cost_inputs=inputs)
(public/'expected-legacy.json').write_text(canonical_json(legacy)+'\n')
(public/'expected-accounting.json').write_text(canonical_json(report)+'\n')
(public/'expected-accounting.md').write_text(render_markdown(report))
assert not any(f.name in ('scenarios.json','packet.json','preparation-claim.json') for f in public.rglob('*'))
assert not any(f.name.startswith('case-') for f in public.rglob('*'))
pins={str(f.relative_to(public)):hashlib.sha256(f.read_bytes()).hexdigest() for f in sorted(public.rglob('*')) if f.is_file()}
(p/'public-pins.json').write_text(json.dumps(pins,indent=2)+'\n')
print(json.dumps({'public_files':len(pins),'private_payloads_copied':0,'new_model_starts':0,
                  'model_start_quantity':complete['coupled_model_starts'],'complete_cost_evidence':report['complete_cost_evidence']}))
