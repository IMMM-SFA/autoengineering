"""Explicit local candidate owner for the fixed coupled-load Linux image.

No image pull, executable selection or private-source access is provided. One
preflight and one runtime dispatch are consumed for each grant. Ordinary cleanup
is bounded. Owner-loss supervision and runtime admission remain unqualified.
"""

def source_identity():
    return hashlib.sha256(Path(__file__).read_bytes()).hexdigest()


def configuration(nonce, packet_sha256, grant_digest):
    if re.fullmatch("[0-9a-f]{32}", nonce) is None:
        raise ValueError("coupled container nonce differs")
    for pin in (packet_sha256, grant_digest):
        if re.fullmatch("[0-9a-f]{64}", pin) is None:
            raise ValueError("coupled container pin differs")
    value = policy._configuration(
        "read_only_root", nonce, "fixed coupled entrypoint", image_id=policy.CURRENT_IMAGE
    )
    value.update(
        Image=IMAGE,
        Entrypoint=list(ENTRYPOINT),
        Cmd=[packet_sha256],
        AttachStdin=True,
        OpenStdin=True,
        StdinOnce=True,
    )
    value["Labels"] = {
        **LABELS,
        "org.autoengineering.nonce": nonce,
        "org.autoengineering.packet": packet_sha256,
        "org.autoengineering.grant": grant_digest,
    }
    return value


def verify_image(value):
    if (value.get("Id"), value.get("Os"), value.get("Architecture")) != (IMAGE, "linux", "amd64"):
        raise ValueError("coupled image identity differs")
    config = value["Config"]
    expected_env = dict((*policy.BASE_ENV, *ENVIRONMENT))
    if (
        config.get("Labels") != LABELS
        or config.get("Entrypoint") != ENTRYPOINT
        or config.get("Cmd") not in (None, [])
        or config.get("Volumes") not in (None, {})
        or config.get("OnBuild") not in (None, [])
        or config.get("User") != "1000:1000"
        or config.get("WorkingDir") != "/work"
        or _environment(config.get("Env")) != expected_env
    ):
        raise ValueError("coupled inherited image configuration differs")


def verify_owned(value, expected, *, cid, name):
    """Verify exact ownership before every mutation, including cleanup."""
    if (
        not isinstance(value, dict)
        or re.fullmatch("[0-9a-f]{64}", cid) is None
        or (value.get("Id"), value.get("Image"), value.get("Name")) != (cid, IMAGE, "/" + name)
    ):
        raise ValueError("coupled container ownership differs")
    config, host = value["Config"], value["HostConfig"]
    for key, wanted in expected.items():
        if key in ("HostConfig", "Env", "Volumes"):
            continue
        if canonical_json(config.get(key)) != canonical_json(wanted):
            raise ValueError("coupled container command or binding differs")
    if _environment(config.get("Env")) != _environment(expected["Env"]) or config.get(
        "Volumes"
    ) not in (None, {}):
        raise ValueError("coupled container environment differs")
    for key, wanted in expected["HostConfig"].items():
        observed = host.get(key)
        if key == "SecurityOpt" and observed == ["no-new-privileges:true"]:
            continue
        if key in {"Binds", "Mounts", "VolumesFrom", "Devices", "DeviceRequests", "CapAdd"}:
            if observed is None and wanted == []:
                continue
        if key == "PortBindings" and observed is None and wanted == {}:
            continue
        if canonical_json(observed) != canonical_json(wanted):
            raise ValueError("coupled isolation or resource configuration differs")
    for key in (
        "DeviceCgroupRules",
        "GroupAdd",
        "Links",
        "ExtraHosts",
        "Dns",
        "DnsOptions",
        "DnsSearch",
        "Ulimits",
    ):
        if host.get(key) not in (None, []):
            raise ValueError("extra coupled host settings")
    for key in ("Sysctls", "StorageOpt"):
        if host.get(key) not in (None, {}):
            raise ValueError("extra coupled kernel settings")
    for key in ("UsernsMode", "UTSMode", "CgroupParent", "ContainerIDFile", "Isolation"):
        if host.get(key) not in (None, ""):
            raise ValueError("coupled namespace differs")
    if host.get("Runtime", "runc") != "runc":
        raise ValueError("coupled runtime differs")
    mounts = value.get("Mounts")
    if (
        not isinstance(mounts, list)
        or len(mounts) > 1
        or any(
            (m.get("Type"), m.get("Destination"), m.get("RW")) != ("tmpfs", "/work", True)
            or m.get("Source") not in (None, "")
            for m in mounts
        )
    ):
        raise ValueError("coupled container mount differs")
    if type(value.get("RestartCount")) is not int or value["RestartCount"] != 0:
        raise ValueError("coupled restart history differs")
    state = value["State"]
    if state.get("Paused") is not False or state.get("Restarting") is not False:
        raise ValueError("coupled state differs")
    return state


def exchange(argv, packet, *, env, cwd, deadline, check_alive=None):
    """One owned CLI group; bounded simultaneous input/output, no raw log file."""
    if time.monotonic() >= deadline:
        raise TimeoutError("coupled client deadline expired")
    if check_alive is not None:
        check_alive()
    process = subprocess.Popen(
        argv,
        stdin=subprocess.PIPE,
        stdout=subprocess.PIPE,
        stderr=subprocess.STDOUT,
        env=env,
        cwd=cwd,
        start_new_session=True,
    )
    offset = 0
    output = bytearray()
    try:
        os.set_blocking(process.stdin.fileno(), False)
        os.set_blocking(process.stdout.fileno(), False)
        with selectors.DefaultSelector() as selector:
            selector.register(process.stdin, selectors.EVENT_WRITE)
            selector.register(process.stdout, selectors.EVENT_READ)
            while selector.get_map():
                if check_alive is not None:
                    check_alive()
                remaining = deadline - time.monotonic()
                if remaining <= 0:
                    raise TimeoutError("coupled client deadline expired")
                for key, _ in selector.select(min(0.1, remaining)):
                    if check_alive is not None:
                        check_alive()
                    if key.fileobj is process.stdin:
                        offset += os.write(process.stdin.fileno(), packet[offset : offset + 4096])
                        if offset == len(packet):
                            selector.unregister(process.stdin)
                            process.stdin.close()
                    else:
                        data = os.read(process.stdout.fileno(), MAX_OUTPUT + 1 - len(output))
                        if not data:
                            selector.unregister(process.stdout)
                        output.extend(data)
                        if len(output) > MAX_OUTPUT:
                            raise ValueError("coupled output exceeds envelope")
            while True:
                if check_alive is not None:
                    check_alive()
                remaining = deadline - time.monotonic()
                if remaining <= 0:
                    raise TimeoutError("coupled client exit deadline expired")
                try:
                    code = process.wait(timeout=min(0.1, remaining))
                    break
                except subprocess.TimeoutExpired:
                    continue
            if check_alive is not None:
                check_alive()
            if offset != len(packet) or time.monotonic() >= deadline:
                raise TimeoutError("coupled client completion unverified")
            return code, bytes(output)
    finally:
        if process.poll() is None:
            os.killpg(process.pid, signal.SIGKILL)
        process.wait(timeout=5)
        process.stdin.close()
        process.stdout.close()


class CoupledContainerRuntime:
    def __init__(self, *, cli, endpoint, api_version, server_digest):
        self.cli, self.endpoint = str(Path(cli).absolute()), str(Path(endpoint).absolute())
        self.api = GuestDaemonAPI(
            self.endpoint, api_version=api_version, server_digest=server_digest
        )
        self.transport = self._transport()
        self.registration = BoundCoupledRuntimeRegistration(
            schema="utility-coupled-runtime-registration-v2",
            image_id=IMAGE,
            environment_digest=LABELS["org.autoengineering.environment"],
            runtime_source_digest=source_identity(),
            entrypoint_source_digest=ENTRYPOINT_SHA,
            worker_source_digest=WORKER_SHA,
            kernel_source_digest=KERNEL_SHA,
            source_manifest_digest=SOURCE_MANIFEST,
            transport_digest=digest(self.transport),
        )
        self._preflights = {}

    def source_identity(self):
        return source_identity()

    def _transport(self):
        return {
            "daemon_profile_digest": self.api.identity,
            "cli": build._binary(self.cli),
            "endpoint": build._endpoint(self.endpoint),
            "helper_sources": {
                m.__name__: hashlib.sha256(Path(m.__file__).read_bytes()).hexdigest()
                for m in (build, policy, supervision)
            },
            "max_seconds": MAX_SECONDS,
            "cleanup_seconds": CLEANUP_SECONDS,
            "output_limit_bytes": MAX_OUTPUT,
        }

    def _verify_identity(self):
        self.api._identity()
        if (
            self.source_identity() != self.registration.runtime_source_digest
            or digest(self._transport()) != self.registration.transport_digest
        ):
            raise ValueError("coupled owner or transport changed")

    def preflight(self, grant):
        self._verify_identity()
        validate_container_resources(grant, self.registration)
        if grant.digest in self._preflights:
            raise ValueError("coupled preflight already consumed")
        limit = next(r.limit for r in grant.resources if r.name == "active_time")
        if limit < MAX_SECONDS:
            raise ValueError("coupled owner needs its full lifecycle allowance")
        begin = time.monotonic()
        deadline = begin + MAX_SECONDS
        self._preflights[grant.digest] = [begin, deadline, "preflight_intended"]
        version = self.api._json(
            "GET", "/version", deadline=min(deadline - CLEANUP_SECONDS, begin + 10)
        )
        if digest(version) != self.api.profile.server_digest:
            raise ValueError("coupled daemon changed")
        image = self.api._json(
            "GET",
            "/images/" + IMAGE + "/json",
            deadline=min(deadline - CLEANUP_SECONDS, begin + 10),
        )
        verify_image(image)
        self._preflights[grant.digest][2] = "ready"

    def run(self, grant, packet, *, expected_sha256, work_root):
        self._verify_identity()
        prepared = self._preflights.get(grant.digest)
        if prepared is None or prepared[2] != "ready":
            raise ValueError("coupled dispatch already consumed or not preflighted")
        prepared[2] = "consumed"
        begin, deadline, _ = prepared
        active = deadline - CLEANUP_SECONDS
        parsed = parse_packet(packet, expected_sha256=expected_sha256)
        if parsed["execution_digest"] != grant.digest or parsed["kernel_digest"] != KERNEL_SHA:
            raise ValueError("coupled packet belongs to another grant or kernel")
        root = checked_directory(work_root) / "runtime"
        root.mkdir(mode=0o700)

        def save(name, value):
            durable_write(root / name, canonical_json(value).encode())

        nonce = uuid.uuid4().hex
        name = "ae-coupled-load-" + nonce
        config = configuration(nonce, expected_sha256, grant.digest)
        save(
            "intent.json",
            {
                "grant_digest": grant.digest,
                "runtime_digest": self.registration.digest,
                "transport_digest": self.registration.transport_digest,
                "packet_sha256": expected_sha256,
                "packet_bytes": len(packet),
                "container_name": name,
                "nonce": nonce,
                "configuration": config,
                "max_creations": 1,
                "max_starts": 1,
                "automatic_retries": 0,
            },
        )
        cid = None
        create_intended = False
        result = None
        start_receipt = None
        stop_receipt = None
        watchdog = None

        def inspect_name():
            status, raw = self.api._request(
                "GET", "/containers/" + name + "/json", deadline=deadline
            )
            if status == 404:
                return None
            if status != 200:
                raise ValueError("coupled named inspection unverified")
            return unique_json(raw.decode())

        try:
            if inspect_name() is not None:
                raise ValueError("coupled original name already exists")
            save(
                "create-intent.json",
                {"configuration_digest": digest(config), "container_name": name},
            )
            create_intended = True
            created = self.api._json(
                "POST", "/containers/create?name=" + name, config, expected=201, deadline=active
            )
            acknowledged = created.get("Id")
            if (
                not isinstance(acknowledged, str)
                or re.fullmatch("[0-9a-f]{64}", acknowledged) is None
            ):
                raise ValueError("coupled created identity differs")
            cid = acknowledged
            save("created.json", created)
            before = self.api.inspect(cid, deadline=active)
            state = verify_owned(before, config, cid=cid, name=name)
            if state.get("Status") != "created" or state.get("Running") is not False:
                raise ValueError("coupled container was already started")
            save("before.json", before)
            self._verify_identity()
            watchdog = supervision.CoupledWatchdog(
                root,
                self.api,
                config=config,
                cid=cid,
                name=name,
                active_deadline=active,
                deadline=deadline,
            )
            watchdog.ready()
            for dirname in ("cli-config", "cli-home"):
                (root / dirname).mkdir(mode=0o700)
            durable_write(root / "cli-config/config.json", b"{}")
            argv = [
                self.cli,
                "--config",
                str(root / "cli-config"),
                "--host",
                "unix://" + self.endpoint,
                "start",
                "--attach",
                "--interactive",
                cid,
            ]
            start_receipt = {
                "grant_digest": grant.digest,
                "container_id": cid,
                "packet_sha256": expected_sha256,
                "argv_digest": digest(argv),
                "starts_reserved": 1,
            }
            save("start-intent.json", start_receipt)
            env = {
                "PATH": "/usr/local/bin:/usr/bin:/bin",
                "LANG": "C.UTF-8",
                "HOME": str(root / "cli-home"),
                "DOCKER_CONFIG": str(root / "cli-config"),
                "DOCKER_HOST": "unix://" + self.endpoint,
            }
            code, raw = exchange(
                argv,
                packet,
                env=env,
                cwd=root,
                deadline=active,
                check_alive=watchdog.check_alive,
            )
            watchdog.check_alive()
            save(
                "client.json",
                {
                    "exit_code": code,
                    "output_bytes": len(raw),
                    "output_sha256": hashlib.sha256(raw).hexdigest(),
                },
            )
            terminal = self.api.inspect(cid, deadline=active)
            state = verify_owned(terminal, config, cid=cid, name=name)
            save("terminal.json", terminal)
            if code != 0 or (
                state.get("Status"),
                state.get("ExitCode"),
                state.get("Running"),
                state.get("OOMKilled"),
            ) != ("exited", 0, False, False):
                raise ValueError("coupled worker terminal status differs")
            work = decode_worker_output(raw)
            # Only the validated fixed public envelope is retained, never raw
            # input or unvalidated CLI/model diagnostics.
            durable_write(root / "output.json", raw)
            stop_receipt = {
                "container_id": cid,
                "terminal_digest": digest(terminal),
                "output_sha256": hashlib.sha256(raw).hexdigest(),
                "score_digest": work.score.digest,
            }
            save("stop.json", stop_receipt)
            result = (raw, work.score)
        except BaseException as error:
            save(
                "failure.json",
                {
                    "grant_digest": grant.digest,
                    "error_type": type(error).__name__,
                    "attempt_consumed": True,
                    "automatic_retry_forbidden": True,
                    "model_work_total": None,
                },
            )
            raise
        finally:
            try:
                with supervision.cleanup_lock(root, deadline):
                    cleanup = self._cleanup_owned(
                        self.api,
                        cid,
                        create_intended,
                        inspect_name,
                        config,
                        name,
                        deadline,
                        save,
                    )
                    cid = cleanup["container_id"]
            finally:
                if watchdog is not None:
                    watchdog.finish()
        if result is None:
            raise ValueError("coupled result unavailable")
        raw, score = result
        execution = ConfirmationContainerExecution(
            "utility-confirmation-container-execution-v1",
            grant.digest,
            grant.scorer_digest,
            score.digest,
            self.registration.digest,
            cid,
            nonce,
            IMAGE,
            self.registration.runtime_source_digest,
            ENTRYPOINT_SHA,
            WORKER_SHA,
            digest(start_receipt),
            digest(stop_receipt),
            digest(cleanup),
            1,
            1,
            time.monotonic() - begin,
        )
        save(
            "complete.json",
            {
                "execution": execution.to_dict(),
                "packet_sha256": expected_sha256,
                "private_boundary_qualified": False,
                "runtime_qualified": False,
            },
        )
        return ContainerLoadRun(raw, execution, expected_sha256)

    @staticmethod
    def _cleanup_owned(api, cid, create_intended, inspect_name, config, name, deadline, save):
        current = (
            api.inspect(cid, deadline=deadline)
            if cid is not None
            else (inspect_name() if create_intended else None)
        )
        if current is not None:
            actual = current["Id"]
            verify_owned(current, config, cid=actual, name=name)
            if cid is not None and actual != cid:
                raise ValueError("coupled cleanup identity changed")
            cid = actual
            save("delete-intent.json", {"container_id": cid, "force": True})
            status, body = api._request(
                "DELETE", "/containers/" + cid + "?force=1", deadline=deadline
            )
            if status != 204 or body != b"":
                raise ValueError("coupled cleanup unverified")
        absent = (
            api.inspect(cid, deadline=deadline) is None
            if cid is not None
            else inspect_name() is None
        )
        cleanup = {"container_id": cid, "absent": absent, "owner_loss_qualified": False}
        save("cleanup.json", cleanup)
        if not absent:
            raise ValueError("coupled original container remains")
        return cleanup
