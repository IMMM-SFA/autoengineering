from pathlib import Path
import urllib.request,json,hashlib,time
root=Path(__file__).resolve().parent
items=[]
for name,url in [('daymet-v2-guide.html','https://daacweb-prod.ornl.gov/DAYMET/guides/Daymet_mosaics.html'),('ncar-repository-root.json','https://api.github.com/repos/NCAR/CAMELS/contents')]:
    item={'path':name,'url':url,'status':'intent','max_body_bytes':512*1024,'automatic_retries':False}
    items.append(item)
    receipt=root/'guide-requests.json'
    receipt.write_text(json.dumps(items,indent=2)+'\n')
    start=time.monotonic()
    try:
        with urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'Autoengineering-source-metadata-review/0.2'}),timeout=40) as response:
            data=response.read(item['max_body_bytes']+1)
            if len(data)>item['max_body_bytes']:raise ValueError('metadata bound')
            with (root/name).open('xb') as output:output.write(data)
            item.update(status='complete',http_status=response.status,final_url=response.url,body_bytes=len(data),sha256=hashlib.sha256(data).hexdigest())
    except Exception as error:item.update(status='failed',error_type=type(error).__name__)
    item.update(elapsed_seconds=time.monotonic()-start,total_http_exchanges=None,total_network_bytes=None)
    receipt.write_text(json.dumps(items,indent=2)+'\n')
    print(json.dumps(item),flush=True)
