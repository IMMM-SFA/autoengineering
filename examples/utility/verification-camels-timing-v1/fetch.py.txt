from pathlib import Path
import urllib.request,hashlib,json,time
root=Path(__file__).resolve().parent
url='https://hess.copernicus.org/articles/19/209/2015/hess-19-209-2015.pdf'
item={'url':url,'max_body_bytes':8*1024**2,'status':'intent','prior_web_fetch':'timed out; preserved in conversation, no PDF acquired by that tool','automatic_retries':False}
path=root/'request.json'
with path.open('x') as f:json.dump(item,f,indent=2)
start=time.monotonic()
try:
    with urllib.request.urlopen(url,timeout=40) as response:
        data=response.read(item['max_body_bytes']+1)
        if len(data)>item['max_body_bytes'] or not data.startswith(b'%PDF-'):raise ValueError('paper response bound or format')
        with (root/'newman-2015.pdf').open('xb') as f:f.write(data)
        item.update(status='complete',http_status=response.status,final_url=response.url,body_bytes=len(data),sha256=hashlib.sha256(data).hexdigest())
except Exception as error:
    item.update(status='failed',error_type=type(error).__name__)
finally:
    item.update(elapsed_seconds=time.monotonic()-start,total_network_bytes=None,total_http_exchanges=None)
    path.write_text(json.dumps(item,indent=2)+'\n')
    print(json.dumps(item),flush=True)
