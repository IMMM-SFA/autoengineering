"""Recover saved fit metadata and prepare v6 inference; never run training again."""
import hashlib,json
from pathlib import Path
from autoengineering.campaign.serialization import digest
from autoengineering.utility.confirmation import read_selected_state
from autoengineering.utility.store import SuiteStore
from autoengineering.utility.guest_owner import GuestOwnerManifest
from examples.utility import guest_inference_smoke as smoke
root=Path.cwd()
proof=root/'examples/utility/verification-guest-inference-v6'
request=json.loads((proof/'fixture-request.json').read_text())
assert all(hashlib.sha256((root/p).read_bytes()).hexdigest()==h for p,h in request['source_pins'].items())
out=root/'outputs/guest-inference-current-fixture-v6'
owner=out/'template/owner'
suite=out/'authored-guest-metadata/composite'
source=out/'authored-guest-metadata/source'
drivers=out/'template/drivers'
original=SuiteStore(suite)
job=GuestOwnerManifest.from_json((owner/'owner.json').read_text()).job
bound=read_selected_state(original.root,job.grant.inference.selected_state)
assert not original.replay()['confirmation_accesses']
summary={'scope':request['scope'],'request_digest':digest(request),'suite_digest':original.spec.digest,'source_digest':original.spec.source_digest,'fit_sha256':bound.training.fit.sha256,'execution_seed':int(bound.evaluation.provenance['execution_seed']),'training_result':json.loads((original.root/bound.training.result.path).read_text()),'owner':str(owner),'suite':str(suite),'source':str(source),'drivers':str(drivers),'source_pins_unchanged':True,'confirmation_accesses':0,'guest_starts':0,'metadata_only_recovery':True,'new_fitting_starts_during_recovery':0,'runtime_qualified':False,'independent_confirmation':False}
smoke.save(proof,'fixture-completion.json',summary)
prepared=smoke.prepare(owner,suite,source,drivers,root/'examples/utility/verification-guest-resources-v6/evidence',root/'examples/utility/prepared-inference-environment-v6',resource_digest='98079b65f6ec0d7122145a51fe0e8d4f0b1b88aaebdf9afb8674040b3b2c4648',output=root/'outputs/guest-inference-v6')
print(json.dumps({'fixture_request_digest':digest(request),'suite_digest':original.spec.digest,'fit_sha256':bound.training.fit.sha256,'smoke_request_digest':digest(prepared)}))
