"""One current-source authored fit and baseline selection, then stage a fixed guest control.

Test fixture builders supply synthetic exposed inputs and an authored owner template.
That template does not attest an OS. The smoke preparer requires the separate actual
v6 resource/identity evidence and replaces the template's guest/environment metadata.
"""
import hashlib
import json
import os
from pathlib import Path
import sys

from autoengineering.campaign.serialization import digest
from autoengineering.utility.confirmation import read_selected_state
from autoengineering.utility.report import accounting_report
from autoengineering.utility.guest_owner import GuestOwnerManifest
from examples.utility import guest_inference_smoke as smoke

root=Path.cwd()
os.umask(0o077)
sys.path.insert(0,str(root/'tests'))
import test_utility_guest_inputs as fixture

out=root/'outputs/guest-inference-current-fixture-v6'
assert not out.exists()
proof=root/'examples/utility/verification-guest-inference-v6'
pins={str(p.relative_to(root)):hashlib.sha256(p.read_bytes()).hexdigest() for base in ('src/autoengineering','tests') for p in (root/base).rglob('*.py')}
request={'scope':'one fresh current-source authored trained baseline fixture, no independent observations','training_starts_reserved':1,'training_routing_rollouts_cap':120,'training_prefix_days':80,'baseline_full_evaluations_reserved':1,'candidate_count':1,'confirmation_starts_allowed':0,'automatic_retries':0,'private_observations':0,'source_pins':pins}
(proof/'fixture-request.json').write_text(json.dumps(request,indent=2)+'\n')
out.mkdir(mode=0o700)
class Factory:
    def mktemp(self,name):
        p=out/name
        p.mkdir(mode=0o700)
        return p
metadata=fixture.metadata.__wrapped__(Factory())
original=metadata[0]
template=out/'template'
template.mkdir(mode=0o700)
owner,_,source,drivers=fixture.prepared(template,metadata,authorize=False)
for p in out.rglob('*'):
    if p.is_dir():p.chmod(0o700)
job=GuestOwnerManifest.from_json((owner.root/'owner.json').read_text()).job
bound=read_selected_state(original.root,job.grant.inference.selected_state)
report=accounting_report(original)
smoke.save(proof,'fixture-accounting.json',report)
assert not original.replay()['confirmation_accesses']
assert len(original.spec.cells)==1
changed=[p for p,h in pins.items() if hashlib.sha256((root/p).read_bytes()).hexdigest()!=h]
assert not changed,changed
summary={'scope':request['scope'],'request_digest':digest(request),'suite_digest':original.spec.digest,'source_digest':original.spec.source_digest,'fit_sha256':bound.training.fit.sha256,'execution_seed':int(bound.evaluation.provenance['execution_seed']),'training':{k:json.loads((original.root/r.path).read_text()) for k,r in [('costs',bound.training.costs)]},'owner':str(owner.root),'suite':str(original.root),'source':str(source),'drivers':str(drivers),'source_pins_unchanged':True,'confirmation_accesses':0,'guest_starts':0,'runtime_qualified':False,'independent_confirmation':False}
smoke.save(proof,'fixture-completion.json',summary)
prepared=smoke.prepare(owner.root,original.root,source,drivers,root/'examples/utility/verification-guest-resources-v6/evidence',root/'examples/utility/prepared-inference-environment-v6',resource_digest='98079b65f6ec0d7122145a51fe0e8d4f0b1b88aaebdf9afb8674040b3b2c4648',output=root/'outputs/guest-inference-v6')
print(json.dumps({'fixture_request_digest':digest(request),'suite_digest':original.spec.digest,'fit_sha256':bound.training.fit.sha256,'smoke_request_digest':digest(prepared)}))
