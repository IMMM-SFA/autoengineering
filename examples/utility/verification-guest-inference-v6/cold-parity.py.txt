"""Relocated parity recomputation after imports and numerical initialization."""
import hashlib,json,os,socket,subprocess,sys,time
from pathlib import Path
root=Path(sys.argv[1]).resolve()
sys.path.insert(0,sys.argv[2])
from examples.utility import guest_parity_reader as reader
expected=sys.argv[3]
# Initialize inherited dependencies before the guarded reconstruction.
reader.read_completion(root,expected_digest=expected)
inodes={(p.stat().st_dev,p.stat().st_ino) for p in root.rglob('*') if p.is_file()}
def forbidden(*args,**kwargs):raise AssertionError('parity reader attempted a live operation')
def guard(event,args):
    if event=='open':
        path,mode,flags=args
        if isinstance(path,int):
            st=os.fstat(path)
            if (st.st_dev,st.st_ino) not in inodes:raise AssertionError('unscoped file descriptor')
        elif not Path(path).resolve().is_relative_to(root):raise AssertionError('outside proof read')
        if any(c in (mode or '') for c in 'wax+') or flags & (os.O_WRONLY|os.O_RDWR|os.O_CREAT|os.O_TRUNC|os.O_APPEND):raise AssertionError('write')
    elif event=='import' or event.startswith(('socket.','subprocess.','os.exec')) or event in ('os.system','os.fork','os.posix_spawn'):raise AssertionError('live operation or import')
socket.socket=subprocess.Popen=time.time=time.monotonic=time.perf_counter=forbidden
sys.addaudithook(guard)
result=reader.read_completion(root,expected_digest=expected)
print(json.dumps(result,sort_keys=True))
