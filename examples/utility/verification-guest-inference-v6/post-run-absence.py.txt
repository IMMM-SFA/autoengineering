"""One read-only GET for the completed original CID. No mutation or retries."""
import json,time
from pathlib import Path
from autoengineering.utility.guest_daemon import GuestDaemonAPI
from examples.utility.guest_inference_smoke import save
root=Path('outputs/guest-inference-v6')
q=json.loads((root/'request.json').read_text())
b=json.loads((root/'bindings.json').read_text())
c=json.loads((root/'complete.json').read_text())
api=GuestDaemonAPI(b['endpoint'],api_version=q['daemon']['api_version'],server_digest=q['daemon']['server_digest'])
assert api.profile.to_dict()==q['daemon']
cid=c['execution']['container_id']
status,_=api._request('GET','/containers/'+cid+'/json',deadline=time.monotonic()+10)
assert status==404
save(Path('examples/utility/verification-guest-inference-v6'),'post-run-absence.json',{'container_id':cid,'http_status':status,'absent':True,'new_start':False})
print('Original v6 inference container absent.')
