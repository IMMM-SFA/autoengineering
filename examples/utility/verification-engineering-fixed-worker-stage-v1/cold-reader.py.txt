"""Replay fixed-worker producer evidence using only the supplied public tree."""
from pathlib import Path
import hashlib
import json
import math
import sys

def unique(pairs):
    out = {}
    for key, value in pairs:
        assert key not in out, 'duplicate proof field'
        out[key] = value
    return out

def decode(raw):
    def refused(_): raise ValueError('nonfinite proof JSON')
    return json.loads(raw, object_pairs_hook=unique, parse_constant=refused)

def canonical(value):
    return json.dumps(value, sort_keys=True, separators=(',', ':'), allow_nan=False).encode()

def digest(value): return hashlib.sha256(canonical(value)).hexdigest()

root = Path(sys.argv[1]).resolve(strict=True)
manifest_raw = (root / 'evidence-manifest.json').read_bytes()
assert hashlib.sha256(manifest_raw).hexdigest() == sys.argv[2], 'manifest differs'
manifest = decode(manifest_raw)
assert manifest['schema'] == 'utility-fixed-coupled-worker-proof-v1'
for name, expected in manifest['files'].items():
    relative = Path(name)
    assert not relative.is_absolute() and '..' not in relative.parts
    path = root / relative
    assert not any(p.is_symlink() for p in (path, *path.parents))
    assert path.is_file() and path.stat().st_size <= 1048576
    raw = path.read_bytes()
    assert len(raw) == expected['bytes'] and hashlib.sha256(raw).hexdigest() == expected['sha256']

def raw(name):
    assert name in manifest['files'], 'unbound evidence'
    return (root / name).read_bytes()

def read(name): return decode(raw(name))

build = read('image/request.json'); built = read('image/complete.json')
assert built['request_digest'] == digest(build)
assert built['image_id'] == 'sha256:62f6c7b58e34f4ec5d5517af49c8b9e1e7d1d7235f550384ac52c1281fb262ad'
assert built['source_image'] == 'sha256:39fb6f877a99e384ec2af08363a598ea23fd6ef7ec3bae076e0ab1ffe99508b0'
assert build['environment_digest'] == digest(build['environment']) == built['environment_digest']
sources = read('image/source-manifest.json')
assert sources == build['environment']['source_manifest'] == read('entrypoint/manifest.json')
for name, pin in sources['files'].items():
    assert hashlib.sha256(raw('source/' + name)).hexdigest() == pin
assert hashlib.sha256(raw('entrypoint/entrypoint.py')).hexdigest() == sources['entrypoint_sha256']
for i in range(9):
    receipt = read(f'image/{i:02d}-result.json')
    assert receipt['exit_code'] == 0
    assert hashlib.sha256(raw(f'image/{i:02d}.log')).hexdigest() == receipt['log_sha256']
    assert read(f'image/{i:02d}-intent.json')['request_digest'] == digest(build)
assert built['pull_commands'] == built['install_commands'] == built['scientific_starts'] == 0
image = decode(raw('image/08.log'))
assert image['Id'] == built['image_id']
assert image['Config']['Entrypoint'] == build['entrypoint']
assert image['Config']['Labels'] == build['labels']

q = read('probe/request.json'); complete = read('probe/complete.json')
assert complete['request_digest'] == digest(q)
assert q['image_id'] == built['image_id'] == complete['image_id']
assert q['image_environment_digest'] == built['environment_digest']
assert len(raw('probe/packet.json')) == q['packet_bytes'] <= 32768
assert hashlib.sha256(raw('probe/packet.json')).hexdigest() == q['packet_sha256']
packet = read('probe/packet.json')
assert len(packet['cases']) == q['expected_model_starts'] == 24
assert q['configuration']['Cmd'] == [q['packet_sha256']]
assert q['configuration']['Entrypoint'] == build['entrypoint']
assert q['configuration']['Labels'] == {**build['labels'], 'org.autoengineering.nonce': q['nonce'],
                                       'org.autoengineering.packet': q['packet_sha256']}
assert q['entrypoint_source_sha256'] == sources['entrypoint_sha256']
assert q['worker_source_sha256'] == sources['files']['examples/utility/engineering_load_worker.py']
assert packet['kernel_digest'] == sources['files']['examples/utility/engineering_load_scoring.py']
cid = read('probe/created.json')['Id']
for name in ('before', 'terminal'):
    observed = read('probe/' + name + '.json')
    assert observed['Id'] == cid and observed['Image'] == built['image_id']
    assert observed['Name'] == '/' + q['container_name']
    for key, wanted in q['configuration'].items():
        if key in ('HostConfig', 'Env', 'Volumes'): continue
        assert canonical(observed['Config'].get(key)) == canonical(wanted)
    assert sorted(observed['Config']['Env']) == sorted(q['configuration']['Env'])
    assert observed['Config'].get('Volumes') in (None, {})
    host = observed['HostConfig']
    for key, wanted in q['configuration']['HostConfig'].items():
        actual = host.get(key)
        if key == 'SecurityOpt' and actual == ['no-new-privileges:true']: actual = ['no-new-privileges']
        if wanted == [] and actual is None: actual = []
        if key == 'PortBindings' and actual is None: actual = {}
        assert canonical(actual) == canonical(wanted), key
terminal = read('probe/terminal.json')['State']
assert terminal['Status'] == 'exited' and terminal['ExitCode'] == 0 and terminal['OOMKilled'] is False
assert read('probe/cleanup.json')['container_id'] == cid
assert read('probe/cleanup.json')['absent'] is True and complete['original_container_absent'] is True
output_raw = raw('probe/output.log'); client = read('probe/client.json')
assert client['exit_code'] == 0 and client['output_bytes'] == len(output_raw) <= q['max_output_bytes']
assert client['output_sha256'] == hashlib.sha256(output_raw).hexdigest()
output = decode(output_raw)
assert output == complete['runtime_observed']
wrapped = output['worker']; result = wrapped['result']; score = result['score']
assert wrapped['child_exit_code'] == 0 and wrapped['child_signal'] is None
assert wrapped['captured_bytes'] < wrapped['output_limit_bytes'] == 16384
assert wrapped['output_limited'] is False and result['completed_model_starts'] == 24
assert result['attempted_model_starts'] == 24 and result['last_attempt_inner_work_unknown'] is False
assert score['status'] == 'success' and score['feasible'] is True
assert math.isclose(score['value'], q['expected_score']['value'], abs_tol=q['atol'], rel_tol=q['rtol'])
assert abs(score['value'] - q['expected_score']['value']) == complete['absolute_power_difference_W']
for value in (output, wrapped, result, complete): assert value['private_boundary_qualified'] is False
assert output['owner_loss_qualified'] is False
failure = read('probe/preflight-failure.json')
assert failure['daemon_calls'] == failure['model_starts'] == failure['container_starts'] == 0
print(json.dumps({'verified': True, 'container_starts': 1, 'coupled_model_starts': 24,
                  'score_power_W': score['value'], 'original_container_absent': True,
                  'private_boundary_qualified': False}, sort_keys=True))
