"""Execute the one prepared public worker check; never repeat an uncertain start."""
import ast, hashlib, json, math, os, re, selectors, subprocess, time
from pathlib import Path
from autoengineering.campaign.serialization import canonical_json, digest, unique_json
from autoengineering.utility.guest_daemon import GuestDaemonAPI
from examples.utility import guest_image_build as build, guest_resources as policy

root = Path.cwd(); p = root / 'examples/utility/verification-engineering-linux-worker-v1'
work = root / 'outputs/engineering-linux-worker-v1'
q = unique_json((work / 'request.json').read_text())
assert digest(q) == '5a7ab4c5f43d993c0e9deb0ba1914d22211679960de341de2135bca9b00ede4d'
assert not (work / 'execution-intent.json').exists()
assert hashlib.sha256((work / 'packet.json').read_bytes()).hexdigest() == q['packet_sha256']
for name in ('engineering_load_entrypoint', 'engineering_load_worker'):
    key = 'entrypoint_source_sha256' if name.endswith('entrypoint') else 'worker_source_sha256'
    assert hashlib.sha256((root / 'examples/utility' / (name + '.py')).read_bytes()).hexdigest() == q[key]
image_request = unique_json((root / 'examples/utility/verification-engineering-worker-image-v1/request.json').read_text())
assert build._binary('/usr/local/bin/docker') == image_request['binary']
IMAGE = q['image_id']
CONFIG_EXPECTED = q['configuration']; _same = policy._same; _environment = policy._environment
exec((p / 'verify-owned.py.txt').read_text())
# Use the bounded owned-client loop with one fixed, already pinned input fd.
source = Path(build.__file__).read_text(); tree = ast.parse(source)
runner = ast.get_source_segment(source, next(n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == '_run'))
runner = runner.replace('stdin=subprocess.DEVNULL', 'stdin=packet_stream')
MAX_LOG = q['max_output_bytes']; _kill_client = build._kill_client
exec(runner)
(p / 'stdin-client-loop.py.txt').write_text(runner)
api = GuestDaemonAPI('/Users/cam/.orbstack/run/docker.sock', api_version=q['api_version'], server_digest=q['server_digest'])
def save(name, value):
    for target in (p, work): (target / name).write_text(json.dumps(value, indent=2) + '\n')
active = time.monotonic() + q['active_seconds']
save('execution-intent.json', {'request_digest': digest(q), 'creations_reserved': 1,
     'starts_reserved': 1, 'model_starts_reserved': 24, 'automatic_retries': 0})
version = api._json('GET', '/version', deadline=active)
assert digest(version) == q['server_digest']
image = api._json('GET', '/images/' + q['image_id'] + '/json', deadline=active)
save('image.json', image)
assert (image['Id'], image['Os'], image['Architecture']) == (q['image_id'], 'linux', 'amd64')
assert image['Config']['Labels'] == image_request['labels']
assert image['Config']['Entrypoint'] == q['configuration']['Entrypoint']
cid = None; outcome = None
try:
    save('create-intent.json', {'configuration_digest': digest(q['configuration']), 'name': q['container_name']})
    created = api._json('POST', '/containers/create?name=' + q['container_name'], q['configuration'], expected=201, deadline=active)
    cid = created['Id']; assert re.fullmatch('[0-9a-f]{64}', cid); save('created.json', created)
    before = api.inspect(cid, deadline=active)
    verify_owned(before, 'public', q['nonce'], cid=cid, image_id=q['image_id'])
    assert before['State']['Status'] == 'created' and not before['State']['Running']
    save('before.json', before)
    (work / 'config').mkdir(mode=0o700); (work / 'config/config.json').write_text('{}')
    argv = ['/usr/local/bin/docker', '--config', str(work / 'config'), '--host',
            'unix:///Users/cam/.orbstack/run/docker.sock', 'start', '--attach', '--interactive', cid]
    save('start-intent.json', {'container_id': cid, 'starts': 1,
         'argv_digest': digest(argv), 'stdin_sha256': q['packet_sha256']})
    env = {k: os.environ[k] for k in ('PATH', 'HOME', 'LANG', 'LC_ALL', 'TMPDIR') if k in os.environ}
    env.update(DOCKER_CONFIG=str(work / 'config'), DOCKER_HOST='unix:///Users/cam/.orbstack/run/docker.sock')
    with (work / 'packet.json').open('rb') as packet_stream:
        status = _run(argv, env=env, cwd=work, deadline=active, log=work / 'output.log')
    raw = (work / 'output.log').read_bytes(); (p / 'output.log').write_bytes(raw)
    save('client.json', {'exit_code': status, 'output_sha256': hashlib.sha256(raw).hexdigest(),
         'output_bytes': len(raw)})
    terminal = api.inspect(cid, deadline=active)
    verify_owned(terminal, 'public', q['nonce'], cid=cid, image_id=q['image_id'])
    save('terminal.json', terminal)
    assert status == 0 and terminal['State']['Status'] == 'exited'
    assert terminal['State']['ExitCode'] == 0 and terminal['State']['OOMKilled'] is False
    assert 0 < len(raw) <= q['max_output_bytes']
    observed = unique_json(raw.decode())
    assert observed['schema'] == 'utility-coupled-container-result-v1'
    assert observed['observed'] == policy.OBSERVED
    assert observed['private_boundary_qualified'] is False and observed['owner_loss_qualified'] is False
    wrapped = observed['worker']; result = wrapped['result']; score = result['score']
    assert wrapped['child_exit_code'] == 0 and wrapped['child_signal'] is None
    assert not wrapped['output_limited'] and wrapped['captured_bytes'] < 16384
    assert result['attempted_model_starts'] == result['completed_model_starts'] == 24
    assert result['last_attempt_inner_work_unknown'] is False
    for name in ('status', 'outcome', 'units', 'feasible'): assert score[name] == q['expected_score'][name]
    assert score['status'] == 'success' and score['feasible'] is True and math.isfinite(score['value'])
    assert math.isclose(score['value'], q['expected_score']['value'], abs_tol=q['atol'], rel_tol=q['rtol'])
    assert b'authored-' not in raw and b'resistor1_ohm' not in raw
    outcome = {'request_digest': digest(q), 'image_id': q['image_id'], 'container_id': cid,
               'container_creations': 1, 'container_starts': 1, 'coupled_model_starts': 24,
               'runtime_observed': observed, 'score_within_frozen_tolerance': True,
               'absolute_power_difference_W': abs(score['value'] - q['expected_score']['value']),
               'private_data': False, 'private_source_reads': 0,
               'automatic_retries': 0, 'private_boundary_qualified': False, 'runtime_qualified': False}
except BaseException as error:
    save('failure.json', {'request_digest': digest(q), 'error_type': type(error).__name__,
         'container_id': cid, 'attempt_consumed': True, 'automatic_retry_forbidden': True,
         'coupled_work_total': None})
    raise
finally:
    if cid is not None:
        deadline = time.monotonic() + q['cleanup_seconds']
        current = api.inspect(cid, deadline=deadline)
        if current is not None:
            verify_owned(current, 'public', q['nonce'], cid=cid, image_id=q['image_id'])
            save('delete-intent.json', {'container_id': cid, 'force': True})
            status, raw = api._request('DELETE', '/containers/' + cid + '?force=1', deadline=deadline)
            assert status == 204 and raw == b''
        absent = api.inspect(cid, deadline=deadline) is None
        save('cleanup.json', {'container_id': cid, 'absent': absent, 'owner_loss_qualified': False})
        assert absent
if outcome is not None:
    outcome['original_container_absent'] = True
    save('complete.json', outcome)
    print(canonical_json({'container_starts': 1, 'coupled_model_starts': 24,
         'score_within_frozen_tolerance': True, 'original_container_absent': True}))
