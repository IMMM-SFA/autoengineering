"""Prepare exactly one fixed-worker check over exposed authored values."""
import ast, hashlib, json, os, uuid
from pathlib import Path
from autoengineering.campaign.serialization import canonical_json, digest, unique_json
from examples.utility import guest_resources as policy
from examples.utility import engineering_load_entrypoint as entry
from examples.utility import engineering_load_worker as worker

root = Path.cwd(); os.umask(0o077)
p = root / 'examples/utility/verification-engineering-linux-worker-v1'
work = root / 'outputs/engineering-linux-worker-v1'
assert not work.exists(); work.mkdir(mode=0o700)
image_root = root / 'examples/utility/verification-engineering-worker-image-v1'
image = unique_json((image_root / 'complete.json').read_text())
build = unique_json((image_root / 'request.json').read_text())
assert image['request_digest'] == digest(build)
for name, pin in build['environment']['source_manifest']['files'].items():
    assert hashlib.sha256((root / name).read_bytes()).hexdigest() == pin
assert hashlib.sha256(Path(entry.__file__).read_bytes()).hexdigest() == build['environment']['source_manifest']['entrypoint_sha256']
previous = unique_json((root / 'examples/utility/verification-engineering-linux-load-probe-v1/request.json').read_text())
mac = unique_json((root / 'examples/utility/verification-engineering-load-worker-v1/public-worker-output.json').read_text())
packet = {'schema': 'utility-coupled-load-packet-v1',
          'execution_digest': digest({'development': 'linux-fixed-worker-v1'}),
          'source_binding_digest': digest({'source': 'exposed nominal authored worker fixture'}),
          'model_digest': previous['expected_model_digest'],
          'kernel_digest': hashlib.sha256((root / 'examples/utility/engineering_load_scoring.py').read_bytes()).hexdigest(),
          'source_current_A': .05, 'minimum_node2_voltage_V': .62,
          'cases': [{'scenario_id': f'authored-{i}', 'weight': 1 / 24,
                     'forcing': {'ground_voltage': .025, 'resistor1_ohm': 110.,
                                 'resistor2_ohm': 8000.}} for i in range(24)]}
raw = canonical_json(packet).encode(); pin = hashlib.sha256(raw).hexdigest()
worker.parse_packet(raw, expected_sha256=pin)
nonce = uuid.uuid4().hex
config = policy._configuration('read_only_root', nonce, 'fixed image entrypoint',
                               image_id=policy.CURRENT_IMAGE)
config.update(Image=image['image_id'], Entrypoint=build['entrypoint'], Cmd=[pin],
              AttachStdin=True, OpenStdin=True, StdinOnce=True)
config['Labels'] = {**build['labels'], 'org.autoengineering.nonce': nonce,
                    'org.autoengineering.packet': pin}
request = {'scope': 'one fixed Linux worker control with exposed authored packet',
           'configuration': config, 'image_id': image['image_id'],
           'image_environment_digest': image['environment_digest'],
           'nonce': nonce, 'container_name': 'ae-guest-resource-' + nonce,
           'packet_sha256': pin, 'packet_bytes': len(raw),
           'expected_score': mac['result']['score'], 'atol': 1e-10, 'rtol': 1e-12,
           'expected_model_starts': 24, 'max_creations': 1, 'max_starts': 1,
           'active_seconds': 60, 'cleanup_seconds': 10, 'max_output_bytes': 16384,
           'api_version': previous['api_version'], 'server_digest': previous['server_digest'],
           'entrypoint_source_sha256': hashlib.sha256(Path(entry.__file__).read_bytes()).hexdigest(),
           'worker_source_sha256': hashlib.sha256(Path(worker.__file__).read_bytes()).hexdigest(),
           'automatic_retries': 0, 'private_source_reads': 0, 'private_boundary_qualified': False}
for target in (p, work):
    (target / 'request.json').write_text(json.dumps(request, indent=2) + '\n')
    (target / 'packet.json').write_bytes(raw)
# Reuse the strict observation checks, without adding the image to any catalogue.
source = Path(policy.__file__).read_text(); tree = ast.parse(source)
function = ast.get_source_segment(source, next(n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == 'verify_owned'))
a = function.index('    expected = ('); b = function.index('    if (', a)
function = function[:a] + '    expected = CONFIG_EXPECTED\n' + function[b:]
(p / 'verify-owned.py.txt').write_text(function)
print(canonical_json({'request_digest': digest(request), 'packet_sha256': pin,
                      'image_id': image['image_id'], 'model_starts_reserved': 24}))
