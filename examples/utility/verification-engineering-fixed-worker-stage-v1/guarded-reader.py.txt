"""Run the saved stdlib reader with imports, runtime and outside reads denied."""
from pathlib import Path
import hashlib
import json
import math
import os
import socket
import sys
import time

root = Path(sys.argv[1]).resolve(strict=True)
pin = sys.argv[2]
source = (root / 'cold-reader.py.txt').read_bytes()
hashlib.sha256(b'initialize').hexdigest()

def denied(*args, **kwargs): raise RuntimeError('forbidden report-only operation')

class DenyImports:
    def find_spec(self, *args, **kwargs): return denied()

sys.meta_path.insert(0, DenyImports())
for name in ('monotonic', 'time', 'perf_counter', 'process_time', 'thread_time'):
    setattr(time, name, denied)
socket.socket = denied
for name in ('fork', 'forkpty', 'posix_spawn', 'posix_spawnp', 'system'):
    if hasattr(os, name): setattr(os, name, denied)

def audit(event, args):
    if event.startswith(('socket.', 'subprocess.', 'os.exec', 'os.spawn')):
        denied()
    if event == 'open':
        name, mode, flags = args
        if not isinstance(name, (str, bytes, os.PathLike)): denied()
        if flags & (os.O_WRONLY | os.O_RDWR | os.O_CREAT | os.O_TRUNC | os.O_APPEND): denied()
        path = Path(os.fsdecode(name)).resolve(strict=True)
        if not path.is_relative_to(root): denied()

sys.addaudithook(audit)
sys.argv = [str(root / 'cold-reader.py.txt'), str(root), pin]
exec(compile(source, 'fixed-metadata-reader', 'exec'), {'__name__': '__main__'})
