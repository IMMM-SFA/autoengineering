"""Offline copied-input wheel build and preserved/successor environment verification."""
from pathlib import Path
import hashlib
import json
import os
import shutil
import subprocess
import sys
import tempfile
import zipfile

root = Path.cwd()
sys.path.insert(0, str(root))

from autoengineering.campaign.artifacts import reference
from examples.utility.inference_environment import prepare_inputs, verify_inputs

out = root / 'examples/utility/verification-guest-current-wheel-v1'
old = root / 'examples/utility/verification-inference-guest-v1/wheel-source-pins.json'
names = set(json.loads(old.read_text()))
names.update(str(p.relative_to(root)) for p in (root / 'src/autoengineering').rglob('*.py'))
names.update({'pyproject.toml', 'pixi.lock', 'LICENSE', 'NOTICE', 'README.md'})
pins = {name: hashlib.sha256((root / name).read_bytes()).hexdigest() for name in sorted(names)}
(out / 'wheel-source-pins.json').write_text(json.dumps(pins, indent=2) + '\n')
with tempfile.TemporaryDirectory(prefix='guest-current-wheel-', dir='/private/tmp') as temporary:
    build = Path(temporary)
    for name in sorted(names):
        destination = build / name
        destination.parent.mkdir(mode=0o700, parents=True, exist_ok=True)
        shutil.copyfile(root / name, destination)
    script = "from setuptools.build_meta import build_wheel; build_wheel('dist')"
    environment = os.environ.copy()
    environment['SOURCE_DATE_EPOCH'] = '1793318400'
    subprocess.run([sys.executable, '-c', script], cwd=build, env=environment, check=True)
    wheel = next((build / 'dist').glob('*.whl'))
    saved = out / wheel.name
    if saved.exists():
        raise ValueError('wheel output already exists')
    shutil.copyfile(wheel, saved)
with zipfile.ZipFile(saved) as archive:
    expected = {str(p.relative_to(root / 'src')): p.read_bytes()
                for p in (root / 'src/autoengineering').rglob('*.py')}
    found = {name: archive.read(name) for name in archive.namelist()
             if name.startswith('autoengineering/') and name.endswith('.py')}
    assert found == expected
assert all(hashlib.sha256((root / name).read_bytes()).hexdigest() == value
           for name, value in pins.items())
lock = reference(root, 'pixi.lock', media_type='text/plain', producer='pixi')
wheelref = reference(root, str(saved.relative_to(root)), media_type='application/zip', producer='build')
inputs = prepare_inputs(root, lock=lock, wheel=wheelref,
                        output=root / 'examples/utility/prepared-inference-environment-v4')
assert verify_inputs(root / 'examples/utility/prepared-inference-environment-v4',
                     expected_digest=inputs.digest) == inputs
preserved = {
 'prepared-inference-environment-v3': '9402879f689a26da4dcb07573e3f007efbc7cd4fa3451f6e0c41ae40667f0966',
 'prepared-inference-environment-v1': 'd5db58cf78c57526855a070b15cf629524f1f37742eb7bc42f4e199aee127453',
 'prepared-inference-environment-v2': '5eee3a23d3574a7f3c160f9ebfbedd9f1670f7db4e7b79f1edeef1168aeeb60b',
}
for name, digest in preserved.items():
    original = verify_inputs(root / 'examples/utility' / name, expected_digest=digest)
    assert original.archives == inputs.archives
assert any(r.path == 'autoengineering/utility/guest_creation_cleanup.py'
           for r in inputs.wheel_python_sources)
summary = dict(digest=inputs.digest, wheel_sha256=wheelref.sha256,
               build_inputs=len(pins), core_python_files=len(expected),
               wheel_python_sources=len(inputs.wheel_python_sources),
               archives=len(inputs.archives), compressed_bytes=inputs.compressed_archive_bytes,
               preserved_inputs=preserved, installed=False, image_built=False, runtime_qualified=False)
(out / 'prepared-inputs.json').write_text(json.dumps(summary, indent=2) + '\n')
print(json.dumps(summary, sort_keys=True))
