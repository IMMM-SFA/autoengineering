"""Diagnostic owner: acknowledge, arm watchdog, optionally start without input."""
from pathlib import Path
import json,os,sys,time
from autoengineering.campaign.serialization import canonical_json,digest
from autoengineering.campaign.store import durable_write
from autoengineering.utility.guest_daemon import GuestDaemonAPI
from examples.utility import engineering_container_runtime as owner
from examples.utility.engineering_container_watchdog import CoupledWatchdog

root=Path(sys.argv[1]);q=json.loads((root/'request.json').read_bytes())
api=GuestDaemonAPI(q['endpoint'],api_version=q['api_version'],server_digest=q['server_digest'])
begin=time.monotonic();deadline=begin+60;active=deadline-10
def save(name,value):durable_write(root/name,canonical_json(value).encode())
assert digest(api._json('GET','/version',deadline=active))==q['server_digest']
owner.verify_image(api._json('GET','/images/'+owner.IMAGE+'/json',deadline=active))
save('create-intent.json',{'name':q['name'],'configuration_digest':digest(q['configuration'])})
created=api._json('POST','/containers/create?name='+q['name'],q['configuration'],expected=201,deadline=active)
cid=created['Id'];save('created.json',created)
before=api.inspect(cid,deadline=active)
owner.verify_owned(before,q['configuration'],cid=cid,name=q['name']);save('before.json',before)
watchdog=CoupledWatchdog(root,api,config=q['configuration'],cid=cid,name=q['name'],active_deadline=active,deadline=deadline)
ready=watchdog.ready()
if q['mode']=='running_before_input':
    save('start-intent.json',{'container_id':cid,'starts_reserved':1,'packet_sent':False})
    status,body=api._request('POST','/containers/'+cid+'/start',deadline=active)
    assert status==204 and body==b''
    running=api.inspect(cid,deadline=active)
    owner.verify_owned(running,q['configuration'],cid=cid,name=q['name'])
    assert running['State']['Running'] is True
    save('running.json',running)
save('owner-boundary.json',{'owner_pid':os.getpid(),'watchdog_pid':ready['watchdog_pid'],
    'container_id':cid,'mode':q['mode'],'packet_sent':False})
while True:time.sleep(.1)
