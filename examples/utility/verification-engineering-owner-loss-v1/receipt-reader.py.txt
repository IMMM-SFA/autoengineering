"""Join copied watchdog producer receipts without imports from the project."""
from pathlib import Path
import hashlib,json,sys
root=Path(sys.argv[1]).resolve(strict=True)
manifest_raw=(root/'evidence-manifest.json').read_bytes()
assert hashlib.sha256(manifest_raw).hexdigest()==sys.argv[2]
manifest=json.loads(manifest_raw)
for name,expected in manifest['files'].items():
    path=root/name
    assert not Path(name).is_absolute() and '..' not in Path(name).parts
    assert not any(p.is_symlink() for p in (path,*path.parents))
    raw=path.read_bytes()
    assert len(raw)==expected['bytes'] and hashlib.sha256(raw).hexdigest()==expected['sha256']
def read(name):
    assert name in manifest['files']
    return json.loads((root/name).read_bytes())
def digest(value):
    return hashlib.sha256(json.dumps(value,sort_keys=True,separators=(',',':'),allow_nan=False).encode()).hexdigest()
summary=read('summary.json')
for case in summary['cases']:
    prefix=case['mode']+'/'
    q=read(prefix+'request.json');plan=read(prefix+'watchdog-plan.json')
    ready=read(prefix+'watchdog-ready.json');outcome=read(prefix+'watchdog-outcome.json')
    boundary=read(prefix+'owner-boundary.json');kill=read(prefix+'kill-intent.json')
    cid=read(prefix+'created.json')['Id']
    assert plan['container_id']==ready['container_id']==outcome['container_id']==case['container_id']==cid
    assert ready['plan_digest']==outcome['plan_digest']==digest(plan)
    assert plan['configuration']==q['configuration'] and plan['name']==q['name']
    assert plan['owner_source_digest']==q['owner_source_sha256']
    assert plan['watchdog_source_digest']==q['watchdog_source_sha256']
    assert ready['watchdog_pid']==outcome['watchdog_pid']==boundary['watchdog_pid']==kill['watchdog_pid']
    assert boundary['owner_pid']==kill['owner_pid']==case['owner_pid']
    assert kill['signal']=='SIGKILL' and outcome['reason']=='owner_lost'
    assert outcome['absent'] is True and case['original_container_absent'] is True
    absence=read(prefix+'watchdog-process-absence.json')
    assert absence['watchdog_pid']==ready['watchdog_pid'] and absence['absent'] is True
    assert not absence['ps_stdout'].strip()
    assert boundary['packet_sent'] is False and case['source_reads']==0
    if case['mode']=='running_before_input':
        running=read(prefix+'running.json')
        assert running['Id']==cid and running['State']['Running'] is True
        assert read(prefix+'start-intent.json')['packet_sent'] is False
    else:assert read(prefix+'before.json')['State']['Status']=='created'
assert summary['container_creations']==2 and summary['container_starts']==1
assert summary['runtime_qualified'] is False and summary['complete_interruption_matrix'] is False
print(json.dumps({'verified':True,'container_creations':2,'container_starts':1,
                  'owner_sigkills':2,'private_payloads_submitted':0,'runtime_qualified':False},sort_keys=True))
