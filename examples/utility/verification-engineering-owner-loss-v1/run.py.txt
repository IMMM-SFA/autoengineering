"""Two exact-owner SIGKILL controls; no packet, source read or science retry."""
from pathlib import Path
import hashlib,json,os,signal,subprocess,sys,time,uuid
from autoengineering.campaign.serialization import canonical_json,digest
from autoengineering.utility.guest_daemon import GuestDaemonAPI
from examples.utility import engineering_container_runtime as owner

root=Path.cwd();os.umask(0o077)
p=root/'examples/utility/verification-engineering-owner-loss-v1'
work=root/'outputs/engineering-owner-loss-v1';assert not work.exists();work.mkdir(mode=0o700)
old=json.loads((root/'examples/utility/verification-engineering-linux-worker-v1/request.json').read_bytes())
endpoint='/Users/cam/.orbstack/run/docker.sock'
api=GuestDaemonAPI(endpoint,api_version=old['api_version'],server_digest=old['server_digest'])
results=[]
for mode in ('ready_unstarted','running_before_input'):
    scope=work/mode;scope.mkdir(mode=0o700)
    nonce=uuid.uuid4().hex;name='ae-coupled-load-'+nonce
    configuration=owner.configuration(nonce,'1'*64,digest({'control':mode}))
    q={'schema':'utility-coupled-owner-loss-control-v1','mode':mode,'name':name,
       'configuration':configuration,'endpoint':endpoint,'api_version':old['api_version'],
       'server_digest':old['server_digest'],'max_creations':1,'max_starts':int(mode=='running_before_input'),
       'private_payloads_submitted':0,'source_reads':0,'owner_source_sha256':owner.source_identity(),
       'watchdog_source_sha256':owner.supervision.source_identity(),'automatic_science_retries':0}
    (scope/'request.json').write_text(canonical_json(q))
    bootstrap='import sys,runpy;sys.path.insert(0,sys.argv[1]);sys.argv=sys.argv[2:];runpy.run_path(sys.argv[0],run_name="__main__")'
    argv=[sys.executable,'-I','-B','-c',bootstrap,str(root),str(p/'owner.py.txt'),str(scope)]
    with (scope/'owner-output.log').open('xb') as output:
        process=subprocess.Popen(argv,stdin=subprocess.DEVNULL,stdout=output,stderr=subprocess.STDOUT,start_new_session=True)
        try:
            limit=time.monotonic()+15
            while not (scope/'owner-boundary.json').exists():
                assert process.poll() is None,'owner failed before boundary'
                assert time.monotonic()<limit,'owner boundary timed out'
                time.sleep(.05)
            boundary=json.loads((scope/'owner-boundary.json').read_bytes())
            assert boundary['owner_pid']==process.pid and boundary['packet_sent'] is False
            (scope/'kill-intent.json').write_text(canonical_json({'owner_pid':process.pid,'signal':'SIGKILL',
                'watchdog_pid':boundary['watchdog_pid'],'container_id':boundary['container_id']}))
            os.kill(process.pid,signal.SIGKILL)
            assert process.wait(timeout=5)==-signal.SIGKILL
            limit=time.monotonic()+15
            while not (scope/'watchdog-outcome.json').exists():
                assert time.monotonic()<limit,'watchdog outcome timed out'
                time.sleep(.05)
            outcome=json.loads((scope/'watchdog-outcome.json').read_bytes())
            assert outcome['reason']=='owner_lost' and outcome['absent'] is True
            assert api.inspect(boundary['container_id'],deadline=time.monotonic()+5) is None
            limit=time.monotonic()+5
            while True:
                ps=subprocess.run(['/bin/ps','-p',str(boundary['watchdog_pid']),'-o','pid=,stat='],capture_output=True,text=True)
                if not ps.stdout.strip():break
                assert time.monotonic()<limit,'watchdog process still present'
                time.sleep(.05)
            (scope/'watchdog-process-absence.json').write_text(canonical_json({'watchdog_pid':boundary['watchdog_pid'],
                'ps_exit_code':ps.returncode,'ps_stdout':ps.stdout,'absent':True}))
            results.append({'mode':mode,'container_id':boundary['container_id'],'owner_pid':process.pid,
                'watchdog_pid':boundary['watchdog_pid'],'container_creations':1,'container_starts':q['max_starts'],
                'owner_sigkill_observed':True,'original_container_absent':True,'watchdog_process_absent':True,
                'packet_submitted':False,'source_reads':0,'model_results_observed':0})
            print(canonical_json(results[-1]),flush=True)
        finally:
            if process.poll() is None:os.kill(process.pid,signal.SIGKILL);process.wait(timeout=5)
summary={'schema':'utility-coupled-owner-loss-development-v1','cases':results,'container_creations':2,
    'container_starts':1,'owner_sigkills':2,'private_payloads_submitted':0,'source_reads':0,
    'model_results_observed':0,'model_work_total':None,'known_model_starts_lower_bound':0,
    'complete_interruption_matrix':False,'pre_acknowledgement_creation_loss_qualified':False,
    'during_model_computation_loss_qualified':False,'private_boundary_qualified':False,'runtime_qualified':False}
(p/'summary.json').write_text(json.dumps(summary,indent=2)+'\n')
import shutil
for mode in ('ready_unstarted','running_before_input'):shutil.copytree(work/mode,p/mode)
