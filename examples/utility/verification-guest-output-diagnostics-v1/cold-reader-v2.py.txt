import sys
import importlib.abc
from pathlib import Path
sys.path.insert(0,str(Path(sys.argv[2]).resolve()))
class NoModels(importlib.abc.MetaPathFinder):
    def find_spec(self,fullname,path=None,target=None):
        if fullname.split('.')[0] in {'torch','pvlib','openmdao','py_wake','topfarm','bmipy'} or fullname.startswith(('examples.leaf_river','examples.utility.hydrology')):
            raise RuntimeError('model import denied')
sys.meta_path.insert(0,NoModels())
import json,os,socket,subprocess,time
import numpy
from examples.utility.guest_output_reader import read_completion
root=Path(sys.argv[1]).resolve()
expected=json.loads((root/'expected-reader-pins.json').read_text())
inodes={(p.stat().st_dev,p.stat().st_ino) for p in root.rglob('*') if p.is_file()}
def denied(*a,**kw):raise RuntimeError('execution network clock or archive load denied')
subprocess.Popen=socket.socket=numpy.load=time.time=time.monotonic=denied

def audit(event,args):
    if event=='open':
        path,mode,flags=args
        if flags&(os.O_WRONLY|os.O_RDWR|os.O_CREAT|os.O_TRUNC|os.O_APPEND):raise RuntimeError('write denied')
        if isinstance(path,int):
            st=os.fstat(path)
            if (st.st_dev,st.st_ino) not in inodes:raise RuntimeError('outside descriptor denied')
        elif not Path(os.fsdecode(path)).absolute().is_relative_to(root):raise RuntimeError('outside read denied')
    if event=='import' or event.startswith(('socket.','subprocess.','os.exec','os.fork','os.posix_spawn')):
        raise RuntimeError('new import or process denied')
sys.addaudithook(audit)
result=read_completion(root,**expected)
assert result['container_starts']==2 and result['scientific_starts']==0
print(json.dumps({'cold_read':True,'cases':2,'diagnostic_container_starts':2,'scientific_starts':0,'models_imported':False,'new_processes':0,'network':False,'private_input_reads':False,'writes':False}))
