"""Copy completed authored callback metadata; never reopen cases or call runtime."""
from pathlib import Path
import hashlib, json, shutil
from autoengineering.campaign.artifacts import reference
from autoengineering.utility.load_costs import CostEvidenceSpec, LoadCostInput, LoadCostRegistration
from autoengineering.utility.scoring import ConfirmationGrant
from autoengineering.utility.store import SuiteStore
from autoengineering.utility.report import accounting_report, render_markdown

root=Path.cwd(); out=root/'examples/utility/verification-engineering-container-callback-v1'
original=Path('/private/tmp/utility-container-cost-final-v5/test_source_to_runtime_once_an0')
public=out/'public'; public.mkdir(mode=0o700)
shutil.copytree(original/'suite',public/'suite')
store=SuiteStore(public/'suite'); state=store.replay()
access=next(iter(state['confirmation_accesses'].values()))
grant=ConfirmationGrant.from_json((store.root/'artifacts'/access/'confirmation-grant.json').read_text())
# The original test completed this exact scope; registration is recovered from
# the original private completion's registration digest and known source pins.
# No generated forcing rows, seed or private source payload is copied.
from autoengineering.utility.load_container_costs import ContainerLoadCostRegistration,CoupledRuntimeRegistration
from autoengineering.utility.load_costs import LoadScorePolicy
from autoengineering.utility.confirmation import read_manifest,read_selected_state
selected=next(c for c in read_manifest(store.root,state['frozen']).cells if c.cell_digest==grant.cell_digest)
model=read_selected_state(store.root,selected.selected_state).model
H='1'*64
kernel_sha=hashlib.sha256((root/'examples/utility/engineering_load_scoring.py').read_bytes()).hexdigest()
runtime=CoupledRuntimeRegistration('utility-coupled-runtime-registration-v1','sha256:'+H,H,H,H,H,kernel_sha,H)
binding_ref=reference(store.root,'engineering-source-bindings/task.json',media_type='application/json',producer='engineering-source')
registration=ContainerLoadCostRegistration(schema='utility-load-cost-registration-v2',binding_artifact=binding_ref,
    callback_source_digest=hashlib.sha256((root/'examples/utility/engineering_container_callback.py').read_bytes()).hexdigest(),
    policy=LoadScorePolicy('utility-coupled-load-score-v1',model.digest,kernel_sha,.62),runtime=runtime)
anchor=json.loads((original/'work'/grant.digest/'complete.json').read_bytes())
assert anchor['registration_digest']==registration.digest
ref=reference(store.root,'engineering-container-load-work/'+grant.digest+'.json',media_type='application/json',producer='engineering-container-load-work')
assert ref.to_dict()==anchor['work_artifact']
inputs=CostEvidenceSpec('utility-cost-inputs-v1',store.spec.digest,(LoadCostInput(grant,registration,ref),))
(public/'cost-inputs.json').write_text(inputs.to_json()+'\n')
legacy=accounting_report(store); current=accounting_report(store,cost_inputs=inputs)
(public/'expected-legacy.json').write_text(json.dumps(legacy,sort_keys=True,separators=(',',':'))+'\n')
(public/'expected-accounting.json').write_text(json.dumps(current,sort_keys=True,separators=(',',':'))+'\n')
(public/'expected-accounting.md').write_text(render_markdown(current))
pins={str(f.relative_to(public)):hashlib.sha256(f.read_bytes()).hexdigest() for f in sorted(public.rglob('*')) if f.is_file()}
(out/'public-pins.json').write_text(json.dumps(pins,indent=2)+'\n')
print(json.dumps({'public_files':len(pins),'runtime_is_authored_double':True,'actual_model_starts':0,'actual_container_starts':0,'private_payloads_copied':0,'full_cost_evidence':current['complete_cost_evidence']}))
