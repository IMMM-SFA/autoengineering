ROOT = Path(__file__).resolve().parents[1]
H = "1" * 64


class AuthoredRuntime:
    """No daemon, subprocess or physical model. All output/counters are authored."""

    def __init__(self, registration, store, change=None):
        self.registration, self.store, self.change = registration, store, change
        self.preflights = self.runs = 0

    def source_identity(self):
        return H

    def preflight(self, grant):
        self.preflights += 1
        assert not (self.store.root / binding.ACCESS_DIRECTORY).exists()
        if self.change == "preflight":
            raise ValueError("PRIVATE-PREFLIGHT-CANARY")

    def run(self, grant, packet, *, expected_sha256, work_root):
        self.runs += 1
        assert hashlib.sha256(packet).hexdigest() == expected_sha256
        value = json.loads(packet)
        assert value["execution_digest"] == grant.digest and len(value["cases"]) == 24
        if self.change == "runtime":
            raise ValueError("PRIVATE-RUNTIME-CANARY")
        output = json.loads(
            (
                ROOT / "examples/utility/verification-engineering-linux-worker-v1/output.log"
            ).read_bytes()
        )
        work = output["worker"]["result"]
        work["score"]["value"] = 0.25
        work["known_completed_inner_work"] = {
            "solver_iterations": 96,
            "component_calls": 480,
            "derivative_calls": 24,
        }
        if self.change == "partial":
            work.update(
                attempted_model_starts=3,
                completed_model_starts=2,
                last_attempt_inner_work_unknown=True,
            )
            work["score"].update(status="model_failure", value=None, feasible=None)
            work["known_completed_inner_work"] = {
                "solver_iterations": 8,
                "component_calls": 40,
                "derivative_calls": 2,
            }
        output["worker"]["captured_bytes"] = len(canonical_json(work).encode()) + 1
        from autoengineering.utility.scoring import PublicScore

        score = PublicScore.from_dict(work["score"])
        r = self.registration
        execution = ConfirmationContainerExecution(
            "utility-confirmation-container-execution-v1",
            grant.digest,
            grant.scorer_digest,
            score.digest,
            r.digest,
            H,
            "a" * 32,
            r.image_id,
            r.runtime_source_digest,
            r.entrypoint_source_digest,
            r.worker_source_digest,
            H,
            H,
            H,
            1,
            1,
            0.5,
        )
        if self.change == "foreign":
            execution = replace(execution, image_id="sha256:" + "2" * 64)
        return callback.ContainerLoadRun(
            canonical_json(output).encode(), execution, expected_sha256
        )


def setup(tmp_path, monkeypatch, change=None):
    monkeypatch.setattr(engineering, "version", lambda _: "3.41.0")
    model = load.context(0.05, tmp_path, source_root=ROOT).model
    policy = LoadScorePolicy(
        "utility-coupled-load-score-v1", model.digest, kernel.source_identity(), 0.62
    )
    source_policy = source.ScenarioPolicy(
        "utility-engineering-generated-scenarios-v1",
        "task",
        {
            "ground_voltage": (-0.05, 0.05),
            "resistor1_ohm": (95.0, 105.0),
            "resistor2_ohm": (9500.0, 10500.0),
        },
        source.source_identity(),
    )
    public, private, work = (tmp_path / n for n in ("public", "private", "work"))
    for p in (public, private, work):
        p.mkdir(mode=0o700)
    monkeypatch.setattr(source.secrets, "token_bytes", lambda n: b"a" * n)
    runtime_registration = CoupledRuntimeRegistration(
        "utility-coupled-runtime-registration-v1",
        "sha256:" + H,
        H,
        H,
        H,
        H,
        kernel.source_identity(),
        H,
    )
    captured = {}
    resources = (
        ResourceCap("scientific_starts", "count", 1, "reservation"),
        ResourceCap("load_model_starts", "count", 24, "reservation"),
        ResourceCap("load_solver_iterations", "count", 2400, "reservation"),
        ResourceCap("container_starts", "count", 1, "reservation"),
        ResourceCap("active_time", "s", 60, "reservation"),
    )

    def before(store):
        for ref in model.source_bundle:
            target = store.root / ref.path
            target.parent.mkdir(mode=0o700, parents=True, exist_ok=True)
            target.write_bytes(read_bytes(ROOT, ref))
        metadata = source.prepare(
            store, source_policy, public, private, expected_suite_digest=store.spec.digest
        )
        artifact = binding.seal(store, public, expected_manifest_digest=digest(metadata))
        registration = ContainerLoadCostRegistration(
            binding_artifact=artifact,
            callback_source_digest=callback.source_identity(),
            policy=policy,
            schema="utility-load-cost-registration-v2",
            runtime=runtime_registration,
        )
        runtime = AuthoredRuntime(runtime_registration, store, change)
        scorer = callback.ContainerLoadScorer(
            store, public, private, work, registration=registration, runtime=runtime
        )
        captured.update(scorer=scorer, runtime=runtime, registration=registration)

    def transform(spec):
        budget = spec.budgets[0]
        return replace(
            spec,
            tasks=(replace(spec.tasks[0], partition_digest=source_policy.digest),),
            cells=tuple(
                replace(c, scenario_digest=source_policy.digest, regime="fixed_evaluations")
                for c in spec.cells
            ),
            budgets=(
                replace(
                    budget,
                    regime="fixed_evaluations",
                    caps=(*budget.caps, *resources[1:4]),
                    confirmation_reserve=(*budget.confirmation_reserve, *resources[1:]),
                ),
            ),
        )

    # Outcome is unchanged by runtime choice. It is sealed before selection.
    from examples.utility.engineering_load_callback import outcome_contract

    store, manifest, contract = selected_suite(
        tmp_path,
        suite_transform=transform,
        before_selection=before,
        model_override=model,
        selected_configuration={"source_current": 0.05},
        outcome_contract_override=outcome_contract(policy),
    )
    scorer = captured["scorer"]
    grant = ConfirmationGrant(
        store.spec.digest,
        manifest.cells[0].cell_digest,
        source_policy.digest,
        scorer.identity,
        contract.digest,
        manifest.cells[0].model.sha256,
        store.replay()["frozen"],
        resources,
    )
    service = ConfirmationService(
        store, scorer, scorer_digest=scorer.identity, outcome_contract=contract
    )
    return store, grant, service, captured


def work_ref(store, grant):
    return reference(
        store.root,
        f"{WORK_DIRECTORY}/{grant.digest}.json",
        media_type="application/json",
        producer="engineering-container-load-work",
    )


def test_source_to_runtime_once_and_cost_inputs_v2_reconstruct_without_runtime(
    tmp_path, monkeypatch
):
    store, grant, service, captured = setup(tmp_path, monkeypatch)
    score = service.confirm(grant)
    assert score.score.value == 0.25 and score.execution.container_starts == 1
    runtime = captured["runtime"]
    assert (runtime.preflights, runtime.runs) == (1, 1)
    ref = work_ref(store, grant)
    inputs = CostEvidenceSpec(
        "utility-cost-inputs-v1",
        store.spec.digest,
        (LoadCostInput(grant, captured["registration"], ref),),
    )
    assert CostEvidenceSpec.from_json(inputs.to_json()) == inputs
    monkeypatch.setattr(runtime, "preflight", lambda *a: pytest.fail("runtime touched"))
    monkeypatch.setattr(runtime, "run", lambda *a, **kw: pytest.fail("runtime rerun"))
    monkeypatch.setattr(source, "read_private", lambda *a, **kw: pytest.fail("source reopened"))
    assert service.confirm(grant) == score
    records = join_cost_inputs(store, inputs)
    values = {r.resource: r.quantity for r in records}
    assert values["load_model_starts"] == values["load_case_attempts"] == 24
    assert values["load_solver_iterations"] == 96
    generic = {r.resource: r.quantity for r in join_confirmation_costs(store)}
    assert generic["confirmation_container_starts"] == 1
    assert generic["confirmation_container_lifecycle_duration"] == 0.5
    missing = join_load_costs(store, grant, captured["registration"])
    assert {r.resource: r.quantity for r in missing}["load_model_starts"] is None
    assert [r.receipt_id for r in missing] == [r.receipt_id for r in records]


@pytest.mark.parametrize("change", ["preflight", "runtime", "foreign"])
def test_failed_runtime_remains_charged_and_cannot_open_or_execute_again(
    tmp_path, monkeypatch, change
):
    store, grant, service, captured = setup(tmp_path, monkeypatch, change)
    with pytest.raises(StoreError, match="uncertain"):
        service.confirm(grant)
    attempts = captured["runtime"].runs
    assert captured["runtime"].preflights == 1
    assert attempts == (0 if change == "preflight" else 1)
    with pytest.raises(StoreError):
        service.confirm(grant)
    assert captured["runtime"].runs == attempts
    assert (store.root / binding.ACCESS_DIRECTORY).exists() is (change != "preflight")
    values = {
        r.resource: r.quantity for r in join_load_costs(store, grant, captured["registration"])
    }
    assert values["load_model_starts"] is None and values["load_model_start_reservations"] == 24


def test_model_failure_preserves_known_work_without_total_or_objective(tmp_path, monkeypatch):
    store, grant, service, captured = setup(tmp_path, monkeypatch, "partial")
    score = service.confirm(grant)
    assert score.score.status == "model_failure" and score.score.value is None
    values = {
        r.resource: r.quantity
        for r in join_load_costs(
            store, grant, captured["registration"], work_artifact=work_ref(store, grant)
        )
    }
    assert values["load_case_attempts"] == 3 and values["load_completed_evaluations"] == 2
    assert values["load_model_starts"] is None and values["load_solver_iterations"] is None
    assert values["load_known_completed_solver_iterations"] == 8


@pytest.mark.parametrize("change", ["source", "runtime", "allowance", "active_time"])
def test_invalid_registration_or_allowance_reads_no_source(tmp_path, monkeypatch, change):
    store, grant, service, captured = setup(tmp_path, monkeypatch)
    if change == "source":
        monkeypatch.setattr(callback, "source_identity", lambda: H)
    elif change == "runtime":
        captured["runtime"].registration = replace(
            captured["runtime"].registration, image_id="sha256:" + "2" * 64
        )
    elif change == "allowance":
        grant = replace(
            grant, resources=tuple(r for r in grant.resources if r.name != "container_starts")
        )
    else:
        grant = replace(
            grant,
            resources=tuple(
                replace(r, limit=1) if r.name == "active_time" else r for r in grant.resources
            ),
        )
    monkeypatch.setattr(
        source, "read_private", lambda *a, **kw: pytest.fail("private source opened")
    )
    with pytest.raises(StoreError, match="uncertain"):
        service.confirm(grant)
    assert captured["runtime"].preflights == captured["runtime"].runs == 0
    assert not (store.root / binding.ACCESS_DIRECTORY).exists()


@pytest.mark.parametrize("change", ["source", "image", "output", "role"])
def test_rehashed_foreign_work_proof_is_rejected(tmp_path, monkeypatch, change):
    from autoengineering.utility.load_container_costs import ContainerLoadWorkReceipt

    store, grant, service, captured = setup(tmp_path, monkeypatch)
    service.confirm(grant)
    ref = work_ref(store, grant)
    proof = ContainerLoadWorkReceipt.from_json(read_bytes(store.root, ref).decode())
    if change == "source":
        proof = replace(proof, source_binding_digest=H)
    elif change == "image":
        proof = replace(proof, execution=replace(proof.execution, image_id="sha256:" + "2" * 64))
    elif change == "role":
        ref = replace(ref, producer="foreign")
    else:
        (store.root / proof.output_artifact.path).write_bytes(b"PRIVATE-PAYLOAD-CANARY")
    if change in ("source", "image"):
        (store.root / ref.path).write_text(proof.to_json())
        ref = work_ref(store, grant)
    with pytest.raises(ValueError):
        join_load_costs(store, grant, captured["registration"], work_artifact=ref)


def test_score_commit_loss_recovers_only_original_pin_without_source_or_runtime(
    tmp_path, monkeypatch
):
    store, grant, service, captured = setup(tmp_path, monkeypatch)

    def fault(stage):
        if stage == "confirmation_score_created":
            raise RuntimeError("authored commit loss")

    store.fault = fault
    with pytest.raises(RuntimeError, match="commit loss"):
        service.confirm(grant)
    assert captured["runtime"].runs == 1
    ref = work_ref(store, grant)
    records = join_load_costs(store, grant, captured["registration"], work_artifact=ref)
    assert {r.resource: r.quantity for r in records}["load_model_starts"] == 24
    monkeypatch.setattr(
        captured["runtime"], "run", lambda *a, **kw: pytest.fail("runtime restarted")
    )
    monkeypatch.setattr(
        captured["runtime"], "preflight", lambda *a: pytest.fail("runtime contacted")
    )
    monkeypatch.setattr(source, "read_private", lambda *a, **kw: pytest.fail("source reopened"))
    store.fault = lambda _: None
    receipt = service.recover(grant)
    assert receipt.score.value == 0.25 and receipt.execution.container_starts == 1
    assert service.confirm(grant) == receipt


def test_recovery_cannot_refresh_a_changed_completion_record(tmp_path, monkeypatch):
    store, grant, service, captured = setup(tmp_path, monkeypatch)

    def fault(stage):
        if stage == "confirmation_score_created":
            raise RuntimeError("authored commit loss")

    store.fault = fault
    with pytest.raises(RuntimeError):
        service.confirm(grant)
    ref = work_ref(store, grant)
    (store.root / ref.path).write_text((store.root / ref.path).read_text() + " ")
    store.fault = lambda _: None
    with pytest.raises(StoreError):
        service.recover(grant)
    assert captured["runtime"].runs == 1
    assert not store.replay()["confirmation_scores"]


def test_v2_registration_reaches_cli_accounting_and_verify(tmp_path, monkeypatch):
    from click.testing import CliRunner
    from autoengineering.cli import cli

    store, grant, service, captured = setup(tmp_path, monkeypatch)
    service.confirm(grant)
    inputs = CostEvidenceSpec(
        "utility-cost-inputs-v1",
        store.spec.digest,
        (LoadCostInput(grant, captured["registration"], work_ref(store, grant)),),
    )
    path = tmp_path / "cost-inputs.json"
    path.write_text(inputs.to_json())
    monkeypatch.setattr(
        captured["runtime"], "run", lambda *a, **kw: pytest.fail("runtime restarted")
    )
    monkeypatch.setattr(source, "read_private", lambda *a, **kw: pytest.fail("source reopened"))
    runner = CliRunner()
    result = runner.invoke(cli, ["utility", "report", str(store.root), "--cost-inputs", str(path)])
    assert result.exit_code == 0, result.output
    report = json.loads(result.output)
    totals = {
        r["resource"]: r["quantity"] for r in report["physical_receipt_costs"]["known_totals"]
    }
    assert totals["load_model_starts"] == 24 and totals["confirmation_container_starts"] == 1
    assert not report["complete_cost_evidence"]
    result = runner.invoke(cli, ["utility", "verify", str(store.root), "--cost-inputs", str(path)])
    assert (
        result.exit_code == 0 and json.loads(result.output)["cost_inputs_digest"] == inputs.digest
    )


def test_completion_publication_loss_retains_work_but_does_not_retry_or_recover(
    tmp_path, monkeypatch
):
    store, grant, service, captured = setup(tmp_path, monkeypatch)
    original = callback.durable_write

    def lost(path, raw):
        if path.name == "complete.json":
            raise OSError("PRIVATE-PUBLICATION-CANARY")
        return original(path, raw)

    monkeypatch.setattr(callback, "durable_write", lost)
    with pytest.raises(StoreError, match="uncertain") as error:
        service.confirm(grant)
    assert "PRIVATE" not in str(error.value)
    assert captured["runtime"].runs == 1
    values = {
        r.resource: r.quantity
        for r in join_load_costs(
            store, grant, captured["registration"], work_artifact=work_ref(store, grant)
        )
    }
    assert values["load_model_starts"] == 24
    with pytest.raises(StoreError):
        service.confirm(grant)
    with pytest.raises(StoreError):
        service.recover(grant)
    assert captured["runtime"].runs == 1 and not store.replay()["confirmation_scores"]
