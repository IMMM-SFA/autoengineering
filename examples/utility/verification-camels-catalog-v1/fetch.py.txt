import hashlib,json,time,urllib.request,urllib.error
from pathlib import Path
from datetime import datetime,timezone
root=Path(__file__).resolve().parent
requests=[('zenodo-record.json','https://zenodo.org/api/records/15529996',2*1024**2),('datacite.json','https://api.datacite.org/dois/10.5065/D6MW2F4D',512*1024),('ucar-data-terms.html','https://www.ucar.edu/terms-of-use/data',512*1024)]
for name,url,limit in requests:
    intent={'url':url,'max_body_bytes':limit,'started_utc':datetime.now(timezone.utc).isoformat(),'automatic_retries':False,'status':'intent'}
    path=root/(name+'.request.json')
    with path.open('x') as f: json.dump(intent,f,indent=2)
    start=time.monotonic()
    try:
        with urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'Autoengineering-source-metadata-review/0.2'}),timeout=40) as response:
            data=response.read(limit+1)
            if len(data)>limit: raise ValueError('metadata bound exceeded')
            with (root/name).open('xb') as f:f.write(data)
            intent.update(status='complete',http_status=response.status,final_url=response.url,body_bytes=len(data),sha256=hashlib.sha256(data).hexdigest(),content_type=response.headers.get('Content-Type'))
    except Exception as error:
        intent.update(status='failed',error_type=type(error).__name__)
        if isinstance(error,urllib.error.HTTPError):intent['http_status']=error.code
    intent.update(elapsed_seconds=time.monotonic()-start,total_network_bytes=None,total_http_exchanges=None)
    path.write_text(json.dumps(intent,indent=2)+'\n')
    print(json.dumps({'path':name,**intent}),flush=True)
