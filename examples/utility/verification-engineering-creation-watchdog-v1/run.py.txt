"""Before-CREATE, unbound creation, uncertain intent and normal-binding controls."""
from pathlib import Path
import json,os,signal,subprocess,sys,time,uuid,shutil
from autoengineering.campaign.serialization import canonical_json,digest
from autoengineering.utility.guest_daemon import GuestDaemonAPI
from examples.utility import engineering_container_runtime as owner

root=Path.cwd();os.umask(0o077)
p=root/'examples/utility/verification-engineering-creation-watchdog-v1'
work=root/'outputs/engineering-creation-watchdog-v1';assert not work.exists();work.mkdir(mode=0o700)
previous=json.loads((root/'examples/utility/verification-engineering-linux-worker-v1/request.json').read_bytes())
endpoint='/Users/cam/.orbstack/run/docker.sock'
api=GuestDaemonAPI(endpoint,api_version=previous['api_version'],server_digest=previous['server_digest'])
cases=[]
for mode in ('before_create','created_unbound','intent_only','normal_binding'):
    scope=work/mode;scope.mkdir(mode=0o700)
    nonce=uuid.uuid4().hex;name='ae-coupled-load-'+nonce
    q={'mode':mode,'name':name,'configuration':owner.configuration(nonce,'1'*64,digest({'creation_control':mode})),
       'endpoint':endpoint,'api_version':previous['api_version'],'server_digest':previous['server_digest'],
       'owner_source_sha256':owner.source_identity(),'watchdog_source_sha256':owner.supervision.source_identity(),
       'max_creations':int(mode in ('created_unbound','normal_binding')),'max_starts':0,
       'source_reads':0,'packet_submitted':False}
    (scope/'request.json').write_text(canonical_json(q))
    bootstrap='import sys,runpy;sys.path.insert(0,sys.argv[1]);sys.argv=sys.argv[2:];runpy.run_path(sys.argv[0],run_name="__main__")'
    argv=[sys.executable,'-I','-B','-c',bootstrap,str(root),str(p/'owner.py.txt'),str(scope)]
    with (scope/'owner-output.log').open('xb') as log:
        process=subprocess.Popen(argv,stdin=subprocess.DEVNULL,stdout=log,stderr=subprocess.STDOUT,start_new_session=True)
        try:
            filename='owner-complete.json' if mode=='normal_binding' else 'owner-boundary.json'
            limit=time.monotonic()+15
            while not (scope/filename).exists():
                assert process.poll() is None,'owner exited before boundary'
                assert time.monotonic()<limit,'owner boundary timed out'
                time.sleep(.05)
            boundary=json.loads((scope/filename).read_bytes());assert boundary['owner_pid']==process.pid
            if mode=='normal_binding':
                assert process.wait(timeout=5)==0
            else:
                (scope/'kill-intent.json').write_text(canonical_json({'owner_pid':process.pid,
                    'watchdog_pid':boundary['watchdog_pid'],'signal':'SIGKILL'}))
                os.kill(process.pid,signal.SIGKILL);assert process.wait(timeout=5)==-signal.SIGKILL
            limit=time.monotonic()+15
            while not (scope/'watchdog-outcome.json').exists():
                assert time.monotonic()<limit,'watchdog outcome timed out'
                time.sleep(.05)
            outcome=json.loads((scope/'watchdog-outcome.json').read_bytes())
            expected_absent=mode!='intent_only'
            assert outcome['absent'] is expected_absent
            assert outcome['unacknowledged_creation_unverified'] is (mode=='intent_only')
            if mode=='created_unbound':
                assert not (scope/'watchdog-bind.json').exists()
                assert outcome['container_id']==boundary['container_id']
            if boundary['container_id'] is not None:
                assert api.inspect(boundary['container_id'],deadline=time.monotonic()+5) is None
            else:
                status,body=api._request('GET','/containers/'+name+'/json',deadline=time.monotonic()+5)
                assert status==404
            limit=time.monotonic()+5
            while True:
                ps=subprocess.run(['/bin/ps','-p',str(boundary['watchdog_pid']),'-o','pid=,stat='],capture_output=True,text=True)
                if not ps.stdout.strip():break
                assert time.monotonic()<limit,'watchdog process remained'
                time.sleep(.05)
            (scope/'watchdog-process-absence.json').write_text(canonical_json({'watchdog_pid':boundary['watchdog_pid'],
                'ps_exit_code':ps.returncode,'ps_stdout':ps.stdout,'absent':True}))
            case={'mode':mode,'container_creations':boundary['creates_observed'],'container_starts':0,
                'container_id':boundary['container_id'],'owner_pid':process.pid,'watchdog_pid':boundary['watchdog_pid'],
                'watchdog_reported_absent':outcome['absent'],'external_current_absence_observed':True,
                'watchdog_process_absent':True,'owner_sigkill_observed':mode!='normal_binding',
                'unacknowledged_creation_retained_unverified':mode=='intent_only','source_reads':0,'packet_submitted':False}
            cases.append(case);print(canonical_json(case),flush=True)
        finally:
            if process.poll() is None:os.kill(process.pid,signal.SIGKILL);process.wait(timeout=5)
summary={'schema':'utility-prospective-coupled-watchdog-development-v1','cases':cases,
    'container_creations':2,'container_starts':0,'owner_sigkills':3,'watchdog_processes':4,
    'source_reads':0,'packets_submitted':0,'model_results_observed':0,
    'uncertain_intent_correctly_retained':True,'full_interruption_matrix_complete':False,
    'unbounded_delayed_daemon_creation_qualified':False,'during_computation_loss_qualified':False,
    'runtime_qualified':False,'private_boundary_qualified':False}
(p/'summary.json').write_text(json.dumps(summary,indent=2)+'\n')
for mode in ('before_create','created_unbound','intent_only','normal_binding'):shutil.copytree(work/mode,p/mode)
