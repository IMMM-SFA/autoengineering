"""Run the one prepared public model control; exact-owned cleanup, no retry."""
import hashlib,json,math,os,re,time
from pathlib import Path
from autoengineering.campaign.serialization import canonical_json,digest,unique_json
from autoengineering.utility.guest_daemon import GuestDaemonAPI
from examples.utility import guest_image_build as build,guest_resources as policy
root=Path.cwd();p=root/'examples/utility/verification-engineering-linux-load-probe-v1'
work=root/'outputs/engineering-linux-load-probe-v2'
q=unique_json((work/'request.json').read_text())
assert digest(q)=='2390349867da8307ded9d00b10dd58e72649d8c314ee64f1d22497df70af6dd7'
assert not (work/'execution-intent.json').exists()
assert hashlib.sha256((work/'body.py.txt').read_bytes()).hexdigest()==q['body_sha256']
CONFIG_EXPECTED=q['configuration'];IMAGE=q['image_id']
_same=policy._same;_environment=policy._environment
exec((p/'verify-owned.py.txt').read_text())
api=GuestDaemonAPI('/Users/cam/.orbstack/run/docker.sock',api_version=q['api_version'],server_digest=q['server_digest'])
def save(name,value):
 for target in (p,work):(target/name).write_text(json.dumps(value,indent=2)+'\n')
active=time.monotonic()+60
save('execution-intent.json',{'request_digest':digest(q),'creations_reserved':1,'starts_reserved':1,'model_starts_reserved':24,'automatic_retries':0})
version=api._json('GET','/version',deadline=active)
assert digest(version)==q['server_digest']
image=api._json('GET','/images/'+q['image_id']+'/json',deadline=active)
save('image.json',image)
assert (image['Id'],image['Os'],image['Architecture'])==(q['image_id'],'linux','amd64')
assert image['Config']['Labels']['org.autoengineering.environment']==q['image_environment_digest']
cid=None;outcome=None
try:
 save('create-intent.json',{'configuration_digest':digest(q['configuration']),'name':q['container_name']})
 result=api._json('POST','/containers/create?name='+q['container_name'],q['configuration'],expected=201,deadline=active)
 cid=result['Id'];assert re.fullmatch('[0-9a-f]{64}',cid);save('created.json',result)
 before=api.inspect(cid,deadline=active);verify_owned(before,'public',q['nonce'],cid=cid,image_id=q['image_id']);save('before.json',before)
 assert before['State']['Status']=='created' and not before['State']['Running']
 (work/'config').mkdir(mode=0o700);(work/'config/config.json').write_text('{}')
 argv=['/usr/local/bin/docker','--config',str(work/'config'),'--host','unix:///Users/cam/.orbstack/run/docker.sock','start','--attach',cid]
 save('start-intent.json',{'container_id':cid,'starts':1,'argv_digest':digest(argv)})
 env={k:os.environ[k] for k in ('PATH','HOME','LANG','LC_ALL','TMPDIR') if k in os.environ}
 env.update(DOCKER_CONFIG=str(work/'config'),DOCKER_HOST='unix:///Users/cam/.orbstack/run/docker.sock')
 status=build._run(argv,env=env,cwd=work,deadline=active,log=work/'output.log')
 raw=(work/'output.log').read_bytes();(p/'output.log').write_bytes(raw)
 save('client.json',{'exit_code':status,'output_sha256':hashlib.sha256(raw).hexdigest(),'output_bytes':len(raw)})
 terminal=api.inspect(cid,deadline=active);verify_owned(terminal,'public',q['nonce'],cid=cid,image_id=q['image_id']);save('terminal.json',terminal)
 assert status==0 and terminal['State']['Status']=='exited' and terminal['State']['ExitCode']==0 and terminal['State']['OOMKilled'] is False
 assert 0<len(raw)<=q['max_output_bytes']
 observed=unique_json(raw.decode());assert observed['schema']=='utility-public-linux-coupled-probe-v1'
 assert observed['observed']==policy.OBSERVED and observed['model_digest']==q['expected_model_digest'] and observed['openmdao']=='3.41.0'
 assert len(observed['rows'])==24
 comparisons=[]
 for a,b in zip(observed['rows'],q['expected_rows'],strict=True):
  assert a['scenario_id']==b['scenario_id']
  assert all(math.isfinite(v) for k,v in a.items() if k!='scenario_id')
  assert a['converged']==1 and 0<=a['residual_A']<=1e-10 and 0<=a['solver_iterations']<=100 and a['source_power_W']>=0
  row={'scenario_id':a['scenario_id']}
  for key in ('source_power_W','node2_voltage_V'):
   assert math.isclose(a[key],b[key],rel_tol=q['rtol'],abs_tol=q['atol']),(key,a,b)
   row[key+'_abs_difference']=abs(a[key]-b[key])
  comparisons.append(row)
 outcome={'request_digest':digest(q),'image_id':q['image_id'],'container_id':cid,'container_creations':1,'container_starts':1,'coupled_model_starts':24,'model_results_observed':24,'all_model_outputs_within_frozen_tolerance':True,'all_cases_satisfy_strict_voltage':all(r['node2_voltage_V']>=q['minimum_voltage_V'] for r in observed['rows']),'comparisons':comparisons,'runtime_observed':observed,'new_fitting_starts':0,'scoring_starts':0,'private_data':False,'owner_loss_qualified':False,'runtime_qualified':False,'private_boundary_admitted':False}
except BaseException as error:
 save('failure.json',{'request_digest':digest(q),'error_type':type(error).__name__,'container_id':cid,'attempt_consumed':True,'automatic_retry_forbidden':True,'coupled_work_total':None})
 raise
finally:
 if cid is not None:
  deadline=time.monotonic()+10
  current=api.inspect(cid,deadline=deadline)
  if current is not None:
   verify_owned(current,'public',q['nonce'],cid=cid,image_id=q['image_id'])
   save('delete-intent.json',{'container_id':cid,'force':True})
   api._json('DELETE','/containers/'+cid+'?force=1',expected=204,deadline=deadline)
  absent=api.inspect(cid,deadline=deadline) is None
  save('cleanup.json',{'container_id':cid,'absent':absent,'owner_loss_qualified':False})
  assert absent
if outcome is not None:
 outcome['original_container_absent']=True
 save('complete.json',outcome)
 print(canonical_json({'container_starts':1,'coupled_model_starts':24,'all_outputs_within_tolerance':True,'strict_voltage_all_pass':outcome['all_cases_satisfy_strict_voltage'],'original_container_absent':True}))
