"""Authored learned routing from extracted wheel; not independent admission."""
from pathlib import Path
import functools
import hashlib
import json
import os
import sys

root=Path(__file__).resolve().parent
wheel=root/'unpacked'
os.chdir(wheel)
checkout=Path(sys.argv[1]).resolve()
metadata_source=Path(sys.argv[2]).resolve()
sys.path[:0]=[str(wheel),str(checkout/'tests')]
import autoengineering
assert Path(autoengineering.__file__).is_relative_to(wheel)
import pytest
from click.testing import CliRunner
from autoengineering.cli import cli
from autoengineering.utility.command import PreparationInputs
from autoengineering.utility.confirmation import TrainedSelectedState,read_selected_state
from autoengineering.utility.execution import ScriptedCoordinator
from autoengineering.utility.store import SuiteStore
from autoengineering.utility.report import accounting_report
import examples.campaign_contracts.build_fixture as factory
import test_utility_hydrology_learned as learned_fixture
from test_utility_trained_execution import setup

starts=[]
def observe(event,args):
    if event=='subprocess.Popen':
        command=args[1]
        if isinstance(command,(tuple,list)) and 'autoengineering.campaign.worker' in command:
            assert Path(args[2]).resolve()==wheel
            starts.append({'command':'autoengineering.campaign.worker','cwd':str(args[2])})
sys.addaudithook(observe)
with pytest.MonkeyPatch.context() as patch:
    # Only input construction uses test helpers. Bind their documented source-root
    # argument and fixture root to copied metadata, never the checkout's sources.
    patch.setattr(factory,'ROOT',metadata_source)
    patch.setattr(factory.fixture,'__defaults__',(metadata_source,))
    patch.setattr(learned_fixture,'context',functools.partial(learned_fixture.context,source_root=metadata_source))
    draft,plan,source=setup(root/'metadata',patch,routes=('baseline',))
    assert not starts
    assert Path(sys.modules['examples.utility.hydrology_runtime'].__file__).is_relative_to(wheel)
    spec_file=root/'specification.json';spec_file.write_text(draft.spec.to_json())
    plan_file=root/'execution-plan.json';plan_file.write_text(plan.to_json())
    inputs=root/'preparation-inputs.json'
    inputs.write_text(PreparationInputs((),(),plan.lock,plan.source_inventory).to_json())
    runner=CliRunner();calls=[]
    def invoke(arguments):
        result=runner.invoke(cli,['utility',*map(str,arguments)])
        (root/('cli-'+str(len(calls))+'-'+str(arguments[0])+'.log')).write_text(result.output)
        calls.append({'command':str(arguments[0]),'exit_code':result.exit_code})
        if result.exit_code:raise AssertionError((arguments[0],result.output,result.exception))
        return json.loads(result.output)
    invoke(['validate',spec_file])
    prepared=invoke(['prepare',spec_file,'--inputs',inputs,'--source-root',source,'--workdir',root/'suite'])
    assert prepared['status']=='blocked' and prepared['starts_authorized'] is False
    assert not starts
    run=invoke(['run',root/'suite',plan_file,'--source-root',source,'--scripted-development'])
    assert run['finished_cells']==1 and starts
    before=len(starts)
    resumed=invoke(['resume',root/'suite',plan_file,'--source-root',source,'--scripted-development'])
    assert resumed['completed_this_call']==0 and len(starts)==before
    store=SuiteStore(root/'suite')
    manifest=ScriptedCoordinator(store,plan,source).selection_manifest()
    selected=manifest.cells[0]
    assert selected.status=='selected'
    bound=read_selected_state(store.root,selected.selected_state)
    assert isinstance(bound,TrainedSelectedState)
    path=root/'selection.json';path.write_text(manifest.to_json())
    invoke(['freeze',root/'suite',path])
    report=invoke(['report',root/'suite'])
    assert not report['evidence_blockers']
    assert sum(r['quantity'] for r in report['receipts'] if r['resource']=='training_starts')==1
    verified=invoke(['verify',root/'suite'])
    assert verified['status']=='verified_integrity' and verified['complete_cost_evidence'] is False
    assert len(starts)==before
    result={'schema':'utility-extracted-learned-workflow-v1','purpose':'authored exposed development only','preparation':prepared,'cli_calls':calls,'scientific_worker_starts':before,'resume_new_starts':0,'selected_state_schema':bound.schema,'suite_digest':store.spec.digest,'execution_plan_digest':plan.digest,'source_root':str(source),'package_root':str(wheel),'limits':['test helpers construct inputs only','extracted wheel with inherited base dependencies; no installation','preparation correctly blocks independent admission','one baseline route and one authored fit only','no protected scoring, private runtime or independent data qualification']}
    (root/'result.json').write_text(json.dumps(result,indent=2)+'\n')
    (root/'processes.json').write_text(json.dumps(starts,indent=2)+'\n')
    print('Extracted learned workflow completed')
