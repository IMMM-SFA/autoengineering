"""Save compact metadata and verify original or relocated evidence under a guard."""
from pathlib import Path
import hashlib,json,shutil,subprocess,sys
root=Path.cwd();out=root/'examples/utility/verification-guest-image-v6'
stage=Path('/private/tmp/utility-guest-image-v6-evidence');stage.mkdir(mode=0o700)
reader=stage/'cold-reader.py.txt';shutil.copyfile(out/reader.name,reader)
operations=(('build','outputs/guest-image-build-v6','7a242f00663785832e720341bbb1d6add1e80ad1ae7977bc54bfc42a5e9197b7'),
 ('environment','outputs/guest-image-environment-v6','23863a118c578ae45d6babbea59ea9342b4d89c14ebcab3c7e5ef7388101e5cb'),
 ('identity','outputs/guest-identity-v6','d2741dc1bdda3374dcdbbf475d0c3c2e3f9db654913733bc99901e2b5611dda8'))
results=[]
for kind,relative,digest in operations:
 original=root/relative
 # The base verifier also needs its saved 74 MB plugin binary. Keep that binary
 # in original local evidence rather than redistributing it in the source archive.
 if kind=='build':
  scope=original
 else:
  scope=stage/kind;shutil.copytree(original,scope,ignore=shutil.ignore_patterns('config'))
 saved=out/kind;saved.mkdir(mode=0o700)
 for p in original.iterdir():
  if p.is_file():shutil.copyfile(p,saved/(p.name+'.txt' if p.suffix=='.py' else p.name))
 p=subprocess.run([sys.executable,'-I','-B',str(reader),str(scope),str(root),kind,digest],cwd=stage,capture_output=True,text=True)
 (out/(kind+'-cold.stdout.txt')).write_text(p.stdout);(out/(kind+'-cold.stderr.txt')).write_text(p.stderr)
 assert p.returncode==0,(kind,p.stderr)
 result=json.loads(p.stdout);result.update(exit_code=p.returncode,evidence_relocated=kind!='build')
 results.append(result)
# Mutation is confined to the relocated identity diagnostic, never original proof.
path=stage/'identity/cleanup.json';bad=json.loads(path.read_text());bad['absent']=False;path.write_text(json.dumps(bad))
p=subprocess.run([sys.executable,'-I','-B',str(reader),str(stage/'identity'),str(root),'identity',operations[2][2]],cwd=stage,capture_output=True,text=True)
assert p.returncode!=0
(out/'cold-results.json').write_text(json.dumps({'readers':results,'changed_cleanup_refused':True,'tamper_exit_code':p.returncode,
 'base_plugin_binary_not_redistributed':True,'original_evidence_changed':False},indent=2)+'\n')
print(json.dumps({'readers':results,'changed_cleanup_refused':True}))
