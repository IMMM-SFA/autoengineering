"""Reconstruct saved image/identity producer evidence without any live action."""
from pathlib import Path
import hashlib,json,os,socket,subprocess,sys,time
root=Path(sys.argv[1]).resolve();source=Path(sys.argv[2]).resolve();kind,expected=sys.argv[3:5]
sys.path.insert(0,str(source))
from examples.utility import guest_image_build as build,guest_image_environment as env,guest_identity as identity
reader={'build':build,'environment':env,'identity':identity}[kind]
reader.digest({})
inodes={(p.stat().st_dev,p.stat().st_ino) for p in root.rglob('*') if p.is_file()}
def denied(*a,**kw):raise RuntimeError('live or external action denied')
build._binary=build._endpoint=build._run=denied
env.GuestDaemonAPI=identity.GuestDaemonAPI=denied
subprocess.Popen=socket.socket=time.time=time.monotonic=time.perf_counter=denied

def audit(event,args):
 if event=='open':
  path,mode,flags=args
  if flags&(os.O_WRONLY|os.O_RDWR|os.O_CREAT|os.O_TRUNC|os.O_APPEND):raise RuntimeError('write denied')
  if isinstance(path,int):
   st=os.fstat(path)
   if (st.st_dev,st.st_ino) not in inodes:raise RuntimeError('outside descriptor denied')
  elif not Path(os.fsdecode(path)).absolute().is_relative_to(root):raise RuntimeError('external scope denied')
 if event=='import' or event.startswith(('socket.','subprocess.','os.exec','os.fork','os.posix_spawn')):
  raise RuntimeError('new import or execution denied')
sys.addaudithook(audit)
result=reader.read_completion(root,expected_digest=expected)
assert not result['runtime_qualified']
print(json.dumps({'kind':kind,'verified_digest':reader.digest(result),
 'image_id':result.get('image_id',result.get('profile',{}).get('image_id')),
 'new_models':0,'new_private_reads':0,'new_daemon_calls':0,'network':0,'processes':0,
 'runtime_qualified':False}))
