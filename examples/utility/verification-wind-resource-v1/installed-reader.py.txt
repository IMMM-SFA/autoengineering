"""Read a caller-pinned synthetic demo without any model/provider imports."""
import hashlib
import importlib.util
import io
import json
from pathlib import Path
import sys
import zipfile
import numpy as np
from autoengineering.campaign.artifacts import confined_file, read_bytes
from autoengineering.campaign.records import ArtifactRef
from autoengineering.campaign.serialization import unique_json
from examples.utility.wind_resource import WindResource

root=Path(sys.argv[1]); expected=sys.argv[2]
path=confined_file(root,'summary.json')
assert path.stat().st_size<=65536
raw=path.read_bytes(); assert hashlib.sha256(raw).hexdigest()==expected
summary=unique_json(raw.decode())
assert summary['profile']=='wind-resource-demo-v1' and type(summary['turbines']) is int and summary['turbines']==4
resource_path=confined_file(root,'resource.json'); assert resource_path.stat().st_size<=262144
resource=WindResource.from_json(resource_path.read_text()); assert resource.digest==summary['resource_digest']
ref=ArtifactRef.from_dict(summary['prediction']); assert ref.path=='prediction.npz'
raw=read_bytes(root,ref,max_bytes=1048576)
shape=(4,len(resource.directions_deg),len(resource.speeds_m_s))
with zipfile.ZipFile(io.BytesIO(raw)) as archive:
 assert set(archive.namelist())=={'aep_gwh.npy','power_w.npy'} and len(archive.infolist())==2
 assert sum(i.file_size for i in archive.infolist())<=1048576
 for info in archive.infolist():
  assert not info.flag_bits & 1
  with archive.open(info) as stream:
   version=np.lib.format.read_magic(stream)
   assert version in ((1,0),(2,0))
   header=(np.lib.format.read_array_header_1_0 if version==(1,0) else np.lib.format.read_array_header_2_0)
   recorded,fortran,dtype=header(stream,max_header_size=4096)
   assert recorded==shape and not fortran and dtype==np.dtype('float64')
   assert info.file_size-stream.tell()==np.prod(shape)*8
with np.load(io.BytesIO(raw),allow_pickle=False) as data:
 power=data['power_w']; aep=data['aep_gwh']
 assert np.isfinite(power).all() and np.min(power)>=0 and np.max(power)<=2e6*(1+1e-10)
 np.testing.assert_allclose(aep,power*np.array(resource.probability)[None,:,:]*8760/1e9,rtol=1e-12,atol=1e-12)
 np.testing.assert_allclose(aep.sum(),summary['annual_energy_gwh'],rtol=1e-12)
assert all(importlib.util.find_spec(name) is None for name in ('py_wake','topfarm','openmdao','anthropic','torch','botorch'))
print(json.dumps({'status':'verified_synthetic_demo','annual_energy_gwh':summary['annual_energy_gwh'],'shape':shape,'summary_sha256':expected,'optional_models_absent':True,'model_or_provider_calls':0,'original_access_sockets_processes':'denied by startup audit hook'},indent=2))
