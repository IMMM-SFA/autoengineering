"""Replay only original authored owner-loss evidence after relocation."""
from pathlib import Path
import socket
import subprocess
import tarfile
import tempfile
import time
from autoengineering.utility.guest_owner import GuestOwnerJournal
from autoengineering.utility.guest_watchdog import GuestWatchdogPlan, GuestWatchdogOutcome, read_outcome

bundle = Path(__file__).parent
expected = GuestWatchdogOutcome.from_json((bundle / 'expected-outcome.json').read_text())
with tempfile.TemporaryDirectory() as directory:
    destination = Path(directory).resolve()
    with tarfile.open(bundle / 'authored-owner-loss.tar.gz') as archive:
        archive.extractall(destination, filter='data')
    for path in destination.rglob('*'):
        path.chmod(0o700 if path.is_dir() else 0o600)
    owner = GuestOwnerJournal(destination / 'owner')
    root = owner.root / 'control/guest-watchdog'
    plan = GuestWatchdogPlan.from_json((root / 'plan.json').read_text())
    def forbidden(*args, **kwargs):
        raise AssertionError('cold replay cannot execute, connect or sample a clock')
    subprocess.Popen = socket.socket = time.monotonic = forbidden
    assert read_outcome(root, owner, plan) == expected
    print('Relocated authored owner-loss evidence agrees without daemon, processes or new clocks.')
