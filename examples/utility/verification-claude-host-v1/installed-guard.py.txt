import pathlib,sys
original=pathlib.Path('/Users/cam/projects/autoengineering')
allowed={pathlib.Path(p) for p in ('/Users/cam/Library/Caches/rattler/cache/envs/autoengineering-2706574624935427595/envs/default/bin/python3.12',str(pathlib.Path('/bin/ps').resolve()))}
def guard(event,args):
 if event=='open' and isinstance(args[0],(str,bytes)):
  value=args[0].decode() if isinstance(args[0],bytes) else args[0]
  if pathlib.Path(value).absolute().is_relative_to(original):
   raise RuntimeError('original checkout access denied')
 if event=='socket.__new__' and (args[1]!=2 or args[2]&15!=1):
  raise RuntimeError('only IPv4 TCP permitted')
 if event in ('socket.connect','socket.bind'):
  destination=args[1]
  if not isinstance(destination,tuple) or destination[0]!='127.0.0.1':
   raise RuntimeError('only loopback permitted')
 if event=='socket.getaddrinfo' and args[0]!='127.0.0.1':
  raise RuntimeError('external lookup denied')
 if event=='subprocess.Popen':
  target=args[0]
  resolved=pathlib.Path('/bin/ps') if target=='ps' else pathlib.Path(target).resolve()
  if resolved not in allowed: raise RuntimeError('unapproved subprocess')
sys.addaudithook(guard)
