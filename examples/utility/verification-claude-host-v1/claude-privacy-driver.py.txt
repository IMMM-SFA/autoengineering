"""Measure the Claude renderer with existing synthetic syscall controls only."""
from pathlib import Path
import hashlib
import json
import sys

from autoengineering.agents.hosts import native_canaries as canaries
from autoengineering.agents.hosts import claude_policy as policy
from autoengineering.agents.identity import host_source_identity
from autoengineering.campaign.store import durable_write
from autoengineering.campaign.serialization import canonical_json

root = Path(sys.argv[1]).absolute()
source_sha = hashlib.sha256(Path(__file__).read_bytes()).hexdigest()
policy_sha = hashlib.sha256(Path(policy.__file__).read_bytes()).hexdigest()
original_runner_sha = hashlib.sha256(Path(canaries.__file__).read_bytes()).hexdigest()
specs = []

def render(workspace, home, binary, port):
    helper = binary.parent / 'trusted-helper'
    durable_write(helper, b'#!/bin/sh\nexit 0\n')
    helper.chmod(0o700)
    spec = policy.ClaudePolicy(
        policy.VERSION, str(workspace), str(home), str(binary), str(helper),
        str(Path(sys.executable).resolve()), str(Path(sys.prefix).resolve()), (),
        (policy.ClaudeEgress('localhost', port),),
        (str(binary.parent / 'private'), str(Path.cwd())),
    )
    specs.append(spec.to_dict())
    return policy.render_claude_policy(spec)

# These trusted test hooks are explicit execution inputs. No application source
# is changed. The compiled executable substitution remains visible in evidence.
canaries.render_native_policy = render
canaries.boundary_source_identity = lambda: policy_sha
receipt = canaries.probe_native_policy(root)
assert hashlib.sha256(Path(policy.__file__).read_bytes()).hexdigest() == policy_sha
summary = {
    'version': 'claude-renderer-privacy-driver-1.0',
    'source_identity': host_source_identity(),
    'driver_sha256': source_sha,
    'runner_source_sha256': original_runner_sha,
    'renderer_source_sha256': policy_sha,
    'runtime_hooks': ['render_native_policy=ClaudePolicy renderer',
                      'boundary_source_identity=Claude renderer source hash'],
    'policy_specifications': specs,
    'receipt_sha256': hashlib.sha256((root / 'receipt.json').read_bytes()).hexdigest(),
    'status': receipt['status'], 'observed': receipt['observed'],
    'control': receipt['control'], 'unverified': receipt['unverified'],
    'executable_substitution': True, 'host_qualified': False,
    'shared_memory_unlinked': receipt['shared_memory']['unlinked'],
    'owned_environment_carrier_stopped': receipt['environment_probe']['stopped'],
    'scope': 'Synthetic plist, owned environment marker, owned shared memory and local syscall permissions. No real preference data or credentials.',
    'provider_calls': 0, 'scientific_calls': 0,
}
durable_write(root / 'claude-driver.json', canonical_json(summary).encode())
print(json.dumps(summary, indent=2))
raise SystemExit(0 if receipt['status'] == 'passed' else 4)
