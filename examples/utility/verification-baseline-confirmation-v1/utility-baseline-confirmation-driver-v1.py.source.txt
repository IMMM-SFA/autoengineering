from pathlib import Path
import json
import os
import shutil
import subprocess
import sys
from autoengineering.utility.store import SuiteStore

base=Path.cwd()/'examples/utility/verification-baseline-confirmation-v1'
config_path=base/'installed-configuration.json';configuration=json.loads(config_path.read_bytes())
store=SuiteStore(configuration['suite']);initial=store.replay();states={};runs=[]
for mode,expected in (('follower-before',2),('interrupt',2),('normal',2),('recover',0),('replay',0),('follower-after',2)):
    result=subprocess.run([sys.executable,'-I','/private/tmp/utility-baseline-confirmation-wrapper-v1.py',str(config_path),mode],cwd=Path(configuration['suite']).parent,capture_output=True,text=True,env={'PATH':os.defpath,'LANG':'C.UTF-8'},timeout=20)
    (base/('installed-'+mode+'.stdout')).write_text(result.stdout)
    (base/('installed-'+mode+'.stderr')).write_text(result.stderr)
    assert result.returncode==expected,(mode,result.returncode,result.stderr)
    assert not result.stderr and 'PRIVATE-FIXTURE-CANARY' not in result.stdout
    document=json.loads(result.stdout);state=store.replay();states[mode]=state
    if mode=='follower-before':assert state['head']==initial['head']
    elif mode=='interrupt':assert len(state['confirmation_accesses'])==1 and not state['confirmation_scores']
    elif mode=='normal':assert state['head']==states['interrupt']['head']
    else:
        assert state['used']==states['interrupt']['used'] and state['phase_used']==states['interrupt']['phase_used']
        assert state['confirmation_accesses']==states['interrupt']['confirmation_accesses']
    if mode=='replay':assert state['head']==states['recover']['head']
    if mode=='follower-after':assert state['head']==states['replay']['head']
    runs.append({'mode':mode,'exit_code':result.returncode})
record=json.loads(Path(configuration['callback_record']).read_bytes())
assert record['callback_invocations']==record['recoveries']==1
assert len(store.replay()['confirmation_scores'])==1
shutil.copytree(store.root,base/'public-suite')
shutil.copyfile(configuration['callback_record'],base/'callback.json')
(base/'installed-execution.json').write_text(json.dumps({'schema':'utility-baseline-confirmation-installed-execution-v1','runs':runs,'callback_invocations':1,'recoveries':1,'confirmation_accesses':1,'committed_source_scores':1,'follower_reservations':0,'original_charge_retained':True,'fixture_only':True,'private_scoring_qualified':False},indent=2)+'\n')
print('Six installed CLI processes completed; one source callback and one original recovery.')
