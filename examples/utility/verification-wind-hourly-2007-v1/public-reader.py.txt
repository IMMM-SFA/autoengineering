"""Replay saved scalar parity, input identity and charges without running models.

Does not recompute physical power or qualify raw HDF5 directional conventions.
"""
import sys
from pathlib import Path
sys.path.insert(0, str(Path(sys.argv[2]).resolve()))
import importlib.abc
class NoModels(importlib.abc.MetaPathFinder):
    def find_spec(self, fullname, *args):
        if fullname.split('.')[0] in {'h5py','torch','pvlib','openmdao','py_wake','topfarm'}:
            raise RuntimeError('optional model runtime denied')
sys.meta_path.insert(0, NoModels())
import hashlib, json, math, os, socket, subprocess, time
import numpy as np
from examples.utility.wind_hourly import WindHourly, POLICY, read_point
from examples.utility.wind_hourly_check import ASSUMPTIONS
from examples.utility.wind_parameters import WindGeometry
from autoengineering.campaign.records import ArtifactRef
from autoengineering.campaign.serialization import digest, unique_json
root=Path(sys.argv[1]).resolve()
expected=json.loads((root.parent/'expected-public.json').read_text())
inodes={(p.stat().st_dev,p.stat().st_ino) for p in root.rglob('*') if p.is_file()}
def denied(*args,**kwargs):raise RuntimeError('process, network or clock denied')
subprocess.Popen=socket.socket=time.time=time.monotonic=denied

def audit(event,args):
    if event=='open':
        path,mode,flags=args
        if flags&(os.O_WRONLY|os.O_RDWR|os.O_CREAT|os.O_TRUNC|os.O_APPEND):
            raise RuntimeError('write denied')
        if isinstance(path,int):
            st=os.fstat(path)
            if (st.st_dev,st.st_ino) not in inodes:raise RuntimeError('outside descriptor denied')
        elif not Path(os.fsdecode(path)).absolute().is_relative_to(root):
            raise RuntimeError('outside read denied')
    if event=='import' or event.startswith(('socket.','subprocess.','os.exec','os.fork','os.posix_spawn')):
        raise RuntimeError('new import or execution denied')
sys.addaudithook(audit)
for name,sha in expected['files'].items():
    assert hashlib.sha256((root/name).read_bytes()).hexdigest()==sha
load=lambda n:unique_json((root/n).read_text())
q=load('request.json');c=load('complete.json')
assert digest(q)==expected['request_digest']==c['request_digest']
assert digest(c)==expected['completion_digest']
assert q['policy']==POLICY and q['assumptions']==ASSUMPTIONS==load('assumptions.json')
assert load('consumed.json')['request_digest']==digest(q)
a=read_point(root,ArtifactRef.from_dict(q['point_ref']),2007)
profile=WindHourly(WindGeometry((0.,600.,0.,600.),(0.,0.,600.,600.)),2007,
                   tuple(a['time_utc']),a['speed_m_s'],a['direction_source_deg'],
                   q['point_ref']['sha256'],digest(ASSUMPTIONS))
assert profile.identity==q['profile_identity']==c['profile_identity']
assert (q['wake_calls_cap'],q['curve_reference_calls_cap'],q['automatic_retries'])==(2,1,0)
for i,label in enumerate(('direct','wrapped','curve-reference')):
    intent=load(f'{i}-intent.json');receipt=load(f'{i}-receipt.json')
    assert intent['request_digest']==digest(q) and intent['stage']==label
    assert intent['wake_calls_reserved']==min(i+1,2)
    assert intent['curve_reference_calls_reserved']==int(i==2)
    assert receipt==c['receipts'][i] and receipt['stage']==label and receipt['status']=='complete'
    assert math.isfinite(receipt['elapsed_seconds']) and receipt['elapsed_seconds']>=0
    v=receipt['outputs']
    assert set(v)=={'annual_energy','minimum_spacing','boundary_violation','maximum_power','minimum_power'}
    assert all(math.isfinite(x) for x in v.values())
    assert 0<=v['annual_energy']<=4*2e6*8760/1e9*(1+1e-10)
    assert 0<=v['minimum_power']<=v['maximum_power']<=2e6*(1+1e-10)
    assert v['minimum_spacing']==600. and v['boundary_violation']==0.
for name,v in c['receipts'][0]['outputs'].items():
    w=c['receipts'][1]['outputs'][name]
    assert abs(v-w)<=q['atol']+q['rtol']*abs(w)
assert c['wake_calls_reserved']==2 and c['curve_reference_calls_reserved']==1
assert c['direct_wrapped_parity'] and c['status']=='complete'
assert c['elapsed_seconds']<q['elapsed_application_cap_seconds']
assert all(c[k] is False for k in ('task_admitted','independent_confirmation','os_deadline_qualified',
                                  'aggregate_rss_qualified','raw_direction_convention_qualified'))
print(json.dumps({'saved_scalar_parity':True,'profile_and_point_identity':True,'hours':8760,
                  'recorded_wake_calls':2,'recorded_curve_reference_calls':1,'new_model_calls':0,
                  'physical_power_recomputed':False,'new_http_requests':0,'new_processes':0,
                  'optional_runtimes_imported':False,'task_admitted':False}))
