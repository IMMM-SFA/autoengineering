from pathlib import Path
import hashlib
import json
import os
import subprocess
import sys
from autoengineering.utility.store import SuiteStore

base=Path.cwd()/'examples/utility/verification-task-scoring-binding-v1'
config_path=base/'installed-configuration.json';configuration=json.loads(config_path.read_bytes())
store=SuiteStore(configuration['suite']);runs=[];observed={}
for mode,expected in (('interrupt',2),('normal',2),('recover',0),('replay',0)):
    result=subprocess.run([sys.executable,'-I','/private/tmp/utility-confirmation-cli-wrapper-v1.py',str(config_path),mode],cwd=Path(configuration['suite']).parent,capture_output=True,text=True,env={'PATH':os.defpath,'LANG':'C.UTF-8'},timeout=20)
    (base/('installed-'+mode+'.stdout')).write_text(result.stdout)
    (base/('installed-'+mode+'.stderr')).write_text(result.stderr)
    assert result.returncode==expected,(mode,result.returncode,result.stderr)
    assert result.stderr=='' and 'PRIVATE-FIXTURE-CANARY' not in result.stdout
    document=json.loads(result.stdout);state=store.replay();observed[mode]=state
    if mode=='interrupt':
        assert len(state['confirmation_accesses'])==1 and not state['confirmation_scores']
    elif mode=='normal':
        assert state['head']==observed['interrupt']['head']
    else:
        assert state['used']==observed['interrupt']['used']
        assert state['phase_used']==observed['interrupt']['phase_used']
        assert state['confirmation_accesses']==observed['interrupt']['confirmation_accesses']
        assert document['receipt']['score']['value']==3.0
    runs.append({'mode':mode,'exit_code':result.returncode,'stdout_sha256':hashlib.sha256(result.stdout.encode()).hexdigest()})
assert observed['replay']['head']==observed['recover']['head']
record=json.loads(Path(configuration['callback_record']).read_bytes())
assert record['callback_invocations']==record['recoveries']==1
report={'schema':'utility-confirmation-cli-installed-v1','runs':runs,'matched_core_files':201,'callback_invocations':1,'original_recoveries':1,'retry_refused':True,'original_access_and_charge_retained':True,'committed_replay_no_callbacks':True,'wrapper_network_and_process_denied':True,'optional_frameworks_not_imported':True,'private_scoring_qualified':False,'fixture_only':True,'real_confirmation_reads':0,'provider_calls':0,'containers_created':0}
(base/'installed-check.json').write_text(json.dumps(report,indent=2)+'\n')
print(json.dumps(report))
