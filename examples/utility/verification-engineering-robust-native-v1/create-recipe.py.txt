def controlling_scenario(scenarios, minimum_voltage):
    """Return the threshold-controlling supplied case without running a model.

    Kirchhoff and Ohm laws give I = v/R1 + (1 + R2/R1) Is expm1(v/Vt),
    where v = minimum_voltage - ground and the fixed diode has Is=1e-15 A,
    Vt=0.025875 V. An unrepresentable threshold refuses before model construction.
    """
    cases = tuple(islice(scenarios, 25))
    if not cases or len(cases) > 24 or len({s.scenario_id for s in cases}) != len(cases):
        raise ValueError("robust native load recipe requires 1 to 24 unique supplied cases")
    if type(minimum_voltage) not in (int, float) or not math.isfinite(minimum_voltage):
        raise ValueError("robust native load voltage requirement must be finite")
    requirements = []
    for case in cases:
        dominant_scenario((case,))  # Reuse strict finite, positive-resistance validation.
        ground, r1, r2 = (case.forcing[n] for n in
                         ("ground_voltage", "resistor1_ohm", "resistor2_ohm"))
        v = minimum_voltage - ground
        try:
            current = v / r1 + (1 + r2 / r1) * 1e-15 * math.expm1(v / .025875)
        except OverflowError as error:
            raise ValueError("robust native threshold current is not representable") from error
        if not math.isfinite(current):
            raise ValueError("robust native threshold current is not representable")
        requirements.append(current)
    return cases[max(range(len(cases)), key=requirements.__getitem__)]
