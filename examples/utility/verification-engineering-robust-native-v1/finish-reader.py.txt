from pathlib import Path
import hashlib,json,shutil,subprocess,sys
root=Path.cwd();p=root/'examples/utility/verification-engineering-robust-native-v1'
command=[sys.executable,'-I','-B',str(p/'cold-reader-v2.py.txt'),'/private/tmp/utility-robust-native-v1-public','/private/tmp/utility-load-cost-report-v1/installed']
r=subprocess.run(command,cwd='/private/tmp',text=True,capture_output=True)
(p/'cold-v2.stdout.txt').write_text(r.stdout);(p/'cold-v2.stderr.txt').write_text(r.stderr)
assert r.returncode==0,r.stderr
altered=Path('/private/tmp/utility-robust-native-v1-forged-aggregate');shutil.copytree(p/'public-evidence',altered)
f=altered/'scientific-evaluation.json';v=json.loads(f.read_text())
v['scenarios'][0]['values']['source_power']+=1
v['evaluation']['metrics']['source_power']=sum(x['values']['source_power'] for x in v['scenarios'])/24
f.write_text(json.dumps(v))
command[4]=str(altered)
bad=subprocess.run(command,cwd='/private/tmp',text=True,capture_output=True)
(p/'forged-aggregate.stderr.txt').write_text(bad.stderr)
assert bad.returncode!=0 and 'AssertionError' in bad.stderr,bad.stderr
s=json.loads((p/'summary.json').read_text())
s.update(saved_scenario_scores_bound_to_parent_receipts=True,altered_aggregate_rejected_without_model_execution=True,installed_cold_reader_v2_exit_code=0,optional_tests_passed=22,base_tests_passed=33,base_optional_skips=8,documentation_tests_passed=19)
(p/'summary.json').write_text(json.dumps(s,indent=2)+'\n')
with (p/'README.md').open('a') as f:f.write('\nThe strengthened reader also matches every saved scenario score to its original\nverified parent receipt, job, configuration and action. A separate copy with\nconsistently forged scenario and aggregate metrics was rejected without executing\nmodels. The initial reader and the strengthened reader are both preserved.\n')
shutil.copyfile('/private/tmp/finish-robust-native.py',p/'finish-reader.py.txt')
print('Verified aggregate-to-parent binding and rejected a forged aggregate.')
