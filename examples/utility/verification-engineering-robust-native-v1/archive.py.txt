from pathlib import Path
root=Path.cwd();os.umask(0o077)
p=root/'examples/utility/verification-engineering-robust-native-v1'
folders=[f.parent for f in Path('/private/tmp/utility-robust-load-v1-final').glob('*/native-report.json') if not f.parent.is_symlink()]
assert len(folders)==1
original=folders[0];data=p/'public-evidence';data.mkdir(mode=0o700)
shutil.copytree(original/'batch',data/'batch')
for name in ('native-spec.json','native-report.json','scientific-evaluation.json'):shutil.copyfile(original/name,data/name)
for d in data.rglob('*'):
 if d.is_dir():d.chmod(0o700)

with zipfile.ZipFile(wheel) as z:
 names=[n for n in z.namelist() if n.endswith('.py')]
 assert len(names)==225 and all(z.read(n)==(installed/n).read_bytes()==(root/'src'/n).read_bytes() for n in names)
