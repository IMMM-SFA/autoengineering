"""Inspect development-object HDF5 metadata only through a bounded conditional reader."""
import sys
from pathlib import Path
sys.path.insert(0,str(Path(sys.argv[1]).resolve()))
import hashlib,json
from datetime import datetime,timezone
import h5py,numpy
from examples.utility.wind_range import WindObjectPin,RangeLimits,WindRangeFile
root=Path(sys.argv[2]).resolve()
head=json.loads((root/'head-result.json').read_text())
pin=WindObjectPin('utility-wind-development-object-v1',2007,int(head['headers']['content-length']),head['headers']['etag'])
limits=RangeLimits()
source=Path(sys.argv[1])/'examples/utility/wind_range.py'
request={'schema':'utility-wind-source-metadata-probe-v1','pin':pin.to_dict(),'limits':limits.to_dict(),'reader_sha256':hashlib.sha256(source.read_bytes()).hexdigest(),'probe_sha256':hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),'dataset_names':['windspeed_80m','winddirection_80m','coordinates','meta','time_index'],'dataset_values_allowed':False,'reserved_objects_allowed':False,'deadline_mode':'application checks plus socket-operation timeout; not OS-enforced','environment':{'python':sys.version.split()[0],'h5py':h5py.__version__,'hdf5':h5py.version.hdf5_version,'numpy':numpy.__version__}}
with (root/'metadata-request.json').open('x') as file:json.dump(request,file,indent=2)
(root/'ranges').mkdir(mode=0o700)

def record(event,raw):
    name=f"{event['request']:03d}-{event['status']}"
    if raw is not None:
        binary=f"{event['request']:03d}-{event['lo']:016x}.bin"
        (root/'ranges'/binary).write_bytes(raw)
        event['body_file']=binary
    with (root/'ranges'/(name+'.json')).open('x') as file:json.dump(event,file,indent=2)
    if event['request']%16==0 and raw is not None:
        print(json.dumps({'range_requests':event['request'],'last_range_bytes':len(raw)}),flush=True)

def plain(value):
    if isinstance(value,bytes):return value.decode('utf-8')
    if isinstance(value,numpy.generic):return plain(value.item())
    if isinstance(value,numpy.ndarray):
        if value.size>16:raise ValueError('metadata attribute exceeds reviewed size')
        return plain(value.tolist())
    if isinstance(value,(list,tuple)):return [plain(x) for x in value]
    if value is None or isinstance(value,(str,int,float,bool)):
        if isinstance(value,str) and len(value)>4096:raise ValueError('metadata text exceeds bound')
        return value
    raise ValueError('unreviewed metadata attribute type')

reader=WindRangeFile(pin,limits,record=record)
try:
    with h5py.File(reader,'r') as file:
        names=list(file.keys())
        if len(names)>256:raise ValueError('root link count exceeds reviewed scope')
        datasets={}
        for name in request['dataset_names']:
            if name not in file:
                datasets[name]={'present':False};continue
            if not isinstance(file.get(name,getlink=True),h5py.HardLink):
                raise ValueError('external or soft dataset links refused')
            item=file[name]
            if not isinstance(item,h5py.Dataset):raise ValueError('expected fixed dataset')
            attrs=list(item.attrs)
            if len(attrs)>64:raise ValueError('attribute count exceeds reviewed scope')
            kept={key:plain(item.attrs[key]) for key in attrs if key in {'units','scale_factor','add_offset','_FillValue','missing_value'}}
            datasets[name]={'present':True,'shape':list(item.shape),'dtype':str(item.dtype),'chunks':list(item.chunks) if item.chunks else None,'compression':item.compression,'attribute_names':attrs,'attributes':kept}
    result={'status':'complete','pin':pin.to_dict(),'root_dataset_names':names,'datasets':datasets,'stats':reader.stats(),'dataset_values_indexed':False,'metadata_full_source_verified':False,'reserved_objects_accessed':False,'source_qualified':False,'acquired_utc':datetime.now(timezone.utc).isoformat()}
except Exception as error:
    result={'status':'failed','exception_type':type(error).__name__,'message':str(error)[:200],'stats':reader.stats(),'reserved_objects_accessed':False,'source_qualified':False}
    (root/'metadata-result.json').write_text(json.dumps(result,indent=2)+'\n')
    raise
finally:
    reader.close()
(root/'metadata-result.json').write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps({'status':result['status'],'range_requests':result['stats']['requests'],'received_bytes':result['stats']['received_response_bytes'],'datasets':{name:details.get('shape') for name,details in result['datasets'].items()},'dataset_values_indexed':False}),flush=True)
