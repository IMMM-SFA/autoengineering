from pathlib import Path
import hashlib
import json
import os
import signal
import subprocess
import sys

from autoengineering.utility import private_container_runtime as module

repository=Path.cwd()
evidence=repository/'examples/utility/verification-container-creation-owner-v1'
evidence.mkdir(mode=0o700)
source_names=['src/autoengineering/utility/private_container_runtime.py',
              'src/autoengineering/utility/private_container_worker.py',
              'src/autoengineering/utility/private_score_worker.py']
hashes={name:hashlib.sha256((repository/name).read_bytes()).hexdigest() for name in source_names}
(evidence/'executed-source-hashes.json').write_text(json.dumps(hashes,indent=2)+'\n')
endpoint=Path('/Users/cam/.orbstack/run/docker.sock')
version=module.API(str(endpoint)).json('GET','/version')
server=module.sha(module.encode(version))
registered=module.FixedScorerRuntime(endpoint,server_sha256=server)
api_version=registered.preflight()
original_request=module.API.request
original_read=module.read_regular
controls=[]
for phase in ('before-ack','before-pipe'):
    for kind,signum in (('cancel',signal.SIGTERM),('owner-loss',signal.SIGKILL)):
        name=phase+'-'+kind
        print('Running creation owner control: '+name,flush=True)
        root=Path('/private/tmp')/('utility-creation-owner-'+name+'-v1')
        root.mkdir(mode=0o700)
        work=root/'workspace';work.mkdir(mode=0o700)
        (work/'job.json').write_text('{"outcome":"rmse","units":"1","prediction":[0,1]}')
        (work/'labels.json').write_text('{"labels":[1,2],"mask":[true,true]}')
        settings={'endpoint':str(endpoint),'server_sha256':server,'runtime_identity':registered.identity,
                  'api_version':api_version,'job_sha256':module.sha((work/'job.json').read_bytes()),
                  'labels_sha256':module.sha((work/'labels.json').read_bytes()),
                  'phase':phase,'signal':int(signum)}
        module.write(root/'control.json',settings)
        with (root/'creator.stdout').open('xb') as out,(root/'creator.stderr').open('xb') as err:
            child=subprocess.Popen([sys.executable,'-I','/private/tmp/utility-creation-owner-child-v1.py',
                                    str(repository/'src'),str(root)],stdout=out,stderr=err,
                                   env={'PATH':'/usr/bin:/bin','LANG':'C.UTF-8'})
            code=child.wait(timeout=30)
        assert code==-int(signum),(name,code)
        pending=json.loads((root/'container/creation-pending.json').read_bytes())
        cid=json.loads((root/'transport-created.json').read_bytes())['container_id']
        termination=json.loads((root/'owner-termination-intent.json').read_bytes())
        assert termination['pid']==child.pid
        assert (root/'container/plan.json').exists()==(phase=='before-pipe')
        assert not (root/'container/start-intent.json').exists()
        assert not (root/'container/watchdog-ready.json').exists()
        requests=[json.loads(line) for line in (root/'creator-requests.jsonl').read_bytes().splitlines()]
        assert len(requests)==1 and requests[0]['method']=='POST'
        api=module.API(str(endpoint),api_version)
        before=api.inspect(cid)
        module.verify_owned(before,{**pending,'container_id':cid})
        assert before['State']['Status']=='created' and before['State']['Running'] is False
        assert before['State']['Pid']==0
        module.write(root/'external-before-cleanup.json',before)
        stage=root/'container/stage';stage.chmod(0o755)
        for scope in (work,stage):
            for field in ('job.json','labels.json'): (scope/field).unlink()
        stage.chmod(0o555)
        observer=module.FixedScorerRuntime(endpoint,server_sha256=server)
        observer_ops=[]
        def request(api,method,path,value=None,**kwargs):
            observer_ops.append({'method':method,'path':path})
            if method=='POST': raise AssertionError('reconciliation attempted create/start')
            return original_request(api,method,path,value,**kwargs)
        def metadata(path,limit):
            if path.name in ('job.json','labels.json'):
                raise AssertionError('reconciliation attempted job/label read')
            return original_read(path,limit)
        module.API.request=request
        module.read_regular=metadata
        receipt=observer.reconcile_creation(root,stop=True)
        assert receipt['container_id']==cid and receipt['removed'] is True
        assert observer.reconcile_creation(root,stop=True)['removed'] is True
        assert api.inspect(cid) is None
        assert sum(op['method']=='DELETE' for op in observer_ops)==1
        module.API.request=original_request
        module.read_regular=original_read
        record={'name':name,'root':str(root),'owner_pid':child.pid,'owner_exit':code,
                'phase':phase,'container_id':cid,'nonce':pending['nonce'],'created_objects':1,
                'container_starts':0,'scoring_starts':0,'watchdog_launched':False,
                'metadata_only_reconciliation':True,'all_job_label_copies_deleted':True,
                'original_id_removed':True,'live_absence_confirmed':True,
                'observer_operations':observer_ops}
        module.write(root/'control-result.json',record)
        controls.append(record)
        for path,digest in hashes.items():
            assert hashlib.sha256((repository/path).read_bytes()).hexdigest()==digest
        print(json.dumps({k:v for k,v in record.items() if k!='observer_operations'},sort_keys=True),flush=True)
(evidence/'actual-controls.json').write_text(json.dumps({
    'schema':'synthetic-container-creation-owner-controls-v1','controls':controls,
    'created_objects':4,'container_starts':0,'scoring_starts':0,'private_scoring_qualified':False,
    'real_confirmation_data_reads':0,'provider_calls':0,'image_pulls':0,'remote_jobs':0,
},indent=2)+'\n')
