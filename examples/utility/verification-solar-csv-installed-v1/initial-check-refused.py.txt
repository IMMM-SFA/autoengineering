"""First installed preparation, then public and trusted reserved reading, without science."""
import sys
from pathlib import Path
sys.path.insert(0,sys.argv[1])
site=Path(sys.argv[1]).resolve()
output=Path(sys.argv[2]).absolute()
import importlib.abc
class NoModels(importlib.abc.MetaPathFinder):
    def find_spec(self,fullname,*args):
        if fullname.split('.')[0] in {'torch','pvlib','py_wake','topfarm','openmdao','h5py','bmipy'}:
            raise RuntimeError('optional models denied')
sys.meta_path.insert(0,NoModels())
import json,os,socket,subprocess,time
from examples.utility import solar_csv_demo as demo
from examples.utility.solar_csv import SolarCsvPlan,parse_csv
from examples.utility.solar_metadata import SolarMetadataReview
from examples.utility.solar_inputs import read_development,read_reserved
from autoengineering.campaign.serialization import digest,unique_json
from importlib.resources import files
# Only metadata and resource discovery are initialized. The source asset remains unopened.
metadata=demo.resource(demo.PLAN_RESOURCE)
plan=SolarCsvPlan.from_dict(metadata['csv_plan'])
review=SolarMetadataReview.from_dict(metadata['metadata_review'])
asset=Path(str(files('examples.utility').joinpath(demo.SOURCE_RESOURCE)))
assert asset.is_relative_to(site) and asset.is_file()
for name,module in tuple(sys.modules.items()):
    if name=='autoengineering' or name.startswith('autoengineering.') or name=='examples.utility' or name.startswith('examples.utility.'):
        origin=getattr(module,'__file__',None)
        if origin is not None:assert Path(origin).resolve().is_relative_to(site),name
phase={'value':'prepare'}
source_reads=[]
def denied(*args,**kwargs):raise RuntimeError('models process network or clock denied')
subprocess.Popen=socket.socket=time.monotonic=time.time=denied

def audit(event,args):
    if event=='open':
        path,mode,flags=args
        if isinstance(path,int):raise RuntimeError('unexpected descriptor open')
        path=Path(os.fsdecode(path)).absolute()
        writing=flags&(os.O_WRONLY|os.O_RDWR|os.O_CREAT|os.O_TRUNC|os.O_APPEND)
        if phase['value']=='prepare':
            if writing and not path.is_relative_to(output):raise RuntimeError('outside preparation write')
            if not writing and not (path.is_relative_to(output) or path.is_relative_to(site)):
                raise RuntimeError('outside preparation read')
            if path==asset:
                assert (output/'public/preparation-claim.json').is_file()
                assert (output/'reserved/preparation-claim.json').is_file()
                source_reads.append(1)
        else:
            if writing:raise RuntimeError('reader write denied')
            allowed=output/'public' if phase['value']=='public' else output
            if not path.is_relative_to(allowed):raise RuntimeError('reader outside scope')
    if event=='import' or event.startswith(('socket.','subprocess.','os.exec','os.fork','os.posix_spawn')):
        raise RuntimeError('new import or execution denied')
sys.addaudithook(audit)
manifest=demo.prepare_demo(output)
assert source_reads==[1]
try:parse_csv(plan,b'changed source bytes',review)
except ValueError as error:assert str(error)=='solar CSV normalization failed'
else:raise AssertionError('changed source accepted')
try:demo.prepare_demo(output)
except ValueError:pass
else:raise AssertionError('consumed preparation reopened')
assert source_reads==[1]
kwargs=dict(expected_manifest_digest=digest(manifest),expected_preparer_sources=manifest['preparer_sources'])
phase['value']='public'
public=read_development(output/'public',**kwargs)
assert public.qualification=='not_admitted' and len(public.forcing.start)==12
assert sum(public.labels['fitting'].score_mask)==sum(public.labels['selection'].score_mask)==4
again=read_development(output/'public',**kwargs)
assert again==public
phase['value']='reserved'
reserved=read_reserved(output/'public',output/'reserved',**kwargs)
assert reserved.qualification=='not_admitted' and len(reserved.forcing.start)==16
assert sum(reserved.labels['confirmation'].score_mask)==4
assert reserved.forcing.start[:12]==public.forcing.start
print(json.dumps({'installed_target_used':True,'python_files_in_wheel':231,
                  'source_asset_reads':1,'source_read_after_both_claims':True,
                  'development_intervals':12,'confirmation_intervals':4,
                  'changed_source_refused':True,'consumed_preparation_refused':True,
                  'public_reader_cannot_open_reserved':True,'reader_replay_equal':True,
                  'editable_source_fallback':False,'new_model_calls':0,
                  'new_processes_after_initialization':0,'new_http_requests':0,'task_admitted':False}))
