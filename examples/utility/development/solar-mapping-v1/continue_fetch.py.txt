import hashlib,json,time,urllib.request
from pathlib import Path
root=Path(__file__).parent
base='https://oedi-data-lake.s3.amazonaws.com/pvdaq/2023-solar-data-prize/2107_OEDI/data/'
records=[]
for filename,limit in [('2107_electrical_data_v1.csv',64*1024**2),('2107_environment_data.csv',6*1024**2),('2107_irradiance_data.csv',8*1024**2)]:
 p=root/filename; offset=p.stat().st_size if p.exists() else 0
 overlap=min(offset,256); lo=offset-overlap;start=time.monotonic();n=0;rows=0;last=None;stopped=False
 print('Fetching',filename,'from byte',lo,flush=True)
 request=urllib.request.Request(base+filename,headers={'Range':f'bytes={lo}-{lo+limit-1}'})
 with urllib.request.urlopen(request,timeout=30) as r:
  assert r.status==206
  if overlap:
   prefix=r.read(overlap);n+=len(prefix)
   with p.open('rb') as old:old.seek(lo);assert old.read(overlap)==prefix
  with p.open('ab' if offset else 'xb') as f:
   if not offset:
    header=r.readline(65536);n+=len(header);f.write(header);print('Columns:',header.decode().strip()[:400],flush=True)
   while True:
    line=r.readline(65536);n+=len(line)
    if not line:break
    if n>limit or time.monotonic()-start>240:raise RuntimeError('acquisition cap')
    if not line.endswith(b'\n'):raise RuntimeError('range too short')
    stamp=line.split(b',',1)[0].strip(b'"')
    if stamp[:10]>=b'2019-07-01':stopped=True;break
    if not stamp[:4].isdigit():raise RuntimeError('date format')
    f.write(line);rows+=1;last=stamp.decode()
  records.append({'url':base+filename,'range_start':lo,'status':r.status,'etag':r.headers.get('ETag'),'bytes_read_this_request':n,'rows_added':rows,'last_timestamp':last,'cutoff_reached':stopped,'seconds':time.monotonic()-start,'sha256':hashlib.sha256(p.read_bytes()).hexdigest(),'file_bytes':p.stat().st_size})
 (root/'acquisition.json').write_text(json.dumps(records,indent=2))
 print('Saved through',last,flush=True)
