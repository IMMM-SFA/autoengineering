import importlib.util,json
from pathlib import Path
import numpy as np,pandas as pd
from scipy.optimize import minimize_scalar
root=Path(__file__).parent
spec=importlib.util.spec_from_file_location('model',Path('examples/utility/solar_1278_development.py'))
m=importlib.util.module_from_spec(spec);spec.loader.exec_module(m)
files=[root/'1278_development_sample.csv',*[root/f'1278_day_{i:02d}.csv' for i in range(2,15)]]
d=pd.concat([pd.read_csv(p) for p in files],ignore_index=True);t=pd.to_datetime(d.measured_on)
g=d.poa_irradiance__3056;a=(d.ambient_temp_f__3058-32)*5/9;w=d.wind_speed_mph__3060*.44704;y=d.ac_power__3081/1000
valid=g.between(100,1400)&a.between(-20,60)&w.between(0,40)&y.between(-.5,157.5)&~t.duplicated(keep=False)
fit=valid&(t.dt.day<=7);check=valid&(t.dt.day>7)
# Deliberate development amendment: first fit hit its lower bound.
# Broaden only effective DC derating; never infer changed module identity.
opt=minimize_scalar(lambda s:float(np.mean((m.predict(g[fit],a[fit],w[fit],s)-y[fit])**2)),bounds=(.5,1.05),method='bounded',options={'maxiter':24,'xatol':1e-5})
assert opt.success
p=m.predict(g,a,w,opt.x)
metrics={}
for phase,mask in [('fit',fit),('check',check)]:
 yy=y[mask];pp=p[mask];rmse=float(np.sqrt(np.mean((pp-yy)**2)));bias=float(pp.sum()/yy.sum()-1);corr=float(yy.corr(pp))
 metrics[phase]={'rows':int(mask.sum()),'rmse_kw':rmse,'rmse_fraction_ac_rating':rmse/150,'daylight_energy_bias':bias,'correlation':corr,'screen_pass':mask.sum()>=200 and rmse/150<=.2 and abs(bias)<=.1 and corr>=.85}
out={'reason':'initial fit saturated 0.8 lower bound; development amendment extends effective derating domain to [0.5,1.05]; unchanged screen thresholds; same development check reused and not independent confirmation','scale':float(opt.x),'optimizer_evaluations':opt.nfev,'optimizer_success':bool(opt.success),'metrics':metrics,'scope':'effective calibrated model, not recovered physical equipment or loss mechanism','confirmation_accessed':False}
# convert numpy bool without coercing numeric data
for v in metrics.values():v['screen_pass']=bool(v['screen_pass'])
(root/'alternative_refined.json').write_text(json.dumps(out,indent=2)+'\n');print(json.dumps(out,indent=2))
