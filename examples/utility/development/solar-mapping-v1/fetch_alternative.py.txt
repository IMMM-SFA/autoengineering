import urllib.request,json,hashlib
from pathlib import Path
root=Path(__file__).parent
receipts=[]
for day in range(2,15):
 url=f'https://oedi-data-lake.s3.amazonaws.com/pvdaq/csv/pvdata/system_id=1278/year=2011/month=10/day={day}/system_1278__date_2011_10_{day:02d}.csv'
 with urllib.request.urlopen(url,timeout=30) as r:b=r.read(1000001)
 assert len(b)<=1000000
 (root/f'1278_day_{day:02d}.csv').write_bytes(b)
 receipts.append({'url':url,'bytes':len(b),'sha256':hashlib.sha256(b).hexdigest()})
 (root/'alternative_acquisition.json').write_text(json.dumps(receipts,indent=2))
 print('Saved development day',day,flush=True)
