"""Read only the sealed authored development bundle after guarded initialization."""
import sys
from pathlib import Path
sys.path.insert(0,str(Path(sys.argv[2]).resolve()))
import importlib.abc
class NoModels(importlib.abc.MetaPathFinder):
    def find_spec(self,fullname,*args):
        if fullname.split('.')[0] in {'h5py','py_wake','pvlib','openmdao','torch','topfarm'}:
            raise RuntimeError('optional model imports denied')
sys.meta_path.insert(0,NoModels())
import hashlib,json,os,socket,subprocess,time
from autoengineering.campaign.serialization import digest,unique_json
from examples.utility.solar_csv import SolarCsvPlan
from examples.utility.solar_inputs import read_development
root=Path(sys.argv[1]).resolve()
expected=json.loads((root.parent/'expected-public.json').read_text())
plan=SolarCsvPlan.from_json((root.parent/'csv-plan.json').read_text())
parser=Path(sys.argv[2]).resolve()/'examples/utility/solar_csv.py'
assert hashlib.sha256(parser.read_bytes()).hexdigest()==expected['parser_sha256']
assert plan.digest==expected['csv_plan_digest']
assert digest({'csv_plan':plan.digest,'source_sha256':plan.source_sha256,
               'parser_sha256':expected['parser_sha256']})==expected['csv_source_binding']
inodes={(p.stat().st_dev,p.stat().st_ino) for p in root.rglob('*') if p.is_file()}
def denied(*args,**kwargs):raise RuntimeError('process network or clock denied')
subprocess.Popen=socket.socket=time.time=time.monotonic=denied

def audit(event,args):
    if event=='open':
        path,mode,flags=args
        if flags&(os.O_WRONLY|os.O_RDWR|os.O_CREAT|os.O_TRUNC|os.O_APPEND):raise RuntimeError('write denied')
        if isinstance(path,int):
            st=os.fstat(path)
            if (st.st_dev,st.st_ino) not in inodes:raise RuntimeError('outside descriptor denied')
        elif not Path(os.fsdecode(path)).absolute().is_relative_to(root):raise RuntimeError('outside read denied')
    if event=='import' or event.startswith(('socket.','subprocess.','os.exec','os.fork','os.posix_spawn')):
        raise RuntimeError('new import or execution denied')
sys.addaudithook(audit)
result=read_development(root,expected_manifest_digest=expected['manifest_digest'],
                        expected_preparer_sources=expected['preparer_sources'])
assert result.qualification=='not_admitted' and len(result.forcing.start)==12
assert result.plan.alignment.source_digest==expected['csv_source_binding']
assert set(result.labels)=={'fitting','selection'}
assert sum(result.labels['fitting'].score_mask)==sum(result.labels['selection'].score_mask)==4
assert len(result.forcing.start)==len(result.labels['fitting'].ac_kw)==len(result.labels['selection'].ac_kw)
assert all(result.forcing.drivers_usable)
print(json.dumps({'guarded_public_reader':True,'development_intervals':12,'fitting_score_rows':4,
                  'selection_score_rows':4,'raw_source_read':False,'reserved_artifacts_read':False,
                  'new_model_calls':0,'new_processes':0,'new_http_requests':0,'task_admitted':False}))
