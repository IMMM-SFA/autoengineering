"""Retain one authored facade's public suite, excluding private input scopes."""
from pathlib import Path
import json
import sys
import tarfile
from autoengineering.campaign.artifacts import read_bytes
from autoengineering.campaign.records import ArtifactRef
from autoengineering.utility.guest_receipts import GuestInferenceArchive,verify_guest_archive
from autoengineering.utility.scoring import ConfirmationGrant
origin=Path(sys.argv[1]).resolve()
pointer=next((origin/'guest-inference-archives').glob('*.json'))
ref=ArtifactRef.from_json(pointer.read_text())
archive=GuestInferenceArchive.from_json(read_bytes(origin,ref).decode())
job=json.loads(read_bytes(origin,archive.job));launch=json.loads(read_bytes(origin,archive.launch))
proof=json.loads(read_bytes(origin,archive.dispatch));grant=ConfirmationGrant.from_dict(job['grant'])
args=dict(source_digest=job['source_digest'],lock_digest=job['lock']['sha256'],
 executor_digest=job['executor_digest'],runtime_digest=launch['runtime_digest'],
 publisher_digest=proof['publisher_digest'])
expected=verify_guest_archive(origin,ref,grant,**args)
assert expected.status=='success'
for path in origin.rglob('*'):
 if path.is_file():
  assert path.name not in {'drivers.npz','guest-inputs.tar','input-package.json','private-labels.json'}
  assert b'AUTHORED-PRIVATE-CANARY' not in path.read_bytes()
bundle=Path(__file__).parent
with tarfile.open(bundle/'authored-facade-public.tar.gz','w:gz') as archive:
 archive.add(origin,arcname='public')
(bundle/'expected-recovery.json').write_text(json.dumps(dict(grant=grant.to_dict(),
 execution=expected.to_dict()),indent=2)+'\n')
print('Original authored public facade suite retained.')
