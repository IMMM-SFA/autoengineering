"""Fresh recovery-only facade replay, with private scopes absent and guarded reads."""
import importlib.abc
from pathlib import Path
import json
import os
import socket
import subprocess
import sys
import tarfile
import tempfile
import time
class NoModels(importlib.abc.MetaPathFinder):
 def find_spec(self,fullname,*args):
  if fullname.split('.')[0] in {'examples','torch','openmdao','pvlib','py_wake'}:
   raise AssertionError('model import')
sys.meta_path.insert(0,NoModels())
import numpy as np
from autoengineering.utility.guest_executor import GuestRoutingExecutor
from autoengineering.utility.guest_receipts import VerifiedGuestExecution
from autoengineering.utility.store import SuiteStore
from autoengineering.utility.scoring import ConfirmationGrant
from autoengineering.utility import guest_executor,guest_archive
bundle=Path(__file__).parent
expected=json.loads((bundle/'expected-recovery.json').read_text())
grant=ConfirmationGrant.from_dict(expected['grant'])
with tempfile.TemporaryDirectory() as directory:
 root=Path(directory).resolve()
 with tarfile.open(bundle/'authored-facade-public.tar.gz') as archive:
  archive.extractall(root,filter='data')
 for path in root.rglob('*'):
  path.chmod(0o700 if path.is_dir() else 0o600)
 public=root/'public'
 def forbidden(*args,**kwargs):raise AssertionError('execution, source probe, array load or clock')
 subprocess.Popen=socket.socket=np.load=time.monotonic=time.time=forbidden
 guest_executor.publisher_identity=guest_archive.publisher_identity=forbidden
 store=SuiteStore(public)
 store.replay()  # Initialize metadata-reader imports before the strict read barrier.
 active=[True]
 def reads(event,args):
  if not active[0] or event!='open' or isinstance(args[0],int):return
  path=Path(args[0]).resolve();mode,flags=args[1:3]
  if not path.is_relative_to(public):raise AssertionError('unrelated or private read')
  writing=mode is not None and any(c in mode for c in 'wax+')
  writing=writing or flags&(os.O_WRONLY|os.O_RDWR|os.O_CREAT|os.O_TRUNC|os.O_APPEND)
  if writing:raise AssertionError('write')
 sys.addaudithook(reads)
 executor=GuestRoutingExecutor.for_recovery(store,grant,root/'missing-work')
 assert executor.source_root is executor.driver_root is executor.api is None
 assert executor.recover(grant)==VerifiedGuestExecution.from_dict(expected['execution'])
 try:executor.execute(grant)
 except ValueError as error:assert 'recovery-only' in str(error)
 else:raise AssertionError('recovery facade dispatched')
 print('Original facade result agrees from relocated public metadata only.')
 active[0]=False
