"""Verify public source/service/work commitments without private data or physics."""
import sys
from pathlib import Path
sys.path.insert(0, str(Path(sys.argv[2]).resolve()))
import importlib.abc
class NoModels(importlib.abc.MetaPathFinder):
    def find_spec(self, fullname, *args):
        if fullname.split('.')[0] in {'openmdao', 'py_wake', 'topfarm', 'torch', 'pvlib', 'h5py'}:
            raise RuntimeError('optional model imports denied')
sys.meta_path.insert(0, NoModels())
import hashlib, json, os, socket, subprocess, time
from autoengineering.campaign.serialization import digest, unique_json
from autoengineering.campaign.records import ArtifactRef
from autoengineering.utility.store import SuiteStore
from autoengineering.utility.controller import UtilityController
from autoengineering.agents.batch import BatchSpec, BatchStore
from autoengineering.utility.baseline_selection import follower_group
from autoengineering.utility.confirmation import read_selected_state
from autoengineering.utility.scoring import ConfirmationGrant, read_receipt, validate_grant
from examples.utility import engineering_source_binding as binding
from examples.utility import engineering_scenarios as source
from examples.utility import engineering_load_scoring as kernel
from examples.utility import engineering_load_callback as callback
root = Path(sys.argv[1]).resolve()
expected = json.loads((root.parent / 'expected-public.json').read_text())
for name, sha in expected['source_snapshots'].items():
    assert hashlib.sha256((root.parent / 'executed-source' / name).read_bytes()).hexdigest() == sha
inodes = {(p.stat().st_dev, p.stat().st_ino) for p in root.rglob('*') if p.is_file()}
def denied(*args, **kwargs):
    raise RuntimeError('private source model entropy network process or clock denied')
source.read_private = source.derive = source.secrets.token_bytes = kernel.run_lifecycle = denied
subprocess.Popen = socket.socket = time.time = time.monotonic = denied

def audit(event, args):
    if event == 'open':
        path, mode, flags = args
        if flags & (os.O_WRONLY | os.O_RDWR | os.O_CREAT | os.O_TRUNC | os.O_APPEND):
            raise RuntimeError('write denied')
        if isinstance(path, int):
            st = os.fstat(path)
            if (st.st_dev, st.st_ino) not in inodes:
                raise RuntimeError('outside descriptor denied')
        elif not Path(os.fsdecode(path)).absolute().is_relative_to(root):
            raise RuntimeError('outside read denied')
    if event == 'import' or event.startswith(('socket.', 'subprocess.', 'os.exec', 'os.fork', 'os.posix_spawn')):
        raise RuntimeError('new import or execution denied')
sys.addaudithook(audit)
for name, sha in expected['files'].items():
    assert hashlib.sha256((root / name).read_bytes()).hexdigest() == sha
request = unique_json((root / 'request.json').read_text())
result = unique_json((root / 'result.json').read_text())
intent = unique_json((root / 'callback-intent.json').read_text())
assert digest(request) == expected['request_digest'] == result['request_digest']
assert digest(result) == expected['result_digest']
assert unique_json((root / 'execution-intent.json').read_text())['request_digest'] == digest(request)
store = SuiteStore(root / 'suite')
assert store.spec.digest == expected['suite_digest']
state = store.replay()
grant = ConfirmationGrant.from_dict(intent['grant'])
selected = validate_grant(store, grant, state['frozen'])
bound = read_selected_state(store.root, selected.selected_state)
assert bound.configuration == {'source_current': request['source_current_A']}
fixed = binding.read_binding(store, ArtifactRef.from_dict(expected['binding_artifact']))
metadata = source.read_public(root / 'metadata', expected_manifest_digest=fixed.public_manifest_digest)
assert metadata['private_record_digest'] == fixed.private_record_digest
policy = kernel.LoadScorePolicy('utility-coupled-load-score-v1', bound.model.digest,
    request['pins']['files']['examples/utility/engineering_load_scoring.py'], request['minimum_node2_voltage_V'])
contract = callback.outcome_contract(policy)
code_digest = digest({'schema': 'utility-trusted-load-callback-v1',
    'source': request['pins']['files']['examples/utility/engineering_load_callback.py'],
    'kernel_policy': policy.digest, 'expected_model': bound.model.digest, 'outcome': contract.digest})
assert code_digest == expected['scorer_code_digest'] == intent['scorer_code_digest']
assert grant.scorer_digest == binding.scorer_identity(fixed, code_digest)
assert grant.outcome_digest == contract.digest
attempt = 'confirmation-' + grant.cell_digest[:48]
receipt = read_receipt(store.root, state['confirmation_scores'][attempt])
assert receipt.digest == expected['receipt_digest'] == result['receipt_digest']
assert receipt.grant_digest == grant.digest == result['grant_digest']
assert receipt.measurement.scorer_digest == grant.scorer_digest
work = unique_json((store.root / callback.WORK_DIRECTORY / (grant.digest + '.json')).read_text())
assert digest(work) == result['work_receipt_digest']
assert work['score_digest'] == receipt.score.digest and work['policy_digest'] == policy.digest
assert work['source_binding_digest'] == fixed.digest and work['grant_digest'] == grant.digest
assert work['attempted_model_starts'] == work['completed_model_starts'] == 24
assert work['total_inner_work'] == result['work']
assert work['derivative_counter_scope'] == 'component _linearize invocations, including Newton linearization; not FLOPs'
cases = unique_json((root / 'case-work.json').read_text())
assert [c['index'] for c in cases] == list(range(24))
assert all(c['required_checks_passed'] and c['work']['solver_iterations'] <= 100 for c in cases)
for name, value in work['total_inner_work'].items():
    assert sum(c['work'][name] for c in cases) == value
assert state['phase_used'][('fixed_evaluations', 'confirmation', 'scientific_starts')] == 1
assert state['phase_used'][('fixed_evaluations', 'confirmation', 'load_model_starts')] == 24
assert state['phase_used'][('fixed_evaluations', 'confirmation', 'load_solver_iterations')] == 2400
claim = unique_json((store.root / binding.ACCESS_DIRECTORY / (grant.digest + '.json')).read_text())
assert claim['source_binding_digest'] == fixed.digest and claim['source_reads_reserved'] == 1
assert not work['task_admitted'] and not work['private_boundary_qualified']
assert not work['os_deadline_qualified'] and not work['aggregate_rss_qualified']
assert receipt.score.status == 'success' and receipt.score.feasible
assert not any(p.name in {'scenarios.json', 'private.json', 'complete.json'} for p in root.rglob('*'))
assert 'openmdao' not in sys.modules
print(json.dumps({'source_binding_verified': True, 'saved_receipt_verified': True,
    'work_receipt_verified': True, 'completed_model_starts': 24, 'confirmation_starts_charged': 1,
    'reserved_native_starts_charged': 24, 'source_read_intents': 1,
    'new_models': 0, 'new_scoring': 0, 'new_source_reads': 0, 'new_entropy': 0,
    'network': 0, 'processes': 0, 'task_admitted': False}))
