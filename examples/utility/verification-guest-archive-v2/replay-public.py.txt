"""Cold minimal public guest archive verification, with private work absent."""
import importlib.abc
from pathlib import Path
import json
import os
import socket
import subprocess
import sys
import tarfile
import tempfile
import time

class NoModels(importlib.abc.MetaPathFinder):
    def find_spec(self, fullname, *args):
        if fullname.split('.')[0] in {'examples','torch','openmdao','pvlib','py_wake'}:
            raise AssertionError('public archive reader cannot import a model')
sys.meta_path.insert(0,NoModels())
import numpy as np
from autoengineering.campaign.records import ArtifactRef
from autoengineering.utility import guest_archive
from autoengineering.utility.guest_receipts import VerifiedGuestExecution,verify_guest_archive
from autoengineering.utility.scoring import ConfirmationGrant
bundle=Path(__file__).parent
expected=json.loads((bundle/'expected-public.json').read_text())
ref=ArtifactRef.from_dict(expected['ref'])
grant=ConfirmationGrant.from_dict(expected['grant'])
with tempfile.TemporaryDirectory() as directory:
    root=Path(directory).resolve()
    with tarfile.open(bundle/'authored-public-archive.tar.gz') as archive:
        archive.extractall(root,filter='data')
    for path in root.rglob('*'):
        path.chmod(0o700 if path.is_dir() else 0o600)
    def forbidden(*args,**kwargs):
        raise AssertionError('public archive replay cannot execute, load arrays or sample time')
    subprocess.Popen=socket.socket=time.monotonic=time.time=np.load=forbidden
    guest_archive.publisher_identity=guest_archive.read_completion=forbidden
    active=[True]
    def saved_reads(event,args):
        if not active[0] or event!='open' or isinstance(args[0],int):return
        path=Path(args[0]).resolve()
        mode,flags=args[1:3]
        if not path.is_relative_to(root):raise AssertionError('unrelated or private read')
        if (mode is not None and any(c in mode for c in 'wax+')) or flags&(
            os.O_WRONLY|os.O_RDWR|os.O_CREAT|os.O_TRUNC|os.O_APPEND
        ):raise AssertionError('public archive reader attempted a write')
    sys.addaudithook(saved_reads)
    actual=verify_guest_archive(root/'public',ref,grant,**expected['args'])
    assert actual==VerifiedGuestExecution.from_dict(expected['execution'])
    print('Original guest execution agrees from minimal public bytes only.')
    active[0]=False
