"""Retain only the explicit public closure from one authored test suite."""
from pathlib import Path
import json
import sys
import tarfile
import tempfile
from autoengineering.campaign.artifacts import read_bytes
from autoengineering.campaign.records import ArtifactRef
from autoengineering.utility.guest_receipts import GuestInferenceArchive, verify_guest_archive
from autoengineering.utility.scoring import ConfirmationGrant

origin=Path(sys.argv[1]).resolve()
pointers=list((origin/'guest-inference-archives').glob('*.json'))
assert len(pointers)==1
ref=ArtifactRef.from_json(pointers[0].read_text())
archive=GuestInferenceArchive.from_json(read_bytes(origin,ref).decode())
job=json.loads(read_bytes(origin,archive.job).decode())
proof=json.loads(read_bytes(origin,archive.dispatch).decode())
launch=json.loads(read_bytes(origin,archive.launch).decode())
args=dict(source_digest=job['source_digest'],lock_digest=job['lock']['sha256'],
 executor_digest=job['executor_digest'],runtime_digest=launch['runtime_digest'],
 publisher_digest=proof['publisher_digest'])
grant=ConfirmationGrant.from_dict(job['grant'])
expected=verify_guest_archive(origin,ref,grant,**args)
assert expected.status=='success'
refs=[ref]+[getattr(archive,name) for name in [
 'job','intent','claim','launch','start','stop','cleanup','capture','outputs','predictions','dispatch']]
refs += [ArtifactRef.from_dict(job['environment']),
 ArtifactRef.from_dict(proof['completion']['start_response']),
 ArtifactRef.from_dict(proof['completion']['watchdog']['absence']),
 ArtifactRef.from_dict(proof['completion']['capture']['worker_result'])]
bundle=Path(__file__).parent
with tempfile.TemporaryDirectory() as directory:
 root=Path(directory).resolve()/'public';root.mkdir(mode=0o700)
 for item in refs:
  if item is None:continue
  path=root/item.path;path.parent.mkdir(parents=True,exist_ok=True)
  path.write_bytes(read_bytes(origin,item));path.chmod(0o600)
 claim=root/'inference-claims'/pointers[0].name
 claim.parent.mkdir(mode=0o700)
 claim.write_bytes((origin/'inference-claims'/pointers[0].name).read_bytes())
 for path in root.rglob('*'):
  if path.is_dir():path.chmod(0o700)
 assert verify_guest_archive(root,ref,grant,**args)==expected
 with tarfile.open(bundle/'authored-public-archive.tar.gz','w:gz') as archive:
  archive.add(root,arcname='public')
 print('Minimal public archive retained:',sum(p.is_file() for p in root.rglob('*')),'files')
(bundle/'expected-public.json').write_text(json.dumps(dict(ref=ref.to_dict(),grant=grant.to_dict(),
 args=args,execution=expected.to_dict()),indent=2)+'\n')
