from pathlib import Path
import urllib.request,urllib.error,hashlib,json,time
root=Path(__file__).resolve().parent
items=json.loads((root/'requests.json').read_text())
requests=[('metrics-2107.parquet', 'https://oedi-data-lake.s3.amazonaws.com/pvdaq/parquet/metrics/metrics__system_2107__part000.parquet'), ('inverters.parquet', 'https://oedi-data-lake.s3.amazonaws.com/pvdaq/parquet/inverters/part-00000-a9a6892f-3bca-42b7-a65e-2d8355306319-c000.snappy.parquet'), ('modules.parquet', 'https://oedi-data-lake.s3.amazonaws.com/pvdaq/parquet/modules/part-00000-4a77878b-ac1d-4920-966b-0536b3ff617b-c000.snappy.parquet'), ('prize-change-log-initial.pdf', 'https://oedi-data-lake.s3.amazonaws.com/pvdaq/2023-solar-data-prize/change_log_2023_Solar_Data_Prize_20250227.pdf'), ('prize-change-log-latest.pdf', 'https://oedi-data-lake.s3.amazonaws.com/pvdaq/2023-solar-data-prize/change_log_2023_Solar_Data_Prize_20260519.pdf')]
for name,url in requests:
    item={'path':name,'url':url,'max_body_bytes':512*1024,'status':'intent','automatic_retries':False}
    items.append(item)
    receipt=root/'requests.json'
    receipt.write_text(json.dumps(items,indent=2)+'\n')
    start=time.monotonic()
    try:
        with urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'Autoengineering-source-metadata-review/0.2'}),timeout=40) as response:
            data=response.read(item['max_body_bytes']+1)
            if len(data)>item['max_body_bytes']:raise ValueError('metadata body bound')
            with (root/name).open('xb') as output:output.write(data)
            item.update(status='complete',http_status=response.status,final_url=response.url,body_bytes=len(data),sha256=hashlib.sha256(data).hexdigest())
    except Exception as error:
        item.update(status='failed',error_type=type(error).__name__)
        if isinstance(error,urllib.error.HTTPError):item['http_status']=error.code
    item.update(elapsed_seconds=time.monotonic()-start,total_network_bytes=None,total_http_exchanges=None)
    receipt.write_text(json.dumps(items,indent=2)+'\n')
    print(json.dumps(item),flush=True)
