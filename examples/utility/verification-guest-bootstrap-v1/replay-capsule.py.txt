"""Cold authored stdio capsule replay, optionally using an extracted project wheel."""
import importlib.abc
import hashlib
import io
import json
from pathlib import Path
import socket
import subprocess
import sys
import tarfile
import tempfile
import time

if len(sys.argv) == 2:
    sys.path.insert(0, sys.argv[1])
class NoModels(importlib.abc.MetaPathFinder):
    def find_spec(self, fullname, *args):
        if fullname.split('.')[0] in {'examples', 'torch', 'openmdao', 'pvlib', 'py_wake'}:
            raise AssertionError('model import')
sys.meta_path.insert(0, NoModels())
from autoengineering.utility import guest_bootstrap as module
from autoengineering.utility.guest_owner import GuestOwnerJournal
if len(sys.argv) == 2:
    assert Path(module.__file__).is_relative_to(Path(sys.argv[1]))
bundle = Path(__file__).parent
expected = json.loads((bundle / 'expected-replay.json').read_text())
with tempfile.TemporaryDirectory() as directory:
    destination = Path(directory).resolve()
    with tarfile.open(bundle / 'authored-stdio-capsule.tar.gz') as archive:
        archive.extractall(destination, filter='data')
    for path in destination.rglob('*'):
        path.chmod(0o700 if path.is_dir() else 0o600)
    def forbidden(*args, **kwargs):
        raise AssertionError('replay cannot execute, connect or sample a new clock')
    subprocess.Popen = socket.socket = time.monotonic = forbidden
    owner = GuestOwnerJournal(destination / 'owner')
    m = owner.manifest
    raw = module.encode_request(owner)
    request, contents = module.decode_request(io.BytesIO(raw))
    capture = module.read_capture(destination / 'received', m.job, m.launch, request.package)
    assert owner.manifest.digest == expected['owner_digest']
    assert hashlib.sha256(raw).hexdigest() == expected['request_sha256']
    assert request.package.digest == expected['package_digest']
    assert capture.to_dict() == expected['capture']
    assert len(contents) == expected['input_files']
    print('Cold authored stdio capsule agrees; no model, image or runtime qualification.')
