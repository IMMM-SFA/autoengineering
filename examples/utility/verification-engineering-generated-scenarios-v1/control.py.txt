"""A real entropy draw for an authored no-execution suite. No scenario values printed."""
import sys,hashlib,json
from pathlib import Path
sys.path.insert(0,str(Path(__file__).resolve().parents[3]))
from autoengineering.campaign.serialization import digest
from autoengineering.utility.contracts import TaskSpec,RouteSpec,ResourceCap,BudgetSpec,ComparisonCell,UtilitySuiteSpec
from autoengineering.utility.confirmation import SelectionManifest,SelectedCell,freeze
from autoengineering.utility.store import SuiteStore
from examples.utility import engineering_scenarios as g
root=Path(sys.argv[1]).absolute()
root.mkdir(mode=0o700)
public,private=root/'public',root/'private'
public.mkdir(mode=0o700);private.mkdir(mode=0o700)
policy=g.ScenarioPolicy('utility-engineering-generated-scenarios-v1','engineering',
                        {'ground_voltage':(-.05,.05),'resistor1_ohm':(95.,105.),
                         'resistor2_ohm':(9500.,10500.)},g.source_identity())
purpose={'scope':'authored source-preparation control only','models_allowed':0,'task_admitted':False}
h=digest(purpose)
task=TaskSpec('engineering','engineering',h,policy.digest,h,(h,),())
routes=tuple(RouteSpec(name,kind,digest({'route':name,'control':purpose}),h)
             for name,kind in (('conventional','conventional'),('multi','multi_agent')))
caps=tuple(ResourceCap(name,units,limit,'reservation') for name,units,limit in
           (('scientific_starts','count',1),('host_invocations','count',0),
            ('training_starts','count',0),('active_time','s',60)))
# The suite schema requires a positive confirmation reserve. It remains unused:
# both cells are explicitly not_attempted, and this control dispatches no executor.
budgets=(BudgetSpec('fixed_evaluations',caps,(ResourceCap('scientific_starts','count',1,'reservation'),)),)
cells=tuple(ComparisonCell('engineering',r.route_id,'fixed_evaluations',0,17,policy.digest) for r in routes)
spec=UtilitySuiteSpec('utility-suite-v1','engineering-generated-source-control',(task,),routes,budgets,cells,
                      digest({'sampler':g.source_identity()}),h,h,h,digest({'purpose':purpose,'policy':policy.digest}))
store=SuiteStore.create(root/'suite',spec)
result=g.prepare(store,policy,public,private,expected_suite_digest=spec.digest)
try:g.read_private(store,public,private,expected_manifest_digest=digest(result),expected_selection_digest='0'*64)
except ValueError:early_refused=True
else:raise AssertionError('private source opened without barrier')
barrier=freeze(store,SelectionManifest('utility-selection-v1',spec.digest,spec.protocol_digest,
               tuple(SelectedCell(c.digest,'not_attempted',None,None,None,'Source-preparation control; no model execution') for c in spec.cells)))
rows=g.read_private(store,public,private,expected_manifest_digest=digest(result),expected_selection_digest=barrier)
assert len(rows)==24
for row in rows:
    for name,value in row['forcing'].items():assert policy.bounds[name][0]<=value<=policy.bounds[name][1]
summary={'scope':'Actual fresh source generation in authored two-cell no-execution suite; not independent scientific confirmation',
         'suite_digest':spec.digest,'policy_digest':policy.digest,'public_manifest_digest':digest(result),
         'selection_barrier_digest':barrier,'generation_before_barrier':True,'early_private_read_refused':early_refused,
         'generated_scenarios':24,'private_derivation_verified':True,'private_values_exported':False,
         'unused_suite_scientific_reserve':1,'scientific_starts':0,'host_starts':0,
         'model_calls':0,'training_calls':0,'scoring_calls':0,'provider_calls':0,'task_admitted':False,
         'historical_data_independence_attested':False}
(root/'result.json').write_text(json.dumps(summary,indent=2)+'\n')
print(json.dumps(summary))
