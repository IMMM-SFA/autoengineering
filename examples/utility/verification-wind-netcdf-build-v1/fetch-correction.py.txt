"""Continue only bounded public downloads, retaining the oversized discovery charge."""
from pathlib import Path
import hashlib,json,urllib.request
out=Path.cwd()/'examples/utility/verification-wind-netcdf-build-v1'
stage=Path('/private/tmp/utility-wind-netcdf-build-v1')
assert not (out/'fetch-results.json').exists()
netcdf_raw=(stage/'netcdf-metadata.json').read_bytes();netcdf=json.loads(netcdf_raw)
prior={'first_attempt_error':'Cython all-version metadata exceeded 1 MB bound; no build or install started',
    'logical_http_requests':2,'saved_netcdf_metadata_bytes':len(netcdf_raw),'discarded_oversize_body_bytes_read':1000001,
    'netcdf_metadata_sha256':hashlib.sha256(netcdf_raw).hexdigest(),'http_exchanges':'unknown'}
(out/'initial-fetch-failure.json').write_text(json.dumps(prior,indent=2)+'\n')
correction={'schema':'utility-wind-netcdf-fetch-correction-v1','fixed_Cython_version':'3.2.4',
    'reason':'Use a fixed-version metadata endpoint, avoiding the unbounded release history',
    'previous_requests_charged':2,'additional_requests_allowed':3,'cumulative_request_cap':5,
    'cumulative_download_bytes_cap_including_discarded_body':19000000,'model_calls':0,'build_attempts_so_far':0}
(out/'fetch-correction.json').write_text(json.dumps(correction,indent=2)+'\n')
results=[]
def get(url,name,cap):
    with urllib.request.urlopen(url,timeout=60) as response:
        body=response.read(cap+1);assert len(body)<=cap
        meta={'url':url,'final_url':response.url,'status':response.status,'saved_bytes':len(body),'sha256':hashlib.sha256(body).hexdigest()}
    (stage/name).write_bytes(body);results.append(meta);return body
cython=json.loads(get('https://pypi.org/pypi/Cython/3.2.4/json','cython-metadata-v2.json',1000000))
source=next(file for file in netcdf['urls'] if file['packagetype']=='sdist')
wheel=next(file for file in cython['urls'] if file['filename'].endswith('cp312-cp312-macosx_11_0_arm64.whl'))
assert source['size']<=2000000 and wheel['size']<=15000000
(out/'selected-inputs.json').write_text(json.dumps({'netcdf4':source,'Cython':wheel,'Cython_version':'3.2.4'},indent=2)+'\n')
for file,cap in [(source,2000000),(wheel,15000000)]:
    body=get(file['url'],file['filename'],cap);assert hashlib.sha256(body).hexdigest()==file['digests']['sha256']
total=len(netcdf_raw)+1000001+sum(row['saved_bytes'] for row in results);assert total<=19000000
(out/'fetch-results.json').write_text(json.dumps({'prior':prior,'additional_requests':results,
    'cumulative_logical_requests':5,'bytes_read_including_discarded_body':total,'http_exchanges':'unknown'},indent=2)+'\n')
print(json.dumps({'requests_charged':5,'bytes_read':total,'source':source['filename'],'Cython':wheel['filename']}))
