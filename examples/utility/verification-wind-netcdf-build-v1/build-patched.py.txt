"""Build once against installed NumPy, installing only into new isolated targets."""
from pathlib import Path
import hashlib,json,os,signal,subprocess,sys,time
from importlib.metadata import version
root=Path.cwd();out=root/'examples/utility/verification-wind-netcdf-build-v1/patched';stage=Path('/private/tmp/utility-wind-netcdf-build-v2')
out.mkdir(mode=0o700)
assert version('numpy')=='2.5.2' and version('netCDF4')=='1.7.4'
assert not (stage/'wheel').exists()
selected=json.loads((out.parent/'selected-inputs.json').read_text());scm=json.loads((out.parent/'build-dependency-fetch.json').read_text())['selected_wheel']
inputs=[selected['Cython'],scm]
for item in inputs:
 assert hashlib.sha256((Path('/private/tmp/utility-wind-netcdf-build-v1')/item['filename']).read_bytes()).hexdigest()==item['digests']['sha256']
intent={'schema':'utility-wind-netcdf-build-intent-v1','build_attempts':1,'timeout_seconds':300,
    'python':sys.executable,'python_version':sys.version,'numpy':version('numpy'),'netCDF4':'1.7.4','Cython':'3.2.4',
    'setuptools':version('setuptools'),'setuptools-scm':'8.3.1',
    'netcdf_config':'/opt/homebrew/bin/nc-config','hdf5_prefix':'/opt/homebrew/opt/hdf5',
    'netcdf_prefix':'/opt/homebrew/opt/netcdf','NETCDF4_LIMITED_API':'0',
    'cumulative_build_attempts':2,'local_patch':True,'MACOSX_DEPLOYMENT_TARGET':'26.0','existing_environment_modified':False,'scientific_model_calls':0,'private_reads':0}
(out/'build-intent.json').write_text(json.dumps(intent,indent=2)+'\n')
def install(target,wheels,label):
 result=subprocess.run([sys.executable,'-I','-m','pip','install','--no-index','--no-deps','--no-compile','--no-cache-dir',
    '--target',str(target),*map(str,wheels)],capture_output=True,text=True,timeout=60)
 (out/(label+'.stdout.txt')).write_text(result.stdout);(out/(label+'.stderr.txt')).write_text(result.stderr)
 assert result.returncode==0
assert Path('/private/tmp/utility-wind-netcdf-build-v1/build-deps').is_dir()
env=os.environ.copy();env.update({'PYTHONPATH':'/private/tmp/utility-wind-netcdf-build-v1/build-deps','NETCDF4_DIR':'/opt/homebrew/opt/netcdf',
 'HDF5_DIR':'/opt/homebrew/opt/hdf5','USE_NCCONFIG':'1','NETCDF4_LIMITED_API':'0',
 'PATH':'/opt/homebrew/bin:'+str(Path(sys.executable).parent)+':/usr/bin:/bin',
 'SOURCE_DATE_EPOCH':'1793318400','MACOSX_DEPLOYMENT_TARGET':'26.0','PIP_NO_INDEX':'1'})
started=time.monotonic();p=subprocess.Popen([sys.executable,'-B','-c',
    "from setuptools.build_meta import build_wheel; build_wheel('../../wheel')"],
    cwd=stage/'source/netcdf4-1.7.4',env=env,stdout=subprocess.PIPE,stderr=subprocess.PIPE,text=True,start_new_session=True)
timed_out=False
try:stdout,stderr=p.communicate(timeout=300)
except subprocess.TimeoutExpired:
 timed_out=True;os.killpg(p.pid,signal.SIGKILL);stdout,stderr=p.communicate(timeout=10)
(out/'build.stdout.txt').write_text(stdout);(out/'build.stderr.txt').write_text(stderr)
result={'schema':'utility-wind-netcdf-build-result-v1','returncode':p.returncode,'timed_out':timed_out,'seconds':time.monotonic()-started,
 'build_attempts':1,'compiler_process_group_terminated_on_timeout':timed_out,'new_model_calls':0}
(out/'build-result.json').write_text(json.dumps(result,indent=2)+'\n')
assert p.returncode==0 and not timed_out
wheel=next((stage/'wheel').glob('*.whl'));install(stage/'installed',[wheel],'candidate-install')
shutil=None
import shutil
shutil.copyfile(wheel,out/wheel.name)
result.update(wheel=wheel.name,wheel_sha256=hashlib.sha256(wheel.read_bytes()).hexdigest(),install_target=str(stage/'installed'))
(out/'build-result.json').write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps(result))
