"""Authored NetCDF compatibility checks; no weather source or scientific model."""
from pathlib import Path
import json,sys,warnings
import numpy as np
candidate,scope,mode=sys.argv[1:]
if candidate!='-':sys.path.insert(0,candidate)
root=Path(scope).resolve()
with warnings.catch_warnings(record=True) as observed:
    warnings.simplefilter('always')
    import netCDF4 as nc
    import xarray as xr
    import py_wake
messages=[{'category':item.category.__name__,'message':str(item.message)} for item in observed]
if candidate!='-':assert Path(nc.__file__).resolve().is_relative_to(Path(candidate).resolve())
if mode=='write':
    root.mkdir(mode=0o700)
    for format in ['NETCDF4_CLASSIC','NETCDF4']:
        with nc.Dataset(root/(format+'.nc'),'w',format=format) as ds:
            ds.createDimension('time',None)
            ds.setncattr('authored_scalar',3.5);ds.setncattr('authored_vector',[1,2,3])
            times=ds.createVariable('time','f8',('time',));times.units='hours since 2000-01-01 00:00:00';times.calendar='standard';times[:]=np.arange(48)
            speed=ds.createVariable('speed','f8',('time',),fill_value=-9999.,zlib=True,complevel=4)
            values=np.ma.array(2.+np.arange(48)/10.,mask=np.arange(48)==7);speed[:]=values
            direction=ds.createVariable('direction','f8',('time',));direction[:]=(np.arange(48)*17)%360
            packed=ds.createVariable('packed','i2',('time',));packed.scale_factor=.1;packed.add_offset=0.;packed[:]=2.+np.arange(48)/10.
    with nc.Dataset(root/'enhanced.nc','w',format='NETCDF4') as ds:
        ds.createDimension('site',3)
        compound=ds.createCompoundType(np.dtype([('u','f8'),('v','f8')]),'wind_pair')
        pair=ds.createVariable('pair',compound,('site',));pair[:]=np.array([(1.,2.),(3.,4.),(5.,6.)],dtype=compound.dtype)
        vlen=ds.createVLType(np.int32,'authored_ints');var=ds.createVariable('samples',vlen,('site',))
        names=ds.createVariable('name',str,('site',))
        for i in range(3):var[i]=np.arange(i+1,dtype='i4');names[i]='authored-'+str(i)
for format in ['NETCDF4_CLASSIC','NETCDF4']:
    with nc.Dataset(root/(format+'.nc')) as ds:
        assert np.asarray(ds.getncattr('authored_scalar')).shape==() and float(ds.getncattr('authored_scalar'))==3.5
        np.testing.assert_array_equal(ds.getncattr('authored_vector'),[1,2,3])
        values=ds['speed'][:];assert np.ma.getmaskarray(values).tolist()==(np.arange(48)==7).tolist()
        np.testing.assert_array_equal(values.compressed(),(2.+np.arange(48)/10.)[np.arange(48)!=7])
        np.testing.assert_array_equal(ds['direction'][:],(np.arange(48)*17)%360)
        np.testing.assert_allclose(ds['packed'][:],2.+np.arange(48)/10.,rtol=0,atol=1e-12)
        decoded=nc.num2date(ds['time'][:],ds['time'].units,ds['time'].calendar)
        np.testing.assert_array_equal(nc.date2num(decoded,ds['time'].units,ds['time'].calendar),np.arange(48))
    with xr.open_dataset(root/(format+'.nc'),engine='netcdf4') as dataset:
        loaded=dataset.load();assert loaded.sizes['time']==48 and np.isnan(loaded['speed'].values[7])
        assert loaded['time'].values[0]==np.datetime64('2000-01-01T00:00:00')
with nc.Dataset(root/'enhanced.nc') as ds:
    np.testing.assert_array_equal(ds['pair'][:]['u'],[1.,3.,5.])
    np.testing.assert_array_equal(ds['pair'][:]['v'],[2.,4.,6.])
    for i in range(3):np.testing.assert_array_equal(ds['samples'][i],np.arange(i+1,dtype='i4'))
    assert ds['name'][:].tolist()==['authored-0','authored-1','authored-2']
print(json.dumps({'netcdf4':nc.__version__,'numpy':np.__version__,'pywake':py_wake.__version__,
    'module_path':nc.__file__,'netcdf_c':nc.__netcdf4libversion__,'hdf5':nc.__hdf5libversion__,
    'warnings':messages,'checks':'passed','mode':mode,'scientific_model_calls':0,'private_reads':0}))
