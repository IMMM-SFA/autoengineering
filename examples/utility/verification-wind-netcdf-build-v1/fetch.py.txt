"""Bounded public build inputs for one isolated netCDF4 1.7.4 experiment."""
from pathlib import Path
import hashlib,json,urllib.request
out=Path.cwd()/'examples/utility/verification-wind-netcdf-build-v1'
stage=Path('/private/tmp/utility-wind-netcdf-build-v1');stage.mkdir(mode=0o700)
request={'schema':'utility-wind-netcdf-build-request-v1','purpose':'same-version isolated build against current NumPy; no scientific selection',
    'netcdf4_version':'1.7.4','numpy_version':'2.5.2','maximum_logical_http_requests':4,'maximum_saved_download_bytes':19000000,
    'metadata_timeout_seconds':30,'source_timeout_seconds':60,'build_attempts':1,'build_timeout_seconds':300,
    'install_target':str(stage/'installed'),'build_dependency_target':str(stage/'build-deps'),
    'global_or_existing_environment_mutation':False,'scientific_model_calls':0,'private_source_reads':0,
    'metadata_sources':['https://pypi.org/pypi/netCDF4/1.7.4/json','https://pypi.org/pypi/Cython/json']}
(out/'request.json').write_text(json.dumps(request,indent=2)+'\n')
results=[]
def get(url,name,cap):
    with urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'Autoengineering-development-qualification'}),timeout=60) as response:
        body=response.read(cap+1);assert len(body)<=cap
        meta={'url':url,'final_url':response.url,'status':response.status,'saved_bytes':len(body),'sha256':hashlib.sha256(body).hexdigest(),'http_exchanges':'unknown'}
    (stage/name).write_bytes(body);results.append(meta);return body
netcdf=json.loads(get(request['metadata_sources'][0],'netcdf-metadata.json',1000000))
cython=json.loads(get(request['metadata_sources'][1],'cython-metadata.json',1000000))
source=next(file for file in netcdf['urls'] if file['packagetype']=='sdist')
# The current host is CPython 3.12 macOS ARM64. Refuse other wheel tags.
wheel=next(file for file in cython['urls'] if file['filename'].endswith('cp312-cp312-macosx_11_0_arm64.whl'))
assert source['size']<=2000000 and wheel['size']<=15000000
selected={'netcdf4':source,'Cython':wheel,'Cython_version':cython['info']['version']}
(out/'selected-inputs.json').write_text(json.dumps(selected,indent=2)+'\n')
for file,cap in [(source,2000000),(wheel,15000000)]:
    body=get(file['url'],file['filename'],cap)
    assert hashlib.sha256(body).hexdigest()==file['digests']['sha256']
assert len(results)==4 and sum(row['saved_bytes'] for row in results)<=19000000
(out/'fetch-results.json').write_text(json.dumps({'requests':results,'saved_bytes':sum(row['saved_bytes'] for row in results)},indent=2)+'\n')
print(json.dumps({'downloads':len(results),'saved_bytes':sum(row['saved_bytes'] for row in results),'Cython_version':cython['info']['version'],'sdist':source['filename'],'wheel':wheel['filename']}))
