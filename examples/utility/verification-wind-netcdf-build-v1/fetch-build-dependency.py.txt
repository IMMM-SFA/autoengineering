"""Fetch the missing declared build dependency with two bounded public requests."""
from pathlib import Path
import hashlib,json,urllib.request
out=Path.cwd()/'examples/utility/verification-wind-netcdf-build-v1';stage=Path('/private/tmp/utility-wind-netcdf-build-v1')
prior=json.loads((out/'fetch-results.json').read_text());assert prior['cumulative_logical_requests']==5
intent={'package':'setuptools-scm','version':'8.3.1','reason':'Declared sdist build requirement absent in wind environment',
    'previous_requests_charged':5,'additional_requests_cap':2,'cumulative_requests_cap':7,'cumulative_bytes_cap':19000000}
(out/'build-dependency-fetch-intent.json').write_text(json.dumps(intent,indent=2)+'\n')
rows=[]
def get(url,name):
    with urllib.request.urlopen(url,timeout=30) as response:
        body=response.read(1000001);assert len(body)<=1000000
        row={'url':url,'final_url':response.url,'status':response.status,'saved_bytes':len(body),'sha256':hashlib.sha256(body).hexdigest()}
    (stage/name).write_bytes(body);rows.append(row);return body
metadata=json.loads(get('https://pypi.org/pypi/setuptools-scm/8.3.1/json','scm-metadata.json'))
wheel=next(file for file in metadata['urls'] if file['filename'].endswith('py3-none-any.whl'))
raw=get(wheel['url'],wheel['filename']);assert hashlib.sha256(raw).hexdigest()==wheel['digests']['sha256']
assert prior['bytes_read_including_discarded_body']+sum(row['saved_bytes'] for row in rows)<19000000
(out/'build-dependency-fetch.json').write_text(json.dumps({'inputs':rows,'selected_wheel':wheel,'cumulative_logical_requests':7,
    'cumulative_bytes_read':prior['bytes_read_including_discarded_body']+sum(row['saved_bytes'] for row in rows),'http_exchanges':'unknown'},indent=2)+'\n')
print(json.dumps({'requests_charged':7,'wheel':wheel['filename'],'requires':metadata['info']['requires_dist']}))
