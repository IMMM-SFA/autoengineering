"""Verify recorded experiment metadata without importing NumPy or native binaries."""
from pathlib import Path
import hashlib,json,os,sys
import xml.etree.ElementTree as ET
root=Path(sys.argv[1]).resolve()
def guard(event,args):
 if event=='open':
  path,mode,flags=args
  if isinstance(path,int) or not Path(path).resolve().is_relative_to(root):raise AssertionError('outside evidence read')
  if any(value in (mode or '') for value in 'wax+') or flags&(os.O_WRONLY|os.O_RDWR|os.O_CREAT|os.O_APPEND|os.O_TRUNC):raise AssertionError('write')
 elif event=='import' or event.startswith(('socket.','subprocess.','os.exec')) or event in ('os.system','os.fork','os.posix_spawn'):
  raise AssertionError('runtime or import')
sys.addaudithook(guard)
manifest=json.loads((root/'manifest.json').read_text())
for name,pin in manifest['files'].items():
 path=root/name;assert not path.is_symlink() and path.resolve().is_relative_to(root)
 raw=path.read_bytes();assert len(raw)==pin['size_bytes'] and hashlib.sha256(raw).hexdigest()==pin['sha256']
def read(name):return json.loads((root/name).read_text())
summary=read('verification.json')
assert summary['logical_http_requests_charged']==7 and summary['bytes_read_including_discarded_body']==4925020
assert summary['build_attempts_charged']==3 and not summary['candidate_promoted']
builds=[read(name+'/build-result.json' if name else 'build-result.json') for name in ['', 'patched','patched-v2']]
assert [result['returncode'] for result in builds]==[0,1,0]
assert sum(result['build_attempts'] for result in builds)==3 and not any(result['timed_out'] for result in builds)
for prefix,result in [('',builds[0]),('patched-v2/',builds[2])]:
 assert hashlib.sha256((root/(prefix+result['wheel'])).read_bytes()).hexdigest()==result['wheel_sha256']
probe=read('patched-v2/probe-result.json')
assert probe['candidate_binary_warning_absent'] and probe['original_extension_unchanged']
assert [result['checks'] for result in probe['results']]==['passed']*4
assert [len(result['warnings']) for result in probe['results']]==[1,0,1,0]
assert not probe['candidate_promoted_to_project_environment'] and probe['new_scientific_model_calls']==0
suites=list(ET.parse(root/'upstream-tests.xml').getroot().iter('testsuite'))
assert sum(int(s.get('tests','0')) for s in suites)==13
assert sum(int(s.get('failures','0')) for s in suites)==5
assert sum(int(s.get('errors','0')) for s in suites)==0
assert summary['scientific_model_calls']==summary['private_source_reads']==0
print(json.dumps({'schema':'utility-wind-netcdf-archive-replay-v1','builds_charged':3,'logical_http_requests_charged':7,'upstream_passed':8,'upstream_failed':5,
 'candidate_promoted':False,'new_model_calls':0,'native_binary_imports':0,'verification':'passed'}))
