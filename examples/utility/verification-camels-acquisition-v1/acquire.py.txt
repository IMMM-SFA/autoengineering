"""One bounded full CAMELS acquisition; hash bytes without parsing observations."""
from pathlib import Path
import hashlib,json,os,shutil,time,urllib.request
root=Path.cwd();out=root/'examples/utility/verification-camels-acquisition-v1'
private=root/'outputs/camels-acquisition-v1';private.mkdir(mode=0o700)
expected_size=3406626583;expected_md5='8e9a466710e8270b58f01d332a87184f'
request={'schema':'utility-camels-acquisition-request-v1','metadata_url':'https://zenodo.org/api/records/15529996',
 'archive_name':'basin_timeseries_v1p2_metForcing_obsFlow.zip','expected_size_bytes':expected_size,'expected_md5':expected_md5,
 'metadata_bytes_cap':2097152,'logical_http_requests_cap':2,'socket_timeout_seconds':30,
 'stream_elapsed_logical_limit_seconds':900,'maximum_one_read_overrun_seconds':30,'archive_body_bytes_cap':expected_size,
 'automatic_retries':0,'destination':str(private),'directory_mode':'0700','file_mode':'0600',
 'parse_observations':False,'scientific_model_calls':0,'confirmation_admitted':False,
 'minimum_free_bytes':8*1024**3,'personal_attestation_path':'examples/utility/confirmation-reserve-attestation-v1.json',
 'producer_sha256':hashlib.sha256(Path(__file__).read_bytes()).hexdigest()}
(out/'request.json').write_text(json.dumps(request,indent=2)+'\n')
assert shutil.disk_usage(private).free>=request['minimum_free_bytes']
started=time.monotonic();status={'logical_requests_dispatched':0,'archive_body_bytes_read':0,'archive_bytes_written':0,
 'observation_members_opened':0,'scientific_model_calls':0,'automatic_retry_allowed':False}
def save_failure(error):
 (out/'failure.json').write_text(json.dumps({'error_type':type(error).__name__,**status,'seconds':time.monotonic()-started},indent=2)+'\n')
try:
 status['logical_requests_dispatched']+=1
 with urllib.request.urlopen(request['metadata_url'],timeout=30) as response:
  raw=response.read(request['metadata_bytes_cap']+1);assert len(raw)<=request['metadata_bytes_cap'] and response.status==200
  meta_response={'url':request['metadata_url'],'final_url':response.url,'status':response.status,'body_bytes':len(raw),'sha256':hashlib.sha256(raw).hexdigest()}
 (out/'zenodo-record.json').write_bytes(raw)
 record=json.loads(raw);metadata=record['metadata']
 assert metadata['version']=='1.2' and metadata['access_right']=='open' and metadata['license']['id']=='cc-by-4.0'
 selected=next(file for file in record['files'] if file['key']==request['archive_name'])
 assert selected['size']==expected_size and selected['checksum']=='md5:'+expected_md5
 assert selected['links']['self']=='https://zenodo.org/api/records/15529996/files/'+request['archive_name']+'/content'
 url=selected['links']['self']+'?download=1'
 (out/'metadata-response.json').write_text(json.dumps({'response':meta_response,'selected_file':selected},indent=2)+'\n')
 sha=hashlib.sha256();md5=hashlib.md5(usedforsecurity=False);last_progress=0;download_started=time.monotonic()
 status['logical_requests_dispatched']+=1
 with urllib.request.urlopen(urllib.request.Request(url,headers={'Accept-Encoding':'identity'}),timeout=30) as response:
  assert response.status==200 and response.headers.get('Content-Encoding','identity')=='identity'
  length=response.headers.get('Content-Length');assert length is None or int(length)==expected_size
  fd=os.open(private/'archive.zip.part',os.O_WRONLY|os.O_CREAT|os.O_EXCL,0o600)
  with os.fdopen(fd,'wb') as writer:
   while True:
    assert time.monotonic()-download_started<=900
    chunk=response.read(min(8*1024**2,expected_size-status['archive_body_bytes_read']+1))
    if not chunk:break
    status['archive_body_bytes_read']+=len(chunk);assert status['archive_body_bytes_read']<=expected_size
    writer.write(chunk);status['archive_bytes_written']+=len(chunk);sha.update(chunk);md5.update(chunk)
    if status['archive_body_bytes_read']-last_progress>=64*1024**2:
     last_progress=status['archive_body_bytes_read'];print(json.dumps({'bytes_hashed':last_progress,'expected_bytes':expected_size}),flush=True)
   writer.flush();os.fsync(writer.fileno())
  archive_response={'url':url,'final_url':response.url,'status':response.status,'content_length':length,
   'etag':response.headers.get('ETag'),'last_modified':response.headers.get('Last-Modified')}
 assert status['archive_body_bytes_read']==status['archive_bytes_written']==expected_size
 assert md5.hexdigest()==expected_md5
 os.rename(private/'archive.zip.part',private/'archive.zip')
 receipt={'schema':'utility-camels-full-acquisition-v1','request_sha256':hashlib.sha256((out/'request.json').read_bytes()).hexdigest(),
  'metadata_response':meta_response,'archive_response':archive_response,'archive_bytes':expected_size,
  'archive_md5':md5.hexdigest(),'archive_sha256':sha.hexdigest(),'provider_md5_verified':True,
  'logical_http_requests':status['logical_requests_dispatched'],'http_exchange_count':None,'total_transfer_bytes':None,
  'saved_metadata_body_bytes':len(raw),'archive_download_seconds':time.monotonic()-download_started,'total_seconds':time.monotonic()-started,
  'private_archive':str(private/'archive.zip'),'archive_mode':oct((private/'archive.zip').stat().st_mode&0o777),
  'observation_members_opened':0,'scientific_model_calls':0,'confirmation_admitted':False,'protected_private_runtime_qualified':False,
  'physical_intervals_admitted':False,'cpu_seconds':None,'peak_rss_bytes':None}
 (out/'receipt.json').write_text(json.dumps(receipt,indent=2)+'\n')
 print(json.dumps({'status':'complete','bytes':expected_size,'sha256':sha.hexdigest(),'md5_verified':True,'observation_members_opened':0}),flush=True)
except BaseException as error:
 save_failure(error);raise
