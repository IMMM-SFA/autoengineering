"""Classify the original source refusal using fixed metadata messages, never values."""
from pathlib import Path
import hashlib,json,zipfile
from collections import Counter
from examples.utility.hydrology_camels import CamelsArchiveDaymetSource,_parse
from examples.utility.hydrology_camels_archive import _member
from examples.utility.hydrology_partitions import DailyHydrologyPlan
root=Path.cwd();out=root/'examples/utility/verification-camels-acquisition-v1'
assert (out/'inspection-failure.json').exists() and not (out/'diagnostic.json').exists()
source=CamelsArchiveDaymetSource.from_json((out/'source.json').read_text())
plan=DailyHydrologyPlan.from_json((out/'inspection-plan.json').read_text())
policy=json.loads((out/'inspection-policy.json').read_text())
intent={'schema':'utility-camels-refusal-diagnostic-intent-v1','original_failure_sha256':hashlib.sha256((out/'inspection-failure.json').read_bytes()).hexdigest(),
 'outputs':'Fixed parser refusal category, date/schema metadata and placeholder-hour counts only',
 'observational_values_exported':False,'changes_to_source_policy':False,'task_admitted':False,'model_calls':0}
(out/'diagnostic-intent.json').write_text(json.dumps(intent,indent=2)+'\n')
with zipfile.ZipFile(root/'outputs/camels-acquisition-v1/archive.zip') as archive:
 forcing=_member(archive,source.forcing_member);flow=_member(archive,source.streamflow_member)
assert hashlib.sha256(forcing).hexdigest()==source.forcing_sha256 and hashlib.sha256(flow).hexdigest()==source.streamflow_sha256
whitelist={'unsupported forcing columns or units','forcing basin metadata differs','driver and flow axes differ',
 'unsupported source row','flow gauge or date differs','forcing hour differs; no implicit date shift','missing or invalid drivers',
 'unknown negative streamflow sentinel','invalid flow quality flag','source dates differ from the frozen plan'}
try:_parse(source,plan,forcing,flow);reason='no refusal reproduced'
except Exception as error:reason=str(error) if str(error) in whitelist else 'other sanitized source validation refusal'
driver_lines=forcing.decode('ascii').splitlines()[4:];flow_lines=flow.decode('ascii').splitlines()
hours=Counter(line.split()[3] if len(line.split())==11 and line.split()[3] in ('0','12') else 'other' for line in driver_lines)
record={'schema':'utility-camels-refusal-diagnostic-v1','original_refusal_reason':reason,
 'forcing_row_count':len(driver_lines),'flow_row_count':len(flow_lines),'published_placeholder_hour_counts':dict(sorted(hours.items())),
 'declared_forcing_hour':source.forcing_hour,'metadata_readme_claim':'hour zero for daily values',
 'observational_values_exported':False,'source_policy_changed':False,'model_calls':0,'task_admitted':False}
(out/'diagnostic.json').write_text(json.dumps(record,indent=2)+'\n');print(json.dumps(record))
