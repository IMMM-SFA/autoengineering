"""Reconstruct public source receipts without opening the private archive or rows."""
from pathlib import Path
from datetime import date
import hashlib,json,os,sys
root=Path(sys.argv[1]).resolve()
def guard(event,args):
 if event=='open':
  path,mode,flags=args
  if isinstance(path,int) or not Path(path).resolve().is_relative_to(root):raise AssertionError('outside evidence')
  if any(value in (mode or '') for value in 'wax+') or flags&(os.O_WRONLY|os.O_RDWR|os.O_CREAT|os.O_APPEND|os.O_TRUNC):raise AssertionError('write')
 elif event=='import' or event.startswith(('socket.','subprocess.','os.exec')) or event in ('os.system','os.fork','os.posix_spawn'):
  raise AssertionError('runtime or import')
sys.addaudithook(guard)
def read(name):return json.loads((root/name).read_text())
def canonical_digest(value):return hashlib.sha256(json.dumps(value,sort_keys=True,separators=(',',':'),ensure_ascii=False,allow_nan=False).encode()).hexdigest()
manifest=read('manifest.json')
for name,pin in manifest['files'].items():

 raw=path.read_bytes();assert len(raw)==pin['size_bytes'] and hashlib.sha256(raw).hexdigest()==pin['sha256']

assert receipt['continuation_request_sha256']==hashlib.sha256((root/'continuation-request.json').read_bytes()).hexdigest()
assert continuation['previous_failure_sha256']==hashlib.sha256((root/'failure.json').read_bytes()).hexdigest()

request=read(prefix+'inspection-request.json');source=read(prefix+'source.json');plan=read(prefix+'inspection-plan.json');policy=read(prefix+'inspection-policy.json');result=read(prefix+'inspection-result.json')
