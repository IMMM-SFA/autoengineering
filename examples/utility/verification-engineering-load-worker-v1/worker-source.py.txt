"""Fixed coupled-load packet and process-output boundary, under development.

Only the supplied coupled scoring kernel can execute. Callers must bind the raw
packet hash before dispatch. Paths are trusted runtime arguments, never packet
fields. This module does not open sealed sources, authorize access, enforce a
wall deadline, admit a container, or register an execution/cost receipt.
"""

from __future__ import annotations

import hashlib
import io
import json
import math
import os
from pathlib import Path
import re
import signal
import stat
import sys
import tempfile

PACKET_BYTES = 32768
OUTPUT_BYTES = 16384
COUNTERS = ("solver_iterations", "component_calls", "derivative_calls")
MODES = ("score", "output-small", "output-overrun")


def _unique(pairs):
    result = {}
    for key, value in pairs:
        if key in result:
            raise ValueError("duplicate worker field")
        result[key] = value
    return result


def _json(raw):
    def nonfinite(_):
        raise ValueError("nonfinite worker JSON")

    return json.loads(raw, object_pairs_hook=_unique, parse_constant=nonfinite)


def _hash(value):
    return type(value) is str and re.fullmatch("[0-9a-f]{64}", value) is not None


def _number(value):
    return type(value) in (int, float) and math.isfinite(value)


def parse_packet(raw, *, expected_sha256):
    """Validate every field before imports, filesystem writes or model creation."""
    if (
        type(raw) is not bytes
        or not 0 < len(raw) <= PACKET_BYTES
        or not _hash(expected_sha256)
        or hashlib.sha256(raw).hexdigest() != expected_sha256
    ):
        raise ValueError("coupled packet bytes differ")
    packet = _json(raw.decode("utf-8"))
    fields = {
        "schema",
        "execution_digest",
        "source_binding_digest",
        "model_digest",
        "kernel_digest",
        "source_current_A",
        "minimum_node2_voltage_V",
        "cases",
    }
    if type(packet) is not dict or set(packet) != fields:
        raise ValueError("coupled packet fields differ")
    if packet["schema"] != "utility-coupled-load-packet-v1" or any(
        not _hash(packet[name])
        for name in ("execution_digest", "source_binding_digest", "model_digest", "kernel_digest")
    ):
        raise ValueError("coupled packet identity differs")
    current, voltage = packet["source_current_A"], packet["minimum_node2_voltage_V"]
    if not _number(current) or not 0.01 <= current <= 0.1 or not _number(voltage):
        raise ValueError("coupled packet policy differs")
    cases = packet["cases"]
    if type(cases) is not list or len(cases) != 24:
        raise ValueError("coupled packet needs 24 cases")
    seen = set()
    for row in cases:
        if type(row) is not dict or set(row) != {"scenario_id", "forcing", "weight"}:
            raise ValueError("coupled packet case differs")
        name, forcing, weight = row["scenario_id"], row["forcing"], row["weight"]
        if (
            type(name) is not str
            or re.fullmatch("[a-z][a-z0-9_-]{0,95}", name) is None
            or name in seen
            or not _number(weight)
            or weight != 1 / 24
            or type(forcing) is not dict
            or set(forcing) != {"ground_voltage", "resistor1_ohm", "resistor2_ohm"}
            or not all(_number(v) for v in forcing.values())
            or forcing["resistor1_ohm"] <= 0
            or forcing["resistor2_ohm"] <= 0
        ):
            raise ValueError("coupled packet forcing differs")
        seen.add(name)
    return packet


def _public_result(score, receipt, *, failed):
    """Export an allowlist, never raw receipts, diagnostics or exceptions."""
    expected = ("model_failure",) if failed else ("success", "scientific_infeasible")
    if score.status not in expected or score.outcome != "mean_source_power" or score.units != "W":
        raise ValueError("coupled score contract differs")
    if score.status == "success":
        if score.feasible is not True or not _number(score.value) or score.value < 0:
            raise ValueError("coupled successful score differs")
    elif score.value is not None or score.feasible is not (None if failed else False):
        raise ValueError("coupled failure score differs")
    attempted, completed = (
        receipt[n] for n in ("attempted_model_starts", "completed_model_starts")
    )
    counters = receipt["known_completed_inner_work" if failed else "inner_work"]
    if (
        type(attempted) is not int
        or type(completed) is not int
        or not 0 <= completed <= attempted <= 24
        or (not failed and (attempted, completed) != (24, 24))
        or (failed and attempted != completed + 1)
        or type(counters) is not dict
        or set(counters) != set(COUNTERS)
        or any(type(v) is not int or v < 0 for v in counters.values())
    ):
        raise ValueError("coupled work receipt differs")
    return {
        "schema": "utility-coupled-load-worker-result-v1",
        "score": {
            "status": score.status,
            "outcome": score.outcome,
            "units": score.units,
            "value": score.value,
            "feasible": score.feasible,
        },
        "attempted_model_starts": attempted,
        "completed_model_starts": completed,
        "known_completed_inner_work": dict(counters),
        "last_attempt_inner_work_unknown": failed,
        "automatic_retries": 0,
        "private_boundary_qualified": False,
    }


def score_packet(raw, *, expected_sha256, source_root, private_work):
    """Run the fixed kernel; source and work roots belong to the trusted runtime."""
    packet = parse_packet(raw, expected_sha256=expected_sha256)
    # The packet cannot name an entrypoint or alter executable source paths.
    from examples.utility import engineering_load_scoring as kernel

    policy = kernel.LoadScorePolicy(
        "utility-coupled-load-score-v1",
        packet["model_digest"],
        packet["kernel_digest"],
        packet["minimum_node2_voltage_V"],
    )
    score = kernel.score_loads(
        packet["source_current_A"],
        packet["cases"],
        private_work,
        source_root,
        policy=policy,
        execution_digest=packet["execution_digest"],
    )
    failed = score.status == "model_failure"
    name = "failed.json" if failed else "complete.json"
    descriptor = os.open(Path(private_work) / name, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK)
    try:
        info = os.fstat(descriptor)
        if not stat.S_ISREG(info.st_mode) or not 0 < info.st_size <= OUTPUT_BYTES:
            raise ValueError("coupled receipt size or type differs")
        receipt = _json(os.read(descriptor, OUTPUT_BYTES + 1))
    finally:
        os.close(descriptor)
    if receipt["execution_digest"] != packet["execution_digest"]:
        raise ValueError("coupled execution receipt differs")
    return _public_result(score, receipt, failed=failed)


def capture(raw, *, expected_sha256, source_root, private_work, mode="score"):
    """Hard-cap shared child stdout/stderr; return no child bytes on any failure.

    The trusted outer runtime must enforce the aggregate deadline and terminate
    the whole container on cancellation/owner loss. There is no retry here.
    """
    if mode not in MODES:
        raise ValueError("unsupported coupled worker mode")
    if mode == "score":
        parse_packet(raw, expected_sha256=expected_sha256)
    with tempfile.TemporaryFile() as output:
        pid = os.fork()
        if pid == 0:
            try:
                import resource

                resource.setrlimit(resource.RLIMIT_CORE, (0, 0))
                resource.setrlimit(resource.RLIMIT_FSIZE, (OUTPUT_BYTES, OUTPUT_BYTES))
                signal.signal(signal.SIGXFSZ, signal.SIG_DFL)
                os.dup2(output.fileno(), 1)
                os.dup2(output.fileno(), 2)
                # Keep inherited Python output objects alive until os._exit.
                _previous_output = (sys.stdout, sys.stderr)
                sys.stdout = io.TextIOWrapper(io.FileIO(1, "w", closefd=False), write_through=True)
                sys.stderr = io.TextIOWrapper(io.FileIO(2, "w", closefd=False), write_through=True)
                null = os.open("/dev/null", os.O_RDONLY)
                os.dup2(null, 0)
                maximum = resource.getrlimit(resource.RLIMIT_NOFILE)[0]
                if not 3 <= maximum <= 1048576:
                    os._exit(70)
                os.closerange(3, maximum)
                os.environ.clear()
                if mode == "output-small":
                    os.write(1, b"x" * 4096)
                elif mode == "output-overrun":
                    os.write(1, b"x" * OUTPUT_BYTES)
                    os.write(2, b"y")
                else:
                    result = score_packet(
                        raw,
                        expected_sha256=expected_sha256,
                        source_root=source_root,
                        private_work=private_work,
                    )
                    print(json.dumps(result, allow_nan=False, separators=(",", ":")), flush=True)
                os._exit(0)
            except BaseException:
                os._exit(70)
        _, status = os.waitpid(pid, 0)
        output.seek(0)
        captured = output.read(OUTPUT_BYTES + 1)
    signum = os.WTERMSIG(status) if os.WIFSIGNALED(status) else None
    code = os.WEXITSTATUS(status) if os.WIFEXITED(status) else None
    limited = len(captured) >= OUTPUT_BYTES or signum == signal.SIGXFSZ
    result = None
    if mode == "score" and code == 0 and not limited:
        try:
            value = _json(captured.decode("utf-8"))
            # Revalidate the allowlist instead of exporting arbitrary child JSON.
            from types import SimpleNamespace

            score = SimpleNamespace(**value["score"])
            failed = score.status == "model_failure"
            receipt = dict(value)
            receipt["inner_work"] = value["known_completed_inner_work"]
            checked = _public_result(score, receipt, failed=failed)
            if json.dumps(checked, sort_keys=True) == json.dumps(value, sort_keys=True):
                result = checked
        except Exception:
            pass
    return {
        "schema": "utility-coupled-load-output-v1",
        "mode": mode,
        "output_limit_bytes": OUTPUT_BYTES,
        "captured_bytes": len(captured),
        "child_signal": signum,
        "child_exit_code": code,
        "output_limited": limited,
        "result": result,
        "private_boundary_qualified": False,
    }
