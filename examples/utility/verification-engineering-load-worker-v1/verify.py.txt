"""Check copied authored worker receipts without importing or running models."""
from pathlib import Path
import hashlib
import json
import math
import sys

root = Path(sys.argv[1]).resolve(strict=True)
raw = (root / 'manifest.json').read_bytes()
assert hashlib.sha256(raw).hexdigest() == sys.argv[2], 'manifest differs'
manifest = json.loads(raw)
assert manifest['schema'] == 'utility-public-worker-proof-v1'
for name, expected in manifest['files'].items():
    assert Path(name).name == name, 'unsafe proof path'
    path = root / name
    assert not path.is_symlink() and path.is_file(), 'nonregular proof file'
    data = path.read_bytes()
    assert len(data) == expected['bytes'], 'proof size differs'
    assert hashlib.sha256(data).hexdigest() == expected['sha256'], 'proof bytes differ'

def read(name):
    assert name in manifest['files']
    return json.loads((root / name).read_bytes())

complete = read('complete.json')
output = read('public-worker-output.json')
public = output['result']
assert output['child_exit_code'] == 0 and output['child_signal'] is None
assert not output['output_limited'] and output['captured_bytes'] < output['output_limit_bytes']
assert output['private_boundary_qualified'] is False
assert public['private_boundary_qualified'] is False
assert complete['private_boundary_qualified'] is False and complete['task_admitted'] is False
assert public['score'] == complete['score']
assert public['score']['status'] == 'success' and public['score']['feasible'] is True
assert public['attempted_model_starts'] == public['completed_model_starts'] == 24
assert public['known_completed_inner_work'] == complete['inner_work']
assert public['last_attempt_inner_work_unknown'] is False
rows = [read(f'{i:02d}-result.json') for i in range(24)]
assert all(r['feasible'] is True for r in rows)
mean = math.fsum(r['source_power_W'] / 24 for r in rows)
assert mean == complete['mean_source_power_W'] == public['score']['value']
for name, count in public['known_completed_inner_work'].items():
    assert sum(r['diagnostics'][name] for r in rows) == count
summary = read('summary.json')
assert summary['actual_model_starts_this_increment'] == 48
assert summary['container_starts_this_increment'] == 0
assert summary['private_boundary_qualified'] is False
print(json.dumps({'verified': True, 'archived_model_starts': 24,
                  'increment_model_starts': 48, 'mean_source_power_W': mean,
                  'private_boundary_qualified': False}, sort_keys=True))
