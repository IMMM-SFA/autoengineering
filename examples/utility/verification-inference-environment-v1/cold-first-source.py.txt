"""Read the pinned inference input package without model/acquisition execution."""
import importlib.abc
import json
from pathlib import Path
import socket
import subprocess
import sys
import zipfile

class NoModels(importlib.abc.MetaPathFinder):
    def find_spec(self, fullname, *args):
        if fullname.startswith('examples.') and fullname not in {
            'examples.utility', 'examples.utility.inference_environment',
        }:
            raise AssertionError('model import attempted')
        if fullname.split('.')[0] in {'torch', 'botorch', 'pvlib', 'openmdao', 'py_wake'}:
            raise AssertionError('optional runtime import attempted')

sys.meta_path.insert(0, NoModels())
from examples.utility.inference_environment import verify_inputs
import numpy

def forbidden(*args, **kwargs):
    raise AssertionError('acquisition, extraction, execution or array load attempted')

socket.socket = subprocess.Popen = zipfile.ZipFile.extractall = numpy.load = forbidden
root = Path(sys.argv[1]).absolute()
plan = verify_inputs(root, expected_digest=sys.argv[2])
assert plan.starts_authorized is False and plan.runtime_qualified is False
print(json.dumps({
    'result': 'cold pinned input verification passed',
    'inputs_digest': plan.digest,
    'archives': len(plan.archives),
    'compressed_archive_bytes': plan.compressed_archive_bytes,
    'wheel_python_sources': len(plan.wheel_python_sources),
    'starts_authorized': False,
    'runtime_qualified': False,
}, sort_keys=True))
