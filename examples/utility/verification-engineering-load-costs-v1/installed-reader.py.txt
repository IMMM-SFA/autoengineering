"""Installed public cost reconstruction with private/model/process access denied."""
import sys
from pathlib import Path
site, root, evidence = (Path(p).resolve() for p in sys.argv[1:4])
sys.path.insert(0, str(site))
import importlib.abc
class NoModels(importlib.abc.MetaPathFinder):
    def find_spec(self, fullname, *args):
        if fullname.split('.')[0] in {'openmdao', 'pvlib', 'torch', 'py_wake', 'topfarm', 'h5py', 'bmipy'}:
            raise RuntimeError('optional model imports denied')
sys.meta_path.insert(0, NoModels())
import hashlib, json, os, socket, subprocess, time
from autoengineering.campaign.serialization import digest
from autoengineering.campaign.records import ArtifactRef
from autoengineering.utility.store import SuiteStore
from autoengineering.utility.scoring import ConfirmationGrant
from autoengineering.utility.controller import UtilityController
from autoengineering.agents.batch import BatchSpec, BatchStore
from autoengineering.utility.baseline_selection import follower_group
from autoengineering.utility.baseline_execution import frozen_baseline_allocations
from autoengineering.utility.costs import join_confirmation_costs, reconcile
from examples.utility.engineering_load_costs import LoadCostRegistration, join_load_costs
from examples.utility import engineering_load_scoring as kernel
from examples.utility import engineering_scenarios as source
expected = json.loads((evidence / 'expected-costs.json').read_text())
pins = json.loads((evidence / 'input-pins.json').read_text())
inputs = json.loads((root / 'inputs.json').read_text())
assert digest(inputs) == expected['inputs_digest']
grant = ConfirmationGrant.from_dict(inputs['grant'])
registration = LoadCostRegistration.from_dict(inputs['registration'])
work_ref = ArtifactRef.from_dict(inputs['work_artifact'])
for name, module in tuple(sys.modules.items()):
    if name == 'autoengineering' or name.startswith(('autoengineering.', 'examples.')):
        origin = getattr(module, '__file__', None)
        if origin is not None:
            assert Path(origin).resolve().is_relative_to(site), name
inodes = {(p.stat().st_dev, p.stat().st_ino) for p in root.rglob('*') if p.is_file()}
def denied(*a, **kw):
    raise RuntimeError('private source model entropy process network or clock denied')
kernel.run_lifecycle = source.read_private = source.derive = source.secrets.token_bytes = denied
subprocess.Popen = socket.socket = time.time = time.monotonic = denied

def audit(event, args):
    if event == 'open':
        path, mode, flags = args
        if flags & (os.O_WRONLY | os.O_RDWR | os.O_CREAT | os.O_TRUNC | os.O_APPEND):
            raise RuntimeError('write denied')
        if isinstance(path, int):
            st = os.fstat(path)
            if (st.st_dev, st.st_ino) not in inodes:
                raise RuntimeError('outside descriptor denied')
        elif not Path(os.fsdecode(path)).absolute().is_relative_to(root):
            raise RuntimeError('outside public scope denied')
    if event == 'import' or event.startswith(('socket.', 'subprocess.', 'os.exec', 'os.fork', 'os.posix_spawn')):
        raise RuntimeError('new import or execution denied')
sys.addaudithook(audit)
for name, sha in pins.items():
    assert hashlib.sha256((root / name).read_bytes()).hexdigest() == sha
for name, sha in inputs['native_public_files'].items():
    assert hashlib.sha256((root / 'native' / name).read_bytes()).hexdigest() == sha
store = SuiteStore(root / 'native/suite')
assert store.spec.digest == inputs['suite_digest']
before = store.replay()['head']
records = join_load_costs(store, grant, registration, work_artifact=work_ref)
again = join_load_costs(store, grant, registration, work_artifact=work_ref)
assert again == records and store.replay()['head'] == before
combined = (*join_confirmation_costs(store), *records)
assert [r.to_dict() for r in records] == expected['load_costs']
assert [r.to_dict() for r in combined] == expected['joined_confirmation_costs']
assert reconcile(combined, tuple(r.receipt_id for r in combined)) == expected['reconciliation']
assert not expected['reconciliation']['complete'] and not expected['full_cost_evidence']
unknown = join_load_costs(store, grant, registration)
assert all(r.quantity is None for r in unknown if 'reservations' not in r.resource)
try:
    reconcile((*records, *again), tuple(r.receipt_id for r in records))
except ValueError:
    duplicate_refused = True
else:
    raise AssertionError('duplicate work accepted')
try:
    (root.parent / 'reserved-canary.txt').read_bytes()
except RuntimeError as error:
    assert str(error) == 'outside public scope denied'
else:
    raise AssertionError('outside read accepted')
assert 'openmdao' not in sys.modules
print(json.dumps({'installed_target_used': True, 'editable_source_fallback': False,
    'saved_native_work_reconstructed': True, 'native_cost_records': len(records),
    'completed_evaluations': 24, 'nonlinear_iterations': 395,
    'component_calls': 6440, 'component_linearize_calls': 1975,
    'unknown_without_work_proof': True, 'unknown_generic_process_cost_preserved': True,
    'repeat_equal_without_mutation': True, 'duplicate_work_refused': duplicate_refused,
    'outside_public_scope_refused': True, 'new_models': 0, 'new_source_reads': 0,
    'new_processes_after_initialization': 0, 'network': 0, 'task_admitted': False,
    'full_cost_evidence': False}))
