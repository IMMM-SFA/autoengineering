"""Build copied inputs using existing setuptools and install without dependencies."""
from pathlib import Path
import hashlib, json, os, shutil, subprocess, sys, zipfile

root = Path.cwd()
out = root / 'examples/utility/verification-engineering-load-costs-v1'
stage = Path('/private/tmp/utility-load-costs-install-v1')
stage.mkdir(mode=0o700)
build, target, wheel_root = (stage / name for name in ('source', 'installed', 'wheel'))
for p in (build, target, wheel_root):
    p.mkdir(mode=0o700)
old_inputs = json.loads((root / 'examples/utility/verification-solar-csv-installed-v1/input-pins.json').read_text())
names = {n for n in old_inputs if not n.startswith('examples/')}
names.update({
 'examples/utility/__init__.py', 'examples/utility/engineering_load_costs.py',
 'examples/utility/engineering_load_scoring.py', 'examples/utility/engineering_load.py',
 'examples/utility/engineering.py', 'examples/utility/engineering_source_binding.py',
 'examples/utility/engineering_scenarios.py', 'examples/interoperability/circuit/model.py',
 'examples/interoperability/circuit/fixture.py',
})
pins = {n: hashlib.sha256((root / n).read_bytes()).hexdigest() for n in sorted(names)}
for name in names:
    dest = build / name
    dest.parent.mkdir(mode=0o700, parents=True, exist_ok=True)
    shutil.copyfile(root / name, dest)
(out / 'wheel-input-pins.json').write_text(json.dumps(pins, indent=2) + '\n')
env = os.environ.copy(); env['SOURCE_DATE_EPOCH'] = '1793318400'
command = [sys.executable, '-I', '-B', '-c', "from setuptools.build_meta import build_wheel; build_wheel('../wheel')"]
r = subprocess.run(command, cwd=build, env=env, capture_output=True, text=True)
(out / 'build.stdout.txt').write_text(r.stdout); (out / 'build.stderr.txt').write_text(r.stderr)
assert r.returncode == 0
wheel = next(wheel_root.glob('*.whl'))
with zipfile.ZipFile(wheel) as z:
    python_files = [n for n in z.namelist() if n.endswith('.py')]
    for name in z.namelist():
        if '.dist-info/' in name:
            continue
        source = 'src/' + name if name.startswith('autoengineering/') else name
        assert source in pins and hashlib.sha256(z.read(name)).hexdigest() == pins[source]
    assert len(python_files) == 233
saved = out / wheel.name; shutil.copyfile(wheel, saved)
install = [sys.executable, '-I', '-m', 'pip', 'install', '--no-index', '--no-deps', '--no-compile',
           '--target', str(target), str(wheel)]
r2 = subprocess.run(install, cwd=stage, capture_output=True, text=True)
(out / 'install.stdout.txt').write_text(r2.stdout); (out / 'install.stderr.txt').write_text(r2.stderr)
assert r2.returncode == 0
inputs = stage / 'inputs'; shutil.copytree(out / 'public', inputs)
result = {'wheel_sha256': hashlib.sha256(saved.read_bytes()).hexdigest(), 'python_files': len(python_files),
          'core_files': 224, 'example_files': 9, 'copied_inputs': len(pins),
          'all_package_payloads_match_inputs': True, 'build_exit_code': r.returncode,
          'install_exit_code': r2.returncode, 'target': str(target), 'input_scope': str(inputs),
          'network_or_dependency_installation': False, 'inherited_native_dependencies': True,
          'release_or_linux_qualification': False}
(out / 'build-install-result.json').write_text(json.dumps(result, indent=2) + '\n')
print(json.dumps(result))
