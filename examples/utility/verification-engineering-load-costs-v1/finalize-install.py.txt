"""Finish metadata input copy in the already built and installed target."""
from pathlib import Path
import hashlib, json, shutil, zipfile
root = Path.cwd()
out = root / 'examples/utility/verification-engineering-load-costs-v1'
stage = Path('/private/tmp/utility-load-costs-install-v1')
wheel = next((stage / 'wheel').glob('*.whl'))
saved = out / wheel.name
assert saved.read_bytes() == wheel.read_bytes()
pins = json.loads((out / 'wheel-input-pins.json').read_text())
with zipfile.ZipFile(wheel) as z:
    python_files = [n for n in z.namelist() if n.endswith('.py')]
    for name in z.namelist():
        if '.dist-info/' in name:
            continue
        source = 'src/' + name if name.startswith('autoengineering/') else name
        assert source in pins and hashlib.sha256(z.read(name)).hexdigest() == pins[source]
        assert (stage / 'installed' / name).read_bytes() == z.read(name)
    assert len(python_files) == 233
inputs = stage / 'inputs'; shutil.copytree(out / 'public', inputs)
result = {'wheel_sha256': hashlib.sha256(saved.read_bytes()).hexdigest(), 'python_files': len(python_files),
          'core_files': 224, 'example_files': 9, 'copied_inputs': len(pins),
          'all_package_payloads_match_inputs': True, 'installed_payloads_match_wheel': True,
          'build_exit_code': 0, 'install_exit_code': 0, 'target': str(stage / 'installed'),
          'input_scope': str(inputs), 'network_or_dependency_installation': False,
          'inherited_native_dependencies': True, 'release_or_linux_qualification': False,
          'builds': 1, 'installs': 1, 'first_wrapper_failure': 'initial-refusal.json'}
(out / 'build-install-result.json').write_text(json.dumps(result, indent=2) + '\n')
print(json.dumps(result))
