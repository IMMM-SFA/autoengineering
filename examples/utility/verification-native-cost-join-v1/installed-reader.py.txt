import hashlib,importlib.util,json,socket,subprocess,sys
from pathlib import Path
root=Path(__file__).parent
manifest=json.loads((root/"fixtures.json").read_bytes())
controls=[lambda:Path('/Users/cam/projects/autoengineering/.worktrees/agentic-architecture/README.md').read_bytes(),lambda:socket.socket(),lambda:subprocess.run([sys.executable,"-c","pass"])]
controls += [lambda p=v["original_root"]: (Path(p)/"batch.json").read_bytes() for v in manifest.values()]
for action in controls:
 try: action()
 except RuntimeError as error: assert "offline guard denied" in str(error)
 else: raise AssertionError("denial control did not run")

 batch=BatchStore(fixture)
 values=join_native_costs(batch,{"first":"1"*64,"second":"2"*64})

 records[name]=[r.to_dict() for r in values]
